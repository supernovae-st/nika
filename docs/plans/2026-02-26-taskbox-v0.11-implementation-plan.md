fn render_compact(&self, area: Rect, buf: &mut Buffer) {
    let verb_color = VerbColor::Exec.rgb();
    let status_badge = match (&self.exit_code, &self.state) {
        (Some(0), _) => Span::styled(" ✓ ", Style::default()
            .fg(Color::Green).add_modifier(Modifier::BOLD)),
        (Some(code), _) => Span::styled(format!(" ✗ {} ", code),
            Style::default().fg(exit::code_color(*code))),
        (None, BoxState::Running { .. }) => Span::styled(" ⋯ ",
            Style::default().fg(verb_color)),
        _ => Span::styled(" ⋯ ", Style::default().fg(Color::Gray)),
    };

    let command = Self::truncate(&self.command, 60);
    let line = Line::from(vec![
        Span::styled("📟 ", Style::default().fg(verb_color)),
        Span::styled(command, Style::default().fg(Color::White)),
        status_badge,
    ]);
    buf.set_line(area.x, area.y, &line, area.width);
}
```

2. **Update Widget::render() to dispatch by mode:**

        RenderMode::Compact => self.render_compact(area, buf),
        RenderMode::Expanded => self.render_expanded(area, buf),
        RenderMode::Full => self.render_full(area, buf),

}

fn render_response_preview(&self, body: &str, max_lines: usize) -> Vec<Line<'static>> {
    let mut lines = Vec::new();

    // Try to parse as JSON for pretty printing
    if let Ok(json) = serde_json::from_str::<serde_json::Value>(body) {
        let pretty = serde_json::to_string_pretty(&json)
            .unwrap_or_else(|_| body.to_string());

        for line in pretty.lines().take(max_lines) {
            lines.push(Line::from(
                Span::styled(format!("  {}", line),
                    Style::default().fg(Color::Gray))));
        }

        if pretty.lines().count() > max_lines {
            lines.push(Line::from(
                Span::styled(format!("  ... ({} more lines)",
                    pretty.lines().count() - max_lines),
                    Style::default().fg(Color::DarkGray))));
        }
    } else {
        // Plain text
        for line in body.lines().take(max_lines) {
            lines.push(Line::from(
                Span::styled(format!("  {}", line),
                    Style::default().fg(Color::Gray))));
        }
    }

    lines
}
```

impl InferBox {
    /// Render compact mode (1-line summary)
    pub fn render_compact(&self, area: Rect, buf: &mut Buffer) {
        let block = Block::default()
            .borders(Borders::ALL)
            .border_style(Style::default().fg(VerbColor::Infer.rgb()))
            .title(self.compact_title());

        let inner = block.inner(area);
        block.render(area, buf);

        // Content: truncated prompt + token count
        let prompt_max = inner.width.saturating_sub(15) as usize;
        let truncated = truncate_with_ellipsis(&self.prompt, prompt_max);
        let tokens = format!("{} tokens", self.output_tokens.unwrap_or(0));

        let line = Line::from(vec![
            Span::styled(format!("\"{}\"", truncated), Style::default().fg(Color::White)),
            Span::raw(" "),
            Span::styled(tokens, Style::default().fg(Color::DarkGray)),
        ]);
        line.render(inner, buf);
    }

    fn compact_title(&self) -> Line<'static> {
        let icon = VerbColor::Infer.icon_label();
        let status = self.state.icon();
        let duration = self.duration.map(|d| format!(" {:.1}s", d.as_secs_f32()))
            .unwrap_or_default();

        Line::from(vec![
            Span::styled(format!(" {} ", icon), Style::default().fg(VerbColor::Infer.rgb())),
            Span::raw("─".repeat(40)),
            Span::styled(format!(" {} ", status), self.state.style()),
            Span::styled(duration, Style::default().fg(Color::DarkGray)),
        ])
    }
}

/// Task temperature category
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum TaskTemperature {
    Hot,    // 🔥 Currently executing
    Warm,   // 🌡️ Ready to execute (no pending deps)
    Queued, // 📋 Waiting on dependencies
}

impl TaskTemperature {
    pub fn icon(&self) -> &'static str {
        match self {
            Self::Hot => "🔥",
            Self::Warm => "🌡️",
            Self::Queued => "📋",
        }
    }

    pub fn label(&self) -> &'static str {
        match self {
            Self::Hot => "HOT (currently executing)",
            Self::Warm => "WARM (ready to execute)",
            Self::Queued => "QUEUED (dependencies pending)",
        }
    }

    pub fn color(&self) -> Color {
        match self {
            Self::Hot => Color::Rgb(239, 68, 68),      // Red 500
            Self::Warm => Color::Rgb(245, 158, 11),   // Amber 500
            Self::Queued => Color::Rgb(100, 116, 139), // Slate 500
        }
    }
}

impl ChatTaskQueue {
    /// Categorize task by temperature
    pub fn categorize_task(&self, task: &QueuedTask) -> TaskTemperature {
        match task.state {
            ChatTaskState::Running => TaskTemperature::Hot,
            ChatTaskState::Pending if task.dependencies.is_empty() => TaskTemperature::Warm,
            ChatTaskState::Pending if self.deps_satisfied(task) => TaskTemperature::Warm,
            ChatTaskState::Pending => TaskTemperature::Queued,
            _ => TaskTemperature::Queued, // Complete/Failed don't show in queue
        }
    }

    /// Render categorized task list
    pub fn render_categorized(&self, area: Rect, buf: &mut Buffer) {
        let hot: Vec<_> = self.tasks.iter()
            .filter(|t| self.categorize_task(t) == TaskTemperature::Hot)
            .collect();
        let warm: Vec<_> = self.tasks.iter()
            .filter(|t| self.categorize_task(t) == TaskTemperature::Warm)
            .collect();
        let queued: Vec<_> = self.tasks.iter()
            .filter(|t| self.categorize_task(t) == TaskTemperature::Queued)
            .collect();

        let mut y = area.y;

        // Render each category
        for (temp, tasks) in [
            (TaskTemperature::Hot, hot),
            (TaskTemperature::Warm, warm),
            (TaskTemperature::Queued, queued),
        ] {
            if !tasks.is_empty() {
                // Category header
                let header = Line::from(vec![
                    Span::styled(
                        format!("  {} {} ", temp.icon(), temp.label()),
                        Style::default().fg(temp.color()).add_modifier(Modifier::BOLD),
                    ),
                ]);
                header.render(Rect::new(area.x, y, area.width, 1), buf);
                y += 1;

                // Tasks in category
                for task in tasks {
                    self.render_task_line(task, Rect::new(area.x, y, area.width, 1), buf);
                    y += 1;
                }
                y += 1; // Spacing between categories
            }
        }
    }

    /// Render single task with progress or deps
    fn render_task_line(&self, task: &QueuedTask, area: Rect, buf: &mut Buffer) {
        let verb_icon = task.verb.icon();
        let description = truncate_with_ellipsis(&task.description, 30);

        let suffix = if task.state == ChatTaskState::Running {
            // Progress bar for running tasks
            let progress = task.progress.unwrap_or(0.0);
            format!("[{}] {}%", progress_bar(progress, 6), (progress * 100.0) as u8)
        } else if !task.dependencies.is_empty() {
            // Dependency list for queued tasks
            format!("deps: {:?}", task.dependencies)
        } else {
            "⏳ waiting".to_string()
        };

        let line = Line::from(vec![
            Span::raw("  ├─ "),
            Span::styled(verb_icon, Style::default().fg(task.verb.color())),
            Span::raw(format!(" {}: ", task.verb.label().to_lowercase())),
            Span::styled(description, Style::default().fg(Color::White)),
            Span::raw(" "),
            Span::styled(suffix, Style::default().fg(Color::DarkGray)),
        ]);
        line.render(area, buf);
    }
}

/// Generate progress bar string
fn progress_bar(progress: f32, width: usize) -> String {
    let filled = (progress * width as f32).round() as usize;
    let empty = width.saturating_sub(filled);
    format!("{}{}", "▓".repeat(filled), "░".repeat(empty))
}

impl ChatView {
    /// Handle incoming events from the runtime
    pub fn handle_event(&mut self, event: &EventLog) {
        match &event.kind {
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // WORKFLOW LIFECYCLE
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::WorkflowStarted { workflow_id, .. } => {
                self.workflow_id = Some(workflow_id.clone());
                self.workflow_state = WorkflowState::Running;
                self.task_boxes.clear();
                self.agent_hierarchy.clear();
            }

            EventKind::WorkflowCompleted { duration, .. } => {
                self.workflow_state = WorkflowState::Completed {
                    duration: *duration,
                };
                // Stop all animations
                for task_box in self.task_boxes.values_mut() {
                    task_box.stop_animation();
                }
            }

            EventKind::WorkflowFailed { error, .. } => {
                self.workflow_state = WorkflowState::Failed {
                    error: error.clone(),
                };
            }

            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // TASK LIFECYCLE
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::TaskScheduled { task_id } => {
                // Create task box in Queued state
                let task_box = TaskBox::new_queued(task_id.clone());
                self.task_boxes.insert(task_id.clone(), task_box);
            }

            EventKind::TaskStarted { task_id, action } => {
                if let Some(task_box) = self.task_boxes.get_mut(task_id) {
                    task_box.set_state(BoxState::Running {
                        start: Instant::now(),
                        spinner_frame: 0,
                    });
                    task_box.set_verb(action.verb_color());
                    task_box.start_animation(&self.animation_ticker);
                } else {
                    // Create if not scheduled (direct start)
                    let mut task_box = TaskBox::new_running(task_id.clone(), action);
                    task_box.start_animation(&self.animation_ticker);
                    self.task_boxes.insert(task_id.clone(), task_box);
                }
            }

            EventKind::TaskCompleted { task_id, result } => {
                if let Some(task_box) = self.task_boxes.get_mut(task_id) {
                    task_box.set_state(BoxState::Success {
                        duration: result.duration,
                    });
                    task_box.stop_animation();
                    task_box.set_result(Some(result.clone()));
                }
            }

            EventKind::TaskFailed { task_id, error, duration } => {
                if let Some(task_box) = self.task_boxes.get_mut(task_id) {
                    task_box.set_state(BoxState::Failed {
                        error: error.to_string(),
                        duration: *duration,
                    });
                    task_box.stop_animation();
                }
            }

            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // INFER VERB EVENTS
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::ProviderCalled { task_id, model, .. } => {
                if let Some(TaskBox::Infer(infer_box)) = self.task_boxes.get_mut(task_id) {
                    infer_box.set_model(model.clone());
                    infer_box.set_streaming_cursor(true);
                }
            }

            EventKind::ProviderResponded { task_id, response, tokens, thinking } => {
                if let Some(TaskBox::Infer(infer_box)) = self.task_boxes.get_mut(task_id) {
                    infer_box.append_response(response);
                    if let Some((input, output)) = tokens {
                        infer_box.set_tokens(*input, *output);
                    }
                    if let Some(thinking_text) = thinking {
                        infer_box.set_thinking(thinking_text.clone());
                    }
                    infer_box.set_streaming_cursor(false);
                }
            }

            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // EXEC VERB EVENTS
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::ExecStarted { task_id, command, pid, cwd } => {
                if let Some(TaskBox::Exec(exec_box)) = self.task_boxes.get_mut(task_id) {
                    exec_box.set_command(command.clone());
                    exec_box.set_pid(*pid);
                    if let Some(dir) = cwd {
                        exec_box.set_cwd(dir.clone());
                    }
                }
            }

            EventKind::ExecStdout { task_id, chunk } => {
                if let Some(TaskBox::Exec(exec_box)) = self.task_boxes.get_mut(task_id) {
                    exec_box.append_stdout(chunk);
                }
            }

            EventKind::ExecStderr { task_id, chunk } => {
                if let Some(TaskBox::Exec(exec_box)) = self.task_boxes.get_mut(task_id) {
                    exec_box.append_stderr(chunk);
                }
            }

            EventKind::ExecCompleted { task_id, exit_code, .. } => {
                if let Some(TaskBox::Exec(exec_box)) = self.task_boxes.get_mut(task_id) {
                    exec_box.set_exit_code(*exit_code);
                }
            }

            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // AGENT VERB EVENTS
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::AgentStart { task_id, prompt, max_turns, .. } => {
                if let Some(TaskBox::Agent(agent_box)) = self.task_boxes.get_mut(task_id) {
                    agent_box.set_prompt(prompt.clone());
                    agent_box.set_max_turns(*max_turns);
                }
            }

            EventKind::AgentTurn { task_id, turn, response, metadata, .. } => {
                if let Some(TaskBox::Agent(agent_box)) = self.task_boxes.get_mut(task_id) {
                    agent_box.set_turn(*turn);
                    agent_box.append_response(response);

                    if let Some(meta) = metadata {
                        // Calculate velocity from token output and time
                        let elapsed_ms = self.last_turn_timestamp
                            .map(|t| t.elapsed().as_millis() as u64)
                            .unwrap_or(1000);
                        agent_box.process_agent_turn(meta, elapsed_ms);

                        // Update thinking capture
                        if let Some(ref thinking) = meta.thinking {
                            agent_box.set_thinking(thinking.clone());
                        }
                    }
                    self.last_turn_timestamp = Some(Instant::now());
                }
            }

            EventKind::AgentComplete { task_id, final_response, total_turns, .. } => {
                if let Some(TaskBox::Agent(agent_box)) = self.task_boxes.get_mut(task_id) {
                    agent_box.set_final_response(final_response.clone());
                    agent_box.set_total_turns(*total_turns);
                }
            }

            EventKind::AgentSpawned { parent_task_id, child_task_id, depth, prompt } => {
                // Track parent-child hierarchy
                self.agent_hierarchy
                    .entry(parent_task_id.clone())
                    .or_insert_with(Vec::new)
                    .push(child_task_id.clone());

                // Create child agent box
                let child_box = AgentBox::new_spawned(
                    child_task_id.clone(),
                    prompt.clone(),
                    *depth,
                    parent_task_id.clone(),
                );
                self.task_boxes.insert(
                    child_task_id.clone(),
                    TaskBox::Agent(child_box),
                );
            }

            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // MCP EVENTS (invoke verb)
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::McpInvoke { task_id, server, tool, params } => {
                if let Some(TaskBox::Invoke(invoke_box)) = self.task_boxes.get_mut(task_id) {
                    invoke_box.set_server(server.clone());
                    invoke_box.set_tool(tool.clone());
                    invoke_box.set_params(params.clone());
                }
            }

            EventKind::McpResponse { task_id, result, duration } => {
                if let Some(TaskBox::Invoke(invoke_box)) = self.task_boxes.get_mut(task_id) {
                    invoke_box.set_response(result.clone());
                    invoke_box.set_duration(*duration);
                }
            }

            EventKind::McpError { task_id, error, code } => {
                if let Some(TaskBox::Invoke(invoke_box)) = self.task_boxes.get_mut(task_id) {
                    invoke_box.set_error(error.clone(), *code);
                    invoke_box.increment_retry_count();
                }
            }

            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            // FETCH VERB EVENTS
            //━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
            EventKind::FetchStarted { task_id, url, method } => {
                if let Some(TaskBox::Fetch(fetch_box)) = self.task_boxes.get_mut(task_id) {
                    fetch_box.set_url(url.clone());
                    fetch_box.set_method(method.clone());
                }
            }

            EventKind::FetchCompleted { task_id, status, body, headers } => {
                if let Some(TaskBox::Fetch(fetch_box)) = self.task_boxes.get_mut(task_id) {
                    fetch_box.set_status(*status);
                    fetch_box.set_response_body(body.clone());
                    if let Some(ct) = headers.get("content-type") {
                        fetch_box.set_content_type(ct.clone());
                    }
                }
            }

            _ => {
                // Unhandled events logged for debugging
                tracing::debug!("Unhandled event: {:?}", event.kind);
            }
        }
    }

impl InvokeBox {
    /// Render parameter preview with truncation and JSON formatting
    fn render_params_preview(&self, max_lines: usize, max_width: usize) -> Vec<Line<'static>> {
        let mut lines = Vec::new();

        if self.params.is_null() {
            lines.push(Line::from(Span::styled(
                "  (no params)",
                Style::default().fg(Color::DarkGray),
            )));
            return lines;
        }

        // Try pretty-print JSON
        if let Ok(pretty) = serde_json::to_string_pretty(&self.params) {
            for (i, line) in pretty.lines().enumerate() {
                if i >= max_lines {
                    lines.push(Line::from(Span::styled(
                        format!("  ... ({} more lines)", pretty.lines().count() - max_lines),
                        Style::default().fg(Color::DarkGray),
                    )));
                    break;
                }

                let display = if line.len() > max_width {
                    format!("{}...", &line[..max_width.saturating_sub(3)])
                } else {
                    line.to_string()
                };

                // Syntax highlighting for JSON
                let styled = self.highlight_json_line(&display);
                lines.push(Line::from(vec![
                    Span::raw("  "),
                    styled,
                ]));
            }
        } else {
            // Fallback to raw string
            let raw = self.params.to_string();
            let truncated = if raw.len() > max_width * max_lines {
                format!("{}...", &raw[..max_width * max_lines - 3])
            } else {
                raw
            };
            lines.push(Line::from(Span::styled(
                format!("  {}", truncated),
                Style::default().fg(Color::Gray),
            )));
        }

        lines
    }

    /// Simple JSON syntax highlighting
    fn highlight_json_line(&self, line: &str) -> Span<'static> {
        // Keys in cyan, strings in green, numbers in yellow, booleans in magenta
        // This is a simplified implementation - full highlighting would use a proper parser
        let line = line.to_string();

        if line.contains(':') && line.trim().starts_with('"') {
            // Key-value pair
            Span::styled(line, Style::default().fg(Color::Cyan))
        } else if line.trim().starts_with('"') {
            // String value
            Span::styled(line, Style::default().fg(Color::Green))
        } else if line.trim().parse::<f64>().is_ok() {
            // Number
            Span::styled(line, Style::default().fg(Color::Yellow))
        } else if line.trim() == "true" || line.trim() == "false" {
            // Boolean
            Span::styled(line, Style::default().fg(Color::Magenta))
        } else {
            // Default (braces, brackets)
            Span::styled(line, Style::default().fg(Color::Gray))
        }
    }

    /// Render collapsed single-line param summary
    fn render_params_collapsed(&self) -> Span<'static> {
        if self.params.is_null() {
            return Span::styled("(no params)", Style::default().fg(Color::DarkGray));
        }

        let summary = match &self.params {
            serde_json::Value::Object(map) => {
                let keys: Vec<_> = map.keys().take(3).collect();
                if map.len() > 3 {
                    format!("{{ {}, ... +{} }}", keys.join(", "), map.len() - 3)
                } else {
                    format!("{{ {} }}", keys.join(", "))
                }
            }
            serde_json::Value::Array(arr) => format!("[{} items]", arr.len()),
            other => other.to_string(),
        };

        let truncated = if summary.len() > 40 {
            format!("{}...", &summary[..37])
        } else {
            summary
        };

        Span::styled(truncated, Style::default().fg(Color::Gray))
    }
}

/// Retry badge for MCP tool invocations
#[derive(Debug, Clone, Default)]
pub struct RetryBadge {
    /// Number of retry attempts
    pub retry_count: u32,
    /// Maximum allowed retries
    pub max_retries: u32,
    /// Last error message
    pub last_error: Option<String>,
    /// MCP error code (JSON-RPC)
    pub error_code: Option<i32>,
}

impl RetryBadge {
    pub fn new(max_retries: u32) -> Self {
        Self {
            retry_count: 0,
            max_retries,
            last_error: None,
            error_code: None,
        }
    }

    pub fn increment(&mut self) {
        self.retry_count += 1;
    }

    pub fn set_error(&mut self, error: String, code: Option<i32>) {
        self.last_error = Some(error);
        self.error_code = code;
    }

    pub fn has_retries(&self) -> bool {
        self.retry_count > 0
    }

    pub fn is_exhausted(&self) -> bool {
        self.retry_count >= self.max_retries
    }

    /// Render the retry badge
    pub fn render(&self) -> Line<'static> {
        if self.retry_count == 0 {
            return Line::default();
        }

        let (color, icon) = if self.is_exhausted() {
            (Color::Red, "⛔")
        } else {
            (Color::Yellow, "🔄")
        };

        let mut spans = vec![
            Span::styled(
                format!("{} {}/{}", icon, self.retry_count, self.max_retries),
                Style::default().fg(color).add_modifier(Modifier::BOLD),
            ),
        ];

        // Add error code if present
        if let Some(code) = self.error_code {
            spans.push(Span::styled(
                format!(" ({})", code),
                Style::default().fg(Color::DarkGray),
            ));
        }

        Line::from(spans)
    }

    /// Render error tooltip (for expanded view)
    pub fn render_error_tooltip(&self, max_width: usize) -> Option<Line<'static>> {
        self.last_error.as_ref().map(|err| {
            let truncated = if err.len() > max_width {
                format!("{}...", &err[..max_width - 3])
            } else {
                err.clone()
            };

            Line::from(vec![
                Span::styled("  ⚠ ", Style::default().fg(Color::Yellow)),
                Span::styled(truncated, Style::default().fg(Color::Gray)),
            ])
        })
    }
}

impl InvokeBox {
    pub fn new(tool: impl Into<String>, server: impl Into<String>) -> Self {
        Self {
            tool: tool.into(),
            server: server.into(),
            params: serde_json::Value::Null,
            response: None,
            error: None,
            state: BoxState::default(),
            retry_badge: RetryBadge::new(3), // Default 3 retries
            expanded_params: false,
            expanded_response: false,
            pulse_intensity: 0.0,
            render_mode: RenderMode::default(),
            duration: None,
        }
    }

    pub fn increment_retry_count(&mut self) {
        self.retry_badge.increment();
    }

    pub fn set_error(&mut self, error: String, code: Option<i32>) {
        self.error = Some(error.clone());
        self.retry_badge.set_error(error, code);
    }

    pub fn retry_count(&self) -> u32 {
        self.retry_badge.retry_count
    }

    pub fn with_max_retries(mut self, max: u32) -> Self {
        self.retry_badge.max_retries = max;
        self
    }
}

use std::collections::VecDeque;
use std::time::{Duration, Instant};

/// Ring buffer for tracking token velocity over time
#[derive(Debug, Clone)]
pub struct TokenVelocity {
    /// (timestamp, tokens_per_second) samples
    samples: VecDeque<(Instant, f32)>,
    /// Maximum samples to retain
    capacity: usize,
    /// Window duration for averaging
    window: Duration,
}

impl Default for TokenVelocity {
    fn default() -> Self {
        Self::new(16, Duration::from_secs(10))
    }
}

impl TokenVelocity {
    /// Create with custom capacity and window
    pub fn new(capacity: usize, window: Duration) -> Self {
        Self {
            samples: VecDeque::with_capacity(capacity),
            capacity,
            window,
        }
    }

    /// Push a new velocity sample
    pub fn push(&mut self, tokens_per_second: f32) {
        let now = Instant::now();

        // Remove old samples outside window
        while let Some((timestamp, _)) = self.samples.front() {
            if now.duration_since(*timestamp) > self.window {
                self.samples.pop_front();
            } else {
                break;
            }
        }

        // Add new sample
        if self.samples.len() >= self.capacity {
            self.samples.pop_front();
        }
        self.samples.push_back((now, tokens_per_second));
    }

    /// Calculate average velocity over retained samples
    pub fn average(&self) -> f32 {
        if self.samples.is_empty() {
            return 0.0;
        }

        let sum: f32 = self.samples.iter().map(|(_, v)| v).sum();
        sum / self.samples.len() as f32
    }

    /// Get current (most recent) velocity
    pub fn current(&self) -> f32 {
        self.samples.back().map(|(_, v)| *v).unwrap_or(0.0)
    }

    /// Get min velocity in window
    pub fn min(&self) -> f32 {
        self.samples
            .iter()
            .map(|(_, v)| *v)
            .min_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal))
            .unwrap_or(0.0)
    }

    /// Get max velocity in window
    pub fn max(&self) -> f32 {
        self.samples
            .iter()
            .map(|(_, v)| *v)
            .max_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal))
            .unwrap_or(0.0)
    }

    /// Check if buffer is empty
    pub fn is_empty(&self) -> bool {
        self.samples.is_empty()
    }

    /// Get sample count
    pub fn len(&self) -> usize {
        self.samples.len()
    }

    /// Generate sparkline characters for visualization
    pub fn sparkline_chars(&self) -> String {
        const SPARKLINE: [char; 8] = ['▁', '▂', '▃', '▄', '▅', '▆', '▇', '█'];

        if self.samples.is_empty() {
            return "▁▁▁▁▁▁▁▁".to_string();
        }

        let max = self.max().max(1.0);
        let samples: Vec<f32> = self.samples.iter().map(|(_, v)| *v).collect();

        // Take last 8 samples for sparkline
        let display_samples: Vec<f32> = if samples.len() > 8 {
            samples[samples.len() - 8..].to_vec()
        } else {
            // Pad with zeros at the beginning
            let mut padded = vec![0.0; 8 - samples.len()];
            padded.extend(samples);
            padded
        };

        display_samples
            .iter()
            .map(|&v| {
                let normalized = (v / max).clamp(0.0, 1.0);
                let idx = (normalized * 7.0).round() as usize;
                SPARKLINE[idx.min(7)]
            })
            .collect()
    }

    /// Render as styled line
    pub fn render_line(&self, show_avg: bool) -> Line<'static> {
        let sparkline = self.sparkline_chars();
        let avg = self.average();

        let mut spans = vec![
            Span::styled(sparkline, Style::default().fg(Color::Cyan)),
        ];

        if show_avg && !self.is_empty() {
            spans.push(Span::raw(" "));
            spans.push(Span::styled(
                format!("{:.0} tok/s", avg),
                Style::default().fg(Color::Gray),
            ));
        }

        Line::from(spans)
    }
}

// In runtime/rig_agent_loop.rs

impl RigAgentLoop {
    /// Run with streaming and emit velocity events
    pub async fn run_with_streaming<M: CompletionModel>(
        &mut self,
        model: M,
    ) -> Result<AgentResult, NikaError> {
        let start = Instant::now();
        let mut total_tokens = 0u32;
        let mut last_chunk_time = start;

        let stream = model
            .stream(&self.params.prompt)
            .await
            .map_err(|e| NikaError::ProviderError {
                source: e.to_string(),
            })?;

        let mut response = String::new();

        while let Some(chunk) = stream.next().await {
            let chunk = chunk.map_err(|e| NikaError::ProviderError {
                source: e.to_string(),
            })?;

            match chunk {
                StreamedAssistantContent::Delta(delta) => {
                    // Append content
                    response.push_str(&delta.content);

                    // Calculate velocity
                    let now = Instant::now();
                    let elapsed = now.duration_since(last_chunk_time);
                    let tokens_in_chunk = delta.content.split_whitespace().count() as u32;
                    total_tokens += tokens_in_chunk;

                    if elapsed.as_millis() > 0 {
                        let velocity = tokens_in_chunk as f32
                            / (elapsed.as_millis() as f32 / 1000.0);

                        // Emit streaming chunk event
                        self.log.emit(EventKind::StreamChunk {
                            task_id: self.task_id.clone(),
                            chunk: delta.content.clone(),
                            tokens: tokens_in_chunk,
                            velocity,
                            timestamp: now,
                        });
                    }

                    last_chunk_time = now;
                }

                StreamedAssistantContent::Final(usage) => {
                    // Final token count
                    self.log.emit(EventKind::ProviderResponded {
                        task_id: self.task_id.clone(),
                        response: response.clone(),
                        tokens: Some((usage.input_tokens, usage.output_tokens)),
                        thinking: None,
                    });
                }
            }
        }

        Ok(AgentResult {
            final_response: response,
            total_turns: 1,
            tokens_in: 0, // Updated by Final event
            tokens_out: total_tokens,
            duration: start.elapsed(),
        })
    }
}

    let mut result = String::new();
    for y in 0..buffer.area.height {
