// src/runtime/security.rs
use crate::error::NikaError;

/// Blocklist of dangerous command patterns (case-insensitive)
const BLOCKLIST: &[&str] = &[
    "rm -rf /",
    "rm -rf /*",
    "curl|bash",
    "wget|bash",
    "eval ",
    "mkfifo",
    "nc -e",
    "nc -c",
    "; rm ",
    "&& rm ",
    "| rm ",
];

/// Validate command string for control characters
pub fn validate_command_string(cmd: &str) -> Result<(), NikaError> {
    for (i, c) in cmd.chars().enumerate() {
        let code = c as u32;
        // Reject 0x00-0x1F except \n (0x0A) and \t (0x09)
        if code < 0x20 && code != 0x0A && code != 0x09 {
            return Err(NikaError::CommandInjectionBlocked {
                command: cmd.to_string(),
                reason: format!("Control character 0x{:02X} at position {}", code, i),
            });
        }
    }
    Ok(())
}

/// Check command against blocklist
pub fn check_blocklist(cmd: &str) -> Result<(), NikaError> {
    let lower = cmd.to_lowercase();
    for pattern in BLOCKLIST {
        if lower.contains(pattern) {
            return Err(NikaError::CommandInjectionBlocked {
                command: cmd.to_string(),
                reason: format!("Blocklisted pattern: {}", pattern),
            });
        }
    }
    Ok(())
}

/// Full security validation for exec commands
pub fn validate_exec_command(cmd: &str) -> Result<(), NikaError> {
    validate_command_string(cmd)?;
    check_blocklist(cmd)?;
    Ok(())
}
```

// In src/runtime/executor.rs
use crate::runtime::security::validate_exec_command;

async fn execute_command(command: &str, shell: Option<bool>) -> Result<String, NikaError> {
    // Always validate first
    validate_exec_command(command)?;

    let output = if shell == Some(true) {
        // Shell mode: user explicitly opted in
        // Use shlex::try_quote for any interpolated values
        tokio::process::Command::new("sh")
            .arg("-c")
            .arg(command)
            .output()
            .await?
    } else {
        // Shell-free mode (default): parameterized execution
        let parts = shlex::split(command).ok_or_else(|| NikaError::ExecError {
            task_id: "".into(),
            command: command.to_string(),
            reason: "Failed to parse command (unmatched quotes?)".to_string(),
        })?;

        if parts.is_empty() {
            return Err(NikaError::ExecError {
                task_id: "".into(),
                command: command.to_string(),
                reason: "Empty command".to_string(),
            });
        }

        tokio::process::Command::new(&parts[0])
            .args(&parts[1..])
            .output()
            .await?
    };

    if output.status.success() {
        Ok(String::from_utf8_lossy(&output.stdout).to_string())
    } else {
        Err(NikaError::ExecError {
            task_id: "".into(),
            command: command.to_string(),
            reason: String::from_utf8_lossy(&output.stderr).to_string(),
        })
    }
}
```

// In src/provider/rig.rs
impl RigProvider {
    pub async fn infer_with_options(
        &self,
        prompt: &str,
        model: Option<&str>,
        temperature: Option<f32>,
        system: Option<&str>,
        max_tokens: Option<u32>,
    ) -> Result<String, RigInferError> {
        match self {
            RigProvider::Claude(client) => {
                let model_id = model.unwrap_or("claude-sonnet-4-20250514");
                let mut builder = client.agent(model_id);

                if let Some(temp) = temperature {
                    builder = builder.temperature(temp);
                }
                if let Some(sys) = system {
                    builder = builder.preamble(sys);
                }
                if let Some(max) = max_tokens {
                    builder = builder.max_tokens(max as u64);
                }

                let agent = builder.build();
                agent.prompt(prompt).await.map_err(|e| RigInferError::ProviderError(e.to_string()))
            }
            // Similar for other providers...
            RigProvider::OpenAI(client) => {
                let model_id = model.unwrap_or("gpt-4o");
                let mut builder = client.agent(model_id);

                if let Some(temp) = temperature {
                    builder = builder.temperature(temp);
                }
                if let Some(sys) = system {
                    builder = builder.preamble(sys);
                }
                if let Some(max) = max_tokens {
                    builder = builder.max_tokens(max as u64);
                }

                let agent = builder.build();
                agent.prompt(prompt).await.map_err(|e| RigInferError::ProviderError(e.to_string()))
            }
            // ... other providers
        }
    }
}
