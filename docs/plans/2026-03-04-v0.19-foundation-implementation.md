# Nika v0.19 Foundation Implementation Plan

> **Two-Phase IR Architecture** — Transform Nika from "parsed YAML config" to "real programming language IR" with span tracking, interned identifiers, and proper language semantics.

## Overview

This plan implements LAYER 0: FOUNDATION from the DX Ecosystem Design. All subsequent layers (validation v0.20, LSP v0.21, generation v0.22) depend on this foundation.

### Goals

1. **Span Tracking** — Every AST node has line:col position (marked_yaml)
2. **Two-Phase IR** — raw::Workflow (parsed) → analyzed::Workflow (validated)
3. **Interned Identifiers** — TaskId(u32) instead of String comparisons
4. **Suggestion Engine** — "Did you mean?" with Levenshtein distance (strsim)
5. **Multi-File Support** — SourceRegistry for pkg: includes
6. **Language Reference** — Complete specification document

//! Span and source position tracking for Nika workflows.

use std::ops::Range;

/// Unique identifier for a source file in the SourceRegistry.
#[derive(Copy, Clone, Debug, Eq, PartialEq, Hash)]
pub struct FileId(pub u32);

/// A byte offset in a source file.
#[derive(Copy, Clone, Debug, Eq, PartialEq, Ord, PartialOrd)]
pub struct ByteOffset(pub u32);

/// A span in a source file (byte range + file ID).
#[derive(Copy, Clone, Debug, Eq, PartialEq)]
pub struct Span {
    pub file: FileId,
    pub start: ByteOffset,
    pub end: ByteOffset,
}

impl Span {
    pub fn new(file: FileId, start: u32, end: u32) -> Self {
        Self {
            file,
            start: ByteOffset(start),
            end: ByteOffset(end),
        }
    }

    pub fn dummy() -> Self {
        Self::new(FileId(0), 0, 0)
    }

    pub fn is_dummy(&self) -> bool {
        self.file.0 == 0 && self.start.0 == 0 && self.end.0 == 0
    }

    pub fn range(&self) -> Range<usize> {
        self.start.0 as usize..self.end.0 as usize
    }

    pub fn len(&self) -> usize {
        (self.end.0 - self.start.0) as usize
    }
}

/// A value with an associated span.
#[derive(Clone, Debug)]
pub struct Spanned<T> {
    pub value: T,
    pub span: Span,
}

impl<T> Spanned<T> {
    pub fn new(value: T, span: Span) -> Self {
        Self { value, span }
    }

    pub fn dummy(value: T) -> Self {
        Self { value, span: Span::dummy() }
    }

    pub fn map<U, F: FnOnce(T) -> U>(self, f: F) -> Spanned<U> {
        Spanned {
            value: f(self.value),
            span: self.span,
        }
    }
}

impl<T> std::ops::Deref for Spanned<T> {
    type Target = T;
    fn deref(&self) -> &Self::Target {
        &self.value
    }
}

//! Multi-file source registry for tracking included files.

use std::collections::HashMap;
use std::path::{Path, PathBuf};
use std::sync::Arc;

use super::span::{FileId, Span, ByteOffset};

/// A single source file with its content and path.
#[derive(Debug)]
pub struct SourceFile {
    pub id: FileId,
    pub path: PathBuf,
    pub content: Arc<String>,
    /// Line start byte offsets for fast line:col lookup
    line_starts: Vec<u32>,
}

impl SourceFile {
    pub fn new(id: FileId, path: PathBuf, content: String) -> Self {
        let line_starts = std::iter::once(0)
            .chain(content.match_indices('\n').map(|(i, _)| i as u32 + 1))
            .collect();

        Self {
            id,
            path,
            content: Arc::new(content),
            line_starts,
        }
    }

    /// Convert byte offset to (line, column), 1-indexed.
    pub fn offset_to_line_col(&self, offset: ByteOffset) -> (usize, usize) {
        let offset = offset.0 as usize;
        let line = self.line_starts
            .partition_point(|&start| start as usize <= offset)
            .saturating_sub(1);
        let line_start = self.line_starts.get(line).copied().unwrap_or(0) as usize;
        let column = offset - line_start;
        (line + 1, column + 1)  // 1-indexed
    }

    /// Get the text at a span.
    pub fn text_at(&self, span: Span) -> &str {
        &self.content[span.range()]
    }

    /// Get the line text for a given line number (1-indexed).
    pub fn line_text(&self, line: usize) -> Option<&str> {
        if line == 0 || line > self.line_starts.len() {
            return None;
        }
        let start = self.line_starts[line - 1] as usize;
        let end = self.line_starts
            .get(line)
            .map(|&e| e as usize)
            .unwrap_or(self.content.len());
        Some(self.content[start..end].trim_end_matches('\n'))
    }
}

/// Registry of all source files in a compilation unit.
#[derive(Debug, Default)]
pub struct SourceRegistry {
    files: Vec<SourceFile>,
    path_to_id: HashMap<PathBuf, FileId>,
    next_id: u32,
}

impl SourceRegistry {
    pub fn new() -> Self {
        Self::default()
    }

    /// Add a source file and return its FileId.
    pub fn add_file(&mut self, path: impl AsRef<Path>, content: String) -> FileId {
        let path = path.as_ref().to_path_buf();
        if let Some(&id) = self.path_to_id.get(&path) {
            return id;
        }

        let id = FileId(self.next_id);
        self.next_id += 1;

        let file = SourceFile::new(id, path.clone(), content);
        self.files.push(file);
        self.path_to_id.insert(path, id);

        id
    }

    /// Get a source file by ID.
    pub fn get(&self, id: FileId) -> Option<&SourceFile> {
        self.files.get(id.0 as usize)
    }

    /// Get file path for a FileId.
    pub fn path(&self, id: FileId) -> Option<&Path> {
        self.get(id).map(|f| f.path.as_path())
    }

    /// Format a span for display: "path:line:col"
    pub fn format_location(&self, span: Span) -> String {
        if span.is_dummy() {
            return "<unknown>".to_string();
        }
        if let Some(file) = self.get(span.file) {
            let (line, col) = file.offset_to_line_col(span.start);
            format!("{}:{}:{}", file.path.display(), line, col)
        } else {
            "<unknown>".to_string()
        }
    }
}

//! Helpers for converting marked_yaml spans to our Span type.

use marked_yaml::{Marker, Span as MarkedSpan};
use crate::source::{FileId, Span, ByteOffset};

/// Convert a marked_yaml Marker to our ByteOffset.
pub fn marker_to_offset(marker: &Marker) -> ByteOffset {
    // marked_yaml uses byte offsets internally
    ByteOffset(marker.source() as u32)
}

/// Convert a marked_yaml Span to our Span.
pub fn marked_span_to_span(file: FileId, span: &MarkedSpan) -> Span {
    Span {
        file,
        start: marker_to_offset(span.start()),
        end: marker_to_offset(span.end()),
    }
}

/// Extension trait for easier span extraction from marked_yaml nodes.
pub trait MarkedExt {
    fn to_span(&self, file: FileId) -> Span;
}

impl<'a> MarkedExt for marked_yaml::Node<'a> {
    fn to_span(&self, file: FileId) -> Span {
        match self.span() {
            Some(s) => marked_span_to_span(file, s),
            None => Span::dummy(),
        }
    }
}

//! Interned identifiers for efficient comparison and storage.

use std::collections::HashMap;
use std::sync::Arc;

/// An interned task identifier.
/// Uses u32 for efficient storage and comparison.
#[derive(Copy, Clone, Debug, Eq, PartialEq, Hash)]
pub struct TaskId(pub u32);

/// An interned string identifier (for general use).
#[derive(Copy, Clone, Debug, Eq, PartialEq, Hash)]
pub struct SymbolId(pub u32);

/// String interner for efficient storage.
#[derive(Debug, Default)]
pub struct Interner {
    strings: Vec<Arc<str>>,
    lookup: HashMap<Arc<str>, u32>,
}

impl Interner {
    pub fn new() -> Self {
        Self::default()
    }

    /// Intern a string and return its ID.
    pub fn intern(&mut self, s: &str) -> SymbolId {
        if let Some(&id) = self.lookup.get(s) {
            return SymbolId(id);
        }

        let id = self.strings.len() as u32;
        let arc: Arc<str> = s.into();
        self.strings.push(arc.clone());
        self.lookup.insert(arc, id);
        SymbolId(id)
    }

    /// Get the string for an ID.
    pub fn resolve(&self, id: SymbolId) -> Option<&str> {
        self.strings.get(id.0 as usize).map(|s| s.as_ref())
    }

    /// Number of interned strings.
    pub fn len(&self) -> usize {
        self.strings.len()
    }

    pub fn is_empty(&self) -> bool {
        self.strings.is_empty()
    }
}

/// Task ID interner (specialized for tasks).
#[derive(Debug, Default)]
pub struct TaskInterner {
    inner: Interner,
}

impl TaskInterner {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn intern(&mut self, task_name: &str) -> TaskId {
        TaskId(self.inner.intern(task_name).0)
    }

    pub fn resolve(&self, id: TaskId) -> Option<&str> {
        self.inner.resolve(SymbolId(id.0))
    }

    pub fn len(&self) -> usize {
        self.inner.len()
    }

    pub fn is_empty(&self) -> bool {
        self.inner.is_empty()
    }

    /// Get all task IDs.
    pub fn all_ids(&self) -> impl Iterator<Item = TaskId> {
        (0..self.len() as u32).map(TaskId)
    }
}

//! Symbol table for resolved references.

use std::collections::HashMap;
use indexmap::IndexMap;

use crate::source::Span;
use super::ids::TaskId;

/// Information about a defined symbol (task, variable, etc.).
#[derive(Debug, Clone)]
pub struct SymbolInfo {
    /// Where the symbol is defined
    pub def_span: Span,
    /// Symbol kind
    pub kind: SymbolKind,
    /// All references to this symbol
    pub references: Vec<Span>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SymbolKind {
    Task,
    UseAlias,
    LoopVariable,
    Input,
}

/// Symbol table for a workflow.
#[derive(Debug, Default)]
pub struct SymbolTable {
    /// Task definitions: name -> info
    tasks: IndexMap<TaskId, SymbolInfo>,
    /// Use aliases in each task: task_id -> (alias -> target_task)
    use_bindings: HashMap<TaskId, IndexMap<String, UseBinding>>,
}

#[derive(Debug, Clone)]
pub struct UseBinding {
    pub alias: String,
    pub alias_span: Span,
    pub target: TaskId,
    pub path: Option<String>,
}

impl SymbolTable {
    pub fn new() -> Self {
        Self::default()
    }

    /// Register a task definition.
    pub fn define_task(&mut self, id: TaskId, span: Span) {
        self.tasks.insert(id, SymbolInfo {
            def_span: span,
            kind: SymbolKind::Task,
            references: Vec::new(),
        });
    }

    /// Add a reference to a task.
    pub fn add_task_reference(&mut self, id: TaskId, ref_span: Span) {
        if let Some(info) = self.tasks.get_mut(&id) {
            info.references.push(ref_span);
        }
    }

    /// Get task definition info.
    pub fn get_task(&self, id: TaskId) -> Option<&SymbolInfo> {
        self.tasks.get(&id)
    }

    /// Check if a task is defined.
    pub fn has_task(&self, id: TaskId) -> bool {
        self.tasks.contains_key(&id)
    }

    /// Get all task IDs in definition order.
    pub fn task_ids(&self) -> impl Iterator<Item = TaskId> + '_ {
        self.tasks.keys().copied()
    }

    /// Register a use binding for a task.
    pub fn add_use_binding(&mut self, task: TaskId, binding: UseBinding) {
        self.use_bindings
            .entry(task)
            .or_default()
            .insert(binding.alias.clone(), binding);
    }

    /// Get use bindings for a task.
    pub fn get_use_bindings(&self, task: TaskId) -> Option<&IndexMap<String, UseBinding>> {
        self.use_bindings.get(&task)
    }
}

//! Reference resolution pass.

use crate::source::Span;
use crate::ast::raw::{RawWorkflow, RawTask, RawUseRef, RawUseTarget};
use crate::ast::analyzed::{TaskId, UseBinding};
use super::context::{AnalysisContext, AnalysisError, AnalysisErrorKind};

/// Resolve all references in a workflow.
pub fn resolve_references(
    ctx: &mut AnalysisContext<'_>,
    workflow: &RawWorkflow,
) -> Result<(), Vec<AnalysisError>> {
    // Pass 1: Collect all task definitions
    for task in &workflow.tasks.value {
        let task_name = &task.value.id.value;
        let task_id = ctx.intern_task(task_name);

        // Check for duplicates
        if ctx.symbols.has_task(task_id) {
            ctx.error(AnalysisError {
                kind: AnalysisErrorKind::DuplicateTask,
                span: task.value.id.span,
                message: format!("duplicate task ID '{}'", task_name),
            });
        } else {
            ctx.symbols.define_task(task_id, task.value.id.span);
        }
    }

    // Pass 2: Resolve references
    for task in &workflow.tasks.value {
        let task_id = ctx.intern_task(&task.value.id.value);
        resolve_task_refs(ctx, task_id, &task.value);
    }

    if ctx.has_errors() {
        Err(ctx.errors.clone())
    } else {
        Ok(())
    }
}

fn resolve_task_refs(
    ctx: &mut AnalysisContext<'_>,
    task_id: TaskId,
    task: &RawTask,
) {
    // Resolve use: references
    if let Some(use_refs) = &task.use_refs {
        for use_ref in &use_refs.value {
            resolve_use_ref(ctx, task_id, use_ref);
        }
    }

    // Resolve flow: references
    if let Some(flow) = &task.flow {
        for flow_ref in &flow.value {
            let target_name = &flow_ref.value;
            let target_id = ctx.intern_task(target_name);

            if !ctx.symbols.has_task(target_id) {
                ctx.error(AnalysisError {
                    kind: AnalysisErrorKind::UnknownTask,
                    span: flow_ref.span,
                    message: format!("unknown task '{}' in flow", target_name),
                });
            } else {
                ctx.symbols.add_task_reference(target_id, flow_ref.span);
            }
        }
    }
}

fn resolve_use_ref(
    ctx: &mut AnalysisContext<'_>,
    task_id: TaskId,
    use_ref: &RawUseRef,
) {
    let (target_name, target_span, path) = match &use_ref.target {
        RawUseTarget::TaskId(id) => (&id.value, id.span, None),
        RawUseTarget::Extended { task, path } => {
            (&task.value, task.span, path.as_ref().map(|p| p.value.clone()))
        }
    };

    let target_id = ctx.intern_task(target_name);

    if !ctx.symbols.has_task(target_id) {
        ctx.error(AnalysisError {
            kind: AnalysisErrorKind::UnknownTask,
            span: target_span,
            message: format!("unknown task '{}' in use reference", target_name),
        });
    } else {
        ctx.symbols.add_task_reference(target_id, target_span);
        ctx.symbols.add_use_binding(task_id, UseBinding {
            alias: use_ref.alias.value.clone(),
            alias_span: use_ref.alias.span,
            target: target_id,
            path,
        });
    }
}

//! "Did you mean?" suggestion engine using Levenshtein distance.

use strsim::levenshtein;

/// Configuration for the suggestion engine.
#[derive(Debug, Clone)]
pub struct SuggestionConfig {
    /// Maximum edit distance to consider a suggestion (default: 3)
    pub max_distance: usize,
    /// Maximum number of suggestions to return (default: 3)
    pub max_suggestions: usize,
    /// Minimum similarity ratio to consider (0.0 - 1.0, default: 0.5)
    pub min_similarity: f64,
}

impl Default for SuggestionConfig {
    fn default() -> Self {
        Self {
            max_distance: 3,
            max_suggestions: 3,
            min_similarity: 0.5,
        }
    }
}

/// A suggestion with its score.
#[derive(Debug, Clone)]
pub struct Suggestion {
    pub text: String,
    pub distance: usize,
    pub similarity: f64,
}

/// Find similar strings to the input.
pub fn find_similar<'a, I>(
    input: &str,
    candidates: I,
    config: &SuggestionConfig,
) -> Vec<Suggestion>
where
    I: IntoIterator<Item = &'a str>,
{
    let input_len = input.len();
    let mut suggestions: Vec<Suggestion> = candidates
        .into_iter()
        .filter_map(|candidate| {
            let distance = levenshtein(input, candidate);

            // Quick reject if too far
            if distance > config.max_distance {
                return None;
            }

            // Calculate similarity ratio
            let max_len = input_len.max(candidate.len());
            let similarity = if max_len == 0 {
                1.0
            } else {
                1.0 - (distance as f64 / max_len as f64)
            };

            if similarity < config.min_similarity {
                return None;
            }

            Some(Suggestion {
                text: candidate.to_string(),
                distance,
                similarity,
            })
        })
        .collect();

    // Sort by distance (ascending), then by similarity (descending)
    suggestions.sort_by(|a, b| {
        a.distance.cmp(&b.distance)
            .then(b.similarity.partial_cmp(&a.similarity).unwrap())
    });

    suggestions.truncate(config.max_suggestions);
    suggestions
}

/// Format suggestions for display.
pub fn format_suggestions(suggestions: &[Suggestion]) -> String {
    if suggestions.is_empty() {
        return String::new();
    }

    if suggestions.len() == 1 {
        return format!("did you mean '{}'?", suggestions[0].text);
    }

    let names: Vec<_> = suggestions.iter().map(|s| format!("'{}'", s.text)).collect();
    format!("did you mean one of: {}?", names.join(", "))
}

        let config = SuggestionConfig::default();

        let suggestions = find_similar("perpare_data", &candidates, &config);
        assert!(!suggestions.is_empty());
        assert_eq!(suggestions[0].text, "prepare_data");
        assert_eq!(suggestions[0].distance, 1);
    }
