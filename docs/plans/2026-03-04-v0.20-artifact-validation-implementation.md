```rust
use jsonpath_rust::JsonPath;

/// Normalize path to full JSONPath syntax
fn normalize_path(path: &str) -> String {
    let trimmed = path.trim();

    // Empty or just "$" returns root
    if trimmed.is_empty() {
        return "$".to_string();
    }
    if trimmed == "$" {
        return "$".to_string();
    }

    // Already has $. or $[ prefix
    if trimmed.starts_with("$.") || trimmed.starts_with("$[") {
        return trimmed.to_string();
    }

    // Shorthand: prepend $.
    format!("$.{}", trimmed)
}

/// Extract a value from JSON using JSONPath expression
pub fn extract_jsonpath(json: &Value, path: &str) -> Result<Value, NikaError> {
    let normalized = normalize_path(path);

    // Parse JSONPath
    let jsonpath = JsonPath::try_from(normalized.as_str()).map_err(|e| {
        NikaError::JsonPathUnsupported {
            path: path.to_string(),
        }
    })?;

    // Execute query
    let results = jsonpath.find(json);

    // Handle results
    match results.len() {
        0 => Err(NikaError::JsonPathNoMatch {
            path: path.to_string(),
            task_id: String::new(), // Will be filled by caller
        }),
        1 => {
            // Single match - return the value directly
            Ok(results.into_iter().next().unwrap().clone())
        }
        _ => {
            // Multiple matches - return as array
            Ok(Value::Array(results.into_iter().cloned().collect()))
        }
    }
}
```

```rust
async fn write_single_artifact(

    output: &str,
    output_spec: &ArtifactOutput,
    workflow_config: Option<&ArtifactsConfig>,
    writer: &ArtifactWriter,
) -> Result<WriteResult, NikaError> {
    // Determine format (task spec > workflow default)
    let format = output_spec
        .format
        .or(workflow_config.map(|c| c.format))
        .unwrap_or(ArtifactFormat::Text);

    // Determine mode (task spec > workflow default)
    let mode = output_spec
        .mode
        .or(workflow_config.map(|c| c.mode))
        .unwrap_or(ArtifactMode::Overwrite);

    // === NEW: Source extraction (v0.20) ===
    let extracted_output = if let Some(ref source_path) = output_spec.source {
        // Parse output as JSON for extraction
        let json: serde_json::Value = serde_json::from_str(output).map_err(|e| {
            NikaError::ArtifactSourceExtractionFailed {
                path: output_spec.path.clone(),
                source: source_path.clone(),
                reason: format!("Task output is not valid JSON: {}", e),
            }
        })?;

        // Extract using JSONPath
        let extracted = extract_jsonpath(&json, source_path).map_err(|e| {
            NikaError::ArtifactSourceExtractionFailed {
                path: output_spec.path.clone(),
                source: source_path.clone(),
                reason: e.to_string(),
            }
        })?;

        // Convert back to string based on format
        match format {
            ArtifactFormat::Json => serde_json::to_string_pretty(&extracted)
                .unwrap_or_else(|_| extracted.to_string()),

            }

        // No source extraction - use raw output
```

```rust
//! JSON Schema validation for artifacts (v0.20)

use crate::ast::output::SchemaRef;
use crate::error::NikaError;
use jsonschema::JSONSchema;
use serde_json::Value;
use std::path::Path;

/// Validate a JSON value against a schema reference
pub fn validate_against_schema(
    value: &Value,
    schema_ref: &SchemaRef,
    base_path: &Path,
    artifact_path: &str,
) -> Result<(), NikaError> {
    // Load the schema
    let schema = load_schema(schema_ref, base_path)?;

    // Compile the schema
    let compiled = JSONSchema::compile(&schema).map_err(|e| {
        NikaError::ArtifactSchemaLoadFailed {
            schema_path: schema_ref_to_string(schema_ref),
            reason: format!("Invalid JSON Schema: {}", e),
        }
    })?;

    // Validate
    match compiled.validate(value) {
        Ok(_) => Ok(()),
        Err(errors) => {
            let error_messages: Vec<String> = errors
                .map(|e| format!("{} at {}", e, e.instance_path))
                .collect();

            Err(NikaError::ArtifactSchemaValidationFailed {
                path: artifact_path.to_string(),
                schema_path: Some(schema_ref_to_string(schema_ref)),
                errors: error_messages,
            })
        }
    }
}

/// Load schema from SchemaRef (inline or file)
fn load_schema(schema_ref: &SchemaRef, base_path: &Path) -> Result<Value, NikaError> {
    match schema_ref {
        SchemaRef::Inline(value) => Ok(value.clone()),
        SchemaRef::File(path) => {
            let schema_path = base_path.join(path);
            let content = std::fs::read_to_string(&schema_path).map_err(|e| {
                NikaError::ArtifactSchemaLoadFailed {
                    schema_path: path.clone(),
                    reason: format!("Failed to read file: {}", e),
                }
            })?;

            serde_json::from_str(&content).map_err(|e| {
                NikaError::ArtifactSchemaLoadFailed {
                    schema_path: path.clone(),
                    reason: format!("Invalid JSON: {}", e),
                }
            })
        }
    }
}

fn schema_ref_to_string(schema_ref: &SchemaRef) -> String {
    match schema_ref {
        SchemaRef::Inline(_) => "<inline>".to_string(),
        SchemaRef::File(path) => path.clone(),
    }
}
```
