/// Parse a task action from the mapping.
fn parse_action(
    file_id: FileId,
    map: &marked_yaml::types::MarkedMappingNode,
) -> Result<Option<RawTaskAction>, ParseError> {
    // Check for infer verb
    if let Some(node) = map.get_node("infer") {
        return parse_infer_action(file_id, node).map(|a| Some(RawTaskAction::Infer(a)));
    }
    // Check for exec verb
    if let Some(node) = map.get_node("exec") {
        return parse_exec_action(file_id, node).map(|a| Some(RawTaskAction::Exec(a)));
    }
    // Check for fetch verb
    if let Some(node) = map.get_node("fetch") {
        return parse_fetch_action(file_id, node).map(|a| Some(RawTaskAction::Fetch(a)));
    }
    // Check for invoke verb
    if let Some(node) = map.get_node("invoke") {
        return parse_invoke_action(file_id, node).map(|a| Some(RawTaskAction::Invoke(a)));
    }
    // Check for agent verb
    if let Some(node) = map.get_node("agent") {

/// Parse infer action - supports both shorthand and full form
fn parse_infer_action(
    file_id: FileId,
    node: &Node,
) -> Result<RawInferAction, ParseError> {
    let span = node_to_span(file_id, node);

    match node {
        // Shorthand: infer: "prompt string"
        Node::Scalar(s) => Ok(RawInferAction {
            prompt: Spanned::new(s.to_string(), span),
            temperature: None,
            system: None,
            max_tokens: None,
            model: None,
            extended_thinking: None,
            thinking_budget: None,
            span,
        }),
        // Full form: infer: { prompt: "...", temperature: ... }
        Node::Mapping(m) => {
            let prompt = get_string_field(file_id, m, "prompt")?
                .ok_or_else(|| ParseError {
                    kind: ParseErrorKind::MissingField,
                    span,
                    message: "infer action requires 'prompt' field".to_string(),
                })?;

            let temperature = get_f64_field(file_id, m, "temperature")?;
            let system = get_string_field(file_id, m, "system")?;
            let max_tokens = get_u32_field(file_id, m, "max_tokens")?;
            let model = get_string_field(file_id, m, "model")?;
            let extended_thinking = get_bool_field(file_id, m, "extended_thinking")?;
            let thinking_budget = get_u32_field(file_id, m, "thinking_budget")?;

            Ok(RawInferAction {
                prompt,
                temperature,
                system,
                max_tokens,
                model,
                extended_thinking,
                thinking_budget,
                span,
            })
        }
        _ => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span,
            message: "infer must be a string or mapping".to_string(),
        }),
    }
}

/// Get optional f64 field
fn get_f64_field(
    file_id: FileId,
    map: &marked_yaml::types::MarkedMappingNode,
    key: &str,
) -> Result<Option<Spanned<f64>>, ParseError> {
    match map.get_node(key) {
        Some(Node::Scalar(s)) => {
            let span = marked_span_to_span(file_id, s.span());
            let value: f64 = s.parse().map_err(|_| ParseError {
                kind: ParseErrorKind::InvalidType,
                span,
                message: format!("'{}' must be a number", key),
            })?;
            Ok(Some(Spanned::new(value, span)))

        Some(node) => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span: node_to_span(file_id, node),
            message: format!("'{}' must be a number", key),
        }),
        None => Ok(None),
    }
}

/// Parse exec action - supports both shorthand and full form
fn parse_exec_action(
    file_id: FileId,
    node: &Node,
) -> Result<RawExecAction, ParseError> {
    let span = node_to_span(file_id, node);

    match node {
        // Shorthand: exec: "command string"
        Node::Scalar(s) => Ok(RawExecAction {
            command: Spanned::new(s.to_string(), span),
            shell: None,
            timeout: None,
            cwd: None,
            env: None,
            span,
        }),
        // Full form
        Node::Mapping(m) => {
            let command = get_string_field(file_id, m, "command")?
                .ok_or_else(|| ParseError {
                    kind: ParseErrorKind::MissingField,
                    span,
                    message: "exec action requires 'command' field".to_string(),
                })?;

            Ok(RawExecAction {
                command,
                shell: get_bool_field(file_id, m, "shell")?,
                timeout: get_u32_field(file_id, m, "timeout")?,
                cwd: get_string_field(file_id, m, "cwd")?,
                env: parse_env_map(file_id, m)?,
                span,
            })
        }
        _ => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span,
            message: "exec must be a string or mapping".to_string(),
        }),
    }
}

/// Parse environment variables map
fn parse_env_map(
    file_id: FileId,
    parent: &marked_yaml::types::MarkedMappingNode,
) -> Result<Option<Spanned<IndexMap<String, Spanned<String>>>>, ParseError> {
    match parent.get_node("env") {
        Some(Node::Mapping(m)) => {
            let span = marked_span_to_span(file_id, m.span());
            let mut env = IndexMap::new();

            for (key, value) in m.iter() {
                let key_str = key.as_str().to_string();
                if let Node::Scalar(s) = value {
                    let val_span = marked_span_to_span(file_id, s.span());
                    env.insert(key_str, Spanned::new(s.to_string(), val_span));
                }
            }

            Ok(Some(Spanned::new(env, span)))
        }
        Some(node) => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span: node_to_span(file_id, node),
            message: "env must be a mapping".to_string(),
        }),
        None => Ok(None),
    }
}

/// Parse use: references
fn parse_use_refs(
    file_id: FileId,
    map: &marked_yaml::types::MarkedMappingNode,
) -> Result<Option<Spanned<IndexMap<Spanned<String>, RawUseTarget>>>, ParseError> {
    match map.get_node("use") {
        Some(Node::Mapping(m)) => {
            let span = marked_span_to_span(file_id, m.span());
            let mut refs = IndexMap::new();

            for (key, value) in m.iter() {
                let alias_span = marked_span_to_span(file_id, key.span());
                let alias = Spanned::new(key.as_str().to_string(), alias_span);
                let target = parse_use_target(file_id, value)?;
                refs.insert(alias, target);
            }

            Ok(Some(Spanned::new(refs, span)))
        }
        Some(node) => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span: node_to_span(file_id, node),
            message: "use must be a mapping".to_string(),
        }),
        None => Ok(None),
    }
}

fn parse_use_target(file_id: FileId, node: &Node) -> Result<RawUseTarget, ParseError> {
    let span = node_to_span(file_id, node);

    match node {
        // Simple: alias: task_id
        Node::Scalar(s) => {
            Ok(RawUseTarget::TaskId(Spanned::new(s.to_string(), span)))
        }
        // Extended: { task: id, path: "..." }
        Node::Mapping(m) => {
            let task = get_string_field(file_id, m, "task")?
                .ok_or_else(|| ParseError {
                    kind: ParseErrorKind::MissingField,
                    span,
                    message: "use target requires 'task' field".to_string(),
                })?;
            let path = get_string_field(file_id, m, "path")?;

            Ok(RawUseTarget::Extended { task, path, span })
        }
        _ => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span,
            message: "use target must be string or mapping".to_string(),
        }),
    }
}

            Ok(Some(Spanned::new(RawRetryConfig {
                max_attempts: get_u32_field(file_id, m, "max_attempts")?,
                delay_ms: get_u32_field(file_id, m, "delay_ms")?,
                backoff: get_f64_field(file_id, m, "backoff")?,
            }, span)))
        }
        Some(node) => Err(ParseError {
            kind: ParseErrorKind::InvalidType,
            span: node_to_span(file_id, node),
            message: "retry must be a mapping".to_string(),
        }),
        None => Ok(None),
    }
}

fn analyze_for_each(
    raw: &Spanned<RawForEach>,
    ctx: &mut AnalyzerContext,
) -> Option<AnalyzedForEach> {
    let items = &raw.value.items.value;

    let parsed_items = if items.starts_with('[') {
        // Try parse as JSON array
        match serde_json::from_str::<Vec<serde_json::Value>>(items) {
            Ok(arr) => ForEachItems::Literal(arr),
            Err(_) => {
                ctx.add_error(AnalyzeError::invalid_for_each(
                    raw.value.items.span,
                    "invalid array literal",
                ));
                return None;
            }
        }
    } else {
        // Binding expression
        ForEachItems::Binding(items.clone())
    };

    Some(AnalyzedForEach {
        items: parsed_items,
        as_var: raw.value.as_var.as_ref()
            .map(|s| s.value.clone())
            .unwrap_or_else(|| "item".to_string()),
        concurrency: raw.value.parallel.as_ref()
            .map(|s| s.value)
            .unwrap_or(1),
    })
}

fn analyze_retry(raw: &Spanned<RawRetryConfig>) -> Option<AnalyzedRetryConfig> {
    Some(AnalyzedRetryConfig {
        max_attempts: raw.value.max_attempts.as_ref()
            .map(|s| s.value)
            .unwrap_or(3),
        delay_ms: raw.value.delay_ms.as_ref()
            .map(|s| s.value)
            .unwrap_or(1000),
        backoff_multiplier: raw.value.backoff.as_ref()
            .map(|s| s.value)
            .unwrap_or(2.0),
    })
}
