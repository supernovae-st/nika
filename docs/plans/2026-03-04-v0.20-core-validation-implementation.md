# Nika v0.20 Core Validation Implementation Plan

> Detailed step-by-step implementation plan for v0.20 validation foundation.

## Overview

Transform Nika's validation from structural-only (JSON Schema) to full semantic validation with:
- Source span tracking (line:col)
- Cargo-style error messages (miette)
- Semantic validations (use: wiring, templates, providers)
- Batch validation and dry-run

## Prerequisites

Before starting, ensure:
- [ ] Nika v0.19.x is stable and merged to main
- [ ] All existing tests pass (`cargo test`)
- [ ] Create feature branch: `git checkout -b feat/v0.20-validation`

```rust
//! Convert marked_yaml spans to miette SourceSpan.

use marked_yaml::{Marker, Span as YamlSpan};
use miette::{NamedSource, SourceOffset, SourceSpan};

/// Convert marked_yaml Marker to miette SourceSpan
pub fn marker_to_span(source: &str, marker: &Marker, length: usize) -> SourceSpan {
    // marked_yaml uses 1-indexed line/column
    // miette's from_location uses 1-indexed as well
    let offset = SourceOffset::from_location(source, marker.line(), marker.column());
    SourceSpan::new(offset, length)
}

/// Convert YamlSpan to miette SourceSpan
pub fn yaml_span_to_source_span(source: &str, span: &YamlSpan) -> Option<SourceSpan> {
    let start = span.start()?;
    let end = span.end().unwrap_or(start);

    let start_offset = SourceOffset::from_location(source, start.line(), start.column());
    let end_offset = SourceOffset::from_location(source, end.line(), end.column());

    let length = end_offset.offset().saturating_sub(start_offset.offset());
    Some(SourceSpan::new(start_offset, length.max(1)))
}

/// Create a NamedSource for error reporting
pub fn create_named_source(path: &str, content: String) -> NamedSource<String> {
    NamedSource::new(path, content).with_language("yaml")
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_marker_to_span() {
        let source = "line1\nline2\nline3";
        let marker = marked_yaml::Marker::new(0, 6, 2, 1); // line 2, col 1
        let span = marker_to_span(source, &marker, 5);

        assert_eq!(span.offset(), 6);
        assert_eq!(span.len(), 5);
    }
}
```

```rust
//! "Did you mean?" suggestion engine using Levenshtein distance.

use strsim::levenshtein;

/// Maximum edit distance to consider a match
const MAX_DISTANCE: usize = 3;

/// Find similar strings to `target` from `candidates`
pub fn find_similar<'a>(target: &str, candidates: &'a [String]) -> Vec<(&'a str, usize)> {
    let mut matches: Vec<_> = candidates
        .iter()
        .map(|c| (c.as_str(), levenshtein(target, c)))
        .filter(|(_, dist)| *dist <= MAX_DISTANCE)
        .collect();

    matches.sort_by_key(|(_, dist)| *dist);
    matches
}

/// Get best suggestion if available
pub fn best_suggestion<'a>(target: &str, candidates: &'a [String]) -> Option<&'a str> {
    find_similar(target, candidates)
        .into_iter()
        .next()
        .map(|(s, _)| s)
}

/// Format suggestions for error message
pub fn format_suggestions(target: &str, candidates: &[String]) -> Option<String> {
    let matches = find_similar(target, candidates);

    match matches.len() {
        0 => None,
        1 => Some(format!("did you mean '{}'?", matches[0].0)),
        _ => {
            let names: Vec<_> = matches.iter().take(3).map(|(s, _)| *s).collect();
            Some(format!("did you mean one of: {}?", names.join(", ")))
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_find_similar() {
        let candidates = vec![
            "fetch_data".to_string(),
            "prepare_data".to_string(),
            "transform".to_string(),
        ];

        let matches = find_similar("fatch_data", &candidates);
        assert_eq!(matches[0].0, "fetch_data");
        assert_eq!(matches[0].1, 1);
    }

    #[test]
    fn test_no_match() {
        let candidates = vec!["abc".to_string(), "def".to_string()];
        let matches = find_similar("xyz_completely_different", &candidates);
        assert!(matches.is_empty());
    }
}
```

```rust
//! Validation phase orchestration.

mod flows;
mod mcp;
mod providers;
mod semantic;
mod structural;

use crate::validation::{ValidationErrors, NamedSource};
use marked_yaml::Node;

pub use flows::validate_flows;
pub use mcp::validate_mcp;
pub use providers::validate_providers;
pub use semantic::validate_semantic;
pub use structural::validate_structural;

/// Validation context passed through all phases
pub struct ValidationContext {
    /// Path to the workflow file
    pub path: String,
    /// Raw source content
    pub source: String,
    /// Named source for error reporting
    pub named_source: NamedSource<String>,
    /// Parsed YAML AST with spans
    pub ast: Node,
    /// Collected errors
    pub errors: ValidationErrors,
    /// Task IDs found in workflow
    pub task_ids: Vec<String>,
    /// Task ID -> span mapping
    pub task_spans: std::collections::HashMap<String, miette::SourceSpan>,
}

impl ValidationContext {
    pub fn new(path: String, source: String, ast: Node) -> Self {
        let named_source = NamedSource::new(&path, source.clone()).with_language("yaml");
        Self {
            path,
            source,
            named_source,
            ast,
            errors: ValidationErrors::new(),
            task_ids: Vec::new(),
            task_spans: std::collections::HashMap::new(),
        }
    }
}

/// Run all validation phases
pub fn validate(path: &str, source: &str) -> Result<(), ValidationErrors> {
    // Phase 0: Parse YAML with spans
    let ast = match marked_yaml::parse_yaml(0, source) {
        Ok(ast) => ast,
        Err(e) => {
            let mut errors = ValidationErrors::new();
            // Convert parse error to diagnostic
            // TODO: implement parse error conversion
            return Err(errors);
        }
    };

    let mut ctx = ValidationContext::new(path.to_string(), source.to_string(), ast);

    // Phase 1: Structural (JSON Schema)
    validate_structural(&mut ctx);

    // Phase 2: Flows (DAG cycles)
    validate_flows(&mut ctx);

    // Phase 3: Semantic (use: wiring, templates)
    validate_semantic(&mut ctx);

    // Phase 4: Providers (provider/model existence)
    validate_providers(&mut ctx);

    // Phase 5: MCP (tool params) - only in strict mode
    // validate_mcp(&mut ctx);

    if ctx.errors.is_empty() {
        Ok(())
    } else {
        Err(ctx.errors)
    }
}
```

### Task 3.2: Semantic Validation (L3)

**File: `tools/nika/src/validation/phases/semantic.rs`**

```rust
//! L3: Semantic validation - use: wiring, templates.

use super::ValidationContext;
use crate::validation::{
    diagnostics::{InvalidUsePath, UnknownTask, UnresolvedTemplate},
    spans::yaml_span_to_source_span,
    suggestions::find_similar,
};
use marked_yaml::Node;
use regex::Regex;
use std::collections::HashSet;

/// Validate semantic correctness
pub fn validate_semantic(ctx: &mut ValidationContext) {
    // Collect all task IDs first
    collect_task_ids(ctx);

    // Validate use: references
    validate_use_references(ctx);

    // Validate template references
    validate_template_references(ctx);

    // Validate flow: references
    validate_flow_references(ctx);
}

fn collect_task_ids(ctx: &mut ValidationContext) {
    if let Some(tasks) = get_tasks_node(&ctx.ast) {
        for task in tasks.as_sequence().unwrap().iter() {
            if let Some(id_node) = task.as_mapping().and_then(|m| m.get("id")) {
                if let Some(id) = id_node.as_scalar() {
                    let task_id = id.as_str().to_string();

                    // Store span
                    if let Some(span) = yaml_span_to_source_span(&ctx.source, id.span()) {
                        ctx.task_spans.insert(task_id.clone(), span);
                    }

                    ctx.task_ids.push(task_id);
                }
            }
        }
    }
}

fn validate_use_references(ctx: &mut ValidationContext) {
    let task_ids: HashSet<_> = ctx.task_ids.iter().cloned().collect();

    if let Some(tasks) = get_tasks_node(&ctx.ast) {
        for task in tasks.as_sequence().unwrap().iter() {
            let task_map = match task.as_mapping() {
                Some(m) => m,
                None => continue,
            };

            // Get the use: block
            if let Some(use_node) = task_map.get("use") {
                if let Some(use_map) = use_node.as_mapping() {
                    for (alias, ref_node) in use_map.iter() {
                        // Get the referenced task ID (before any .)
                        let ref_str = ref_node.as_scalar().map(|s| s.as_str()).unwrap_or("");
                        let ref_task = ref_str.split('.').next().unwrap_or(ref_str);

                        if !task_ids.contains(ref_task) {
                            // Unknown task reference
                            if let Some(span) = yaml_span_to_source_span(&ctx.source, ref_node.span()) {
                                let suggestions: Vec<_> = find_similar(ref_task, &ctx.task_ids)
                                    .into_iter()
                                    .map(|(s, d)| crate::validation::diagnostics::SuggestionHint {
                                        suggestion: s.to_string(),
                                        distance: d,
                                    })
                                    .collect();

                                ctx.errors.push(UnknownTask {
                                    src: ctx.named_source.clone(),
                                    span,
                                    target: ref_task.to_string(),
                                    available: ctx.task_ids.clone(),
                                    suggestions,
                                });
                            }
                        }
                    }
                }
            }
        }
    }
}

fn validate_template_references(ctx: &mut ValidationContext) {
    // Regex for {{use.alias.path}} patterns
    let template_re = Regex::new(r"\{\{use\.(\w+)(?:\.(\w+(?:\.\w+)*))?\}\}").unwrap();

    let task_ids: HashSet<_> = ctx.task_ids.iter().cloned().collect();

    // Scan all string values for templates
    // For now, we'll scan the raw source (could optimize with AST traversal)
    for cap in template_re.captures_iter(&ctx.source) {
        let alias = &cap[1];

        // Check if alias refers to a valid task
        // Note: This is simplified - real implementation needs to check
        // the current task's use: block to resolve aliases
        if !task_ids.contains(alias) {
            // For now, we skip validation of aliases
            // Full implementation would track use: aliases per task
        }
    }
}

fn validate_flow_references(ctx: &mut ValidationContext) {
    let task_ids: HashSet<_> = ctx.task_ids.iter().cloned().collect();

    if let Some(tasks) = get_tasks_node(&ctx.ast) {
        for task in tasks.as_sequence().unwrap().iter() {
            let task_map = match task.as_mapping() {
                Some(m) => m,
                None => continue,
            };

            // Get the flow: array
            if let Some(flow_node) = task_map.get("flow") {
                if let Some(flow_seq) = flow_node.as_sequence() {
                    for ref_node in flow_seq.iter() {
                        let ref_str = ref_node.as_scalar().map(|s| s.as_str()).unwrap_or("");

                        if !task_ids.contains(ref_str) {
                            if let Some(span) = yaml_span_to_source_span(&ctx.source, ref_node.span()) {
                                let suggestions: Vec<_> = find_similar(ref_str, &ctx.task_ids)
                                    .into_iter()
                                    .map(|(s, d)| crate::validation::diagnostics::SuggestionHint {
                                        suggestion: s.to_string(),
                                        distance: d,
                                    })
                                    .collect();

                                ctx.errors.push(UnknownTask {
                                    src: ctx.named_source.clone(),
                                    span,
                                    target: ref_str.to_string(),
                                    available: ctx.task_ids.clone(),
                                    suggestions,
                                });
                            }
                        }
                    }
                }
            }
        }
    }
}

fn get_tasks_node(ast: &Node) -> Option<&Node> {
    ast.as_mapping()?.get("tasks")
}
```

```rust
//! L4: Provider and model validation.

use super::ValidationContext;
use crate::validation::{
    diagnostics::{UnknownModel, UnknownProvider, SuggestionHint},
    spans::yaml_span_to_source_span,
    suggestions::find_similar,
};
use std::collections::HashMap;

/// Known providers and their models
fn known_models() -> HashMap<&'static str, Vec<&'static str>> {
    let mut m = HashMap::new();

    m.insert("claude", vec![
        "claude-sonnet-4-6",
        "claude-opus-4",
        "claude-3-5-sonnet-20241022",
        "claude-3-5-haiku-20241022",
    ]);

    m.insert("openai", vec![
        "gpt-4o",
        "gpt-4o-mini",
        "gpt-4-turbo",
        "o1",
        "o1-mini",
    ]);

    m.insert("ollama", vec![
        "llama3.2",
        "mistral",
        "codellama",
    ]);

    m.insert("mock", vec!["mock-model"]);

    m
}

/// Validate provider and model
pub fn validate_providers(ctx: &mut ValidationContext) {
    let known = known_models();
    let known_providers: Vec<String> = known.keys().map(|s| s.to_string()).collect();

    let root = match ctx.ast.as_mapping() {
        Some(m) => m,
        None => return,
    };

    // Check workflow-level provider
    let provider = root.get("provider")
        .and_then(|n| n.as_scalar())
        .map(|s| s.as_str());

    if let Some(prov) = provider {
        if !known.contains_key(prov) {
            if let Some(span) = root.get("provider")
                .and_then(|n| yaml_span_to_source_span(&ctx.source, n.span()))
            {
                ctx.errors.push(UnknownProvider {
                    src: ctx.named_source.clone(),
                    span,
                    provider: prov.to_string(),
                });
            }
        }
    }

    // Check workflow-level model
    if let (Some(prov), Some(model_node)) = (provider, root.get("model")) {
        if let Some(model) = model_node.as_scalar() {
            let model_str = model.as_str();

            if let Some(valid_models) = known.get(prov) {
                if !valid_models.contains(&model_str) {
                    if let Some(span) = yaml_span_to_source_span(&ctx.source, model_node.span()) {
                        let model_strings: Vec<String> = valid_models.iter().map(|s| s.to_string()).collect();
                        let suggestions: Vec<_> = find_similar(model_str, &model_strings)
                            .into_iter()
                            .map(|(s, d)| SuggestionHint {
                                suggestion: s.to_string(),
                                distance: d,
                            })
                            .collect();

                        ctx.errors.push(UnknownModel {
                            src: ctx.named_source.clone(),
                            span,
                            model: model_str.to_string(),
                            provider: prov.to_string(),
                            available_models: model_strings,
                            suggestions,
                        });
                    }
                }
            }
        }
    }

    // TODO: Check task-level overrides
}
```

```rust
use walkdir::WalkDir;
use rayon::prelude::*;

/// Discover all workflow files
fn discover_workflows(dir: &Path) -> Vec<PathBuf> {
    WalkDir::new(dir)
        .into_iter()
        .filter_map(|e| e.ok())
        .filter(|e| {
            let name = e.file_name().to_string_lossy();
            name.ends_with(".nika.yaml") || name.ends_with(".nika.yml")
        })
        .filter(|e| {
            // Exclude node_modules, target, .git
            !e.path().components().any(|c| {
                matches!(c.as_os_str().to_str(),
                    Some("node_modules") | Some("target") | Some(".git"))
            })
        })
        .map(|e| e.path().to_path_buf())
        .collect()
}

/// Validate all workflows in directory
pub async fn check_all(args: &CheckArgs) -> Result<()> {
    let workflows = discover_workflows(&args.dir);
    let total = workflows.len();

    println!("Checking {} workflows...\n", total);

    let results: Vec<_> = if args.parallel {
        workflows.par_iter()
            .map(|path| validate_file(path))
            .collect()
    } else {
        workflows.iter()
            .map(|path| validate_file(path))
            .collect()
    };

    // Collect results
    let mut passed = 0;
    let mut failed = Vec::new();

    for (path, result) in workflows.iter().zip(results) {
        match result {
            Ok(()) => {
                println!("✓ {}", path.display());
                passed += 1;
            }
            Err(errors) => {
                println!("✗ {}", path.display());
                failed.push((path.clone(), errors));

                if args.fail_fast {
                    break;
                }
            }
        }
    }

    // Print summary
    println!("\n{}", "━".repeat(60));
    println!("Results: {} passed, {} failed ({:.1}%)",
        passed, failed.len(),
        (passed as f64 / total as f64) * 100.0);
    println!("{}", "━".repeat(60));

    // Print failures
    if !failed.is_empty() {
        println!("\nFailures:\n");
        for (path, errors) in failed {
            // Print with miette
            eprintln!("{:?}", miette::Report::new(errors));
        }
    }

    Ok(())
}
```
