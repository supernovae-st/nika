- Hover documentation

- VS Code extension with syntax highlighting

```rust
        self.client
            .log_message(MessageType::INFO, "Nika LSP initialized")
            .await;
    }

        Ok(())
    }

    async fn did_open(&self, params: DidOpenTextDocumentParams) {
        let uri = params.text_document.uri;
        let content = params.text_document.text;
        let version = params.text_document.version;

        // Store document
        self.documents.insert(
            uri.clone(),
            DocumentState::new(content.clone(), version),
        );

        // Request validation
        let _ = self.validation_tx.send(ValidationRequest {
            uri,
            content,
            version,
        }).await;
    }

    async fn did_change(&self, params: DidChangeTextDocumentParams) {
        let uri = params.text_document.uri;
        let version = params.text_document.version;

        if let Some(mut doc) = self.documents.get_mut(&uri) {
            // Apply incremental changes
            for change in params.content_changes {
                doc.apply_change(&change);
            }

            // Request validation
            let _ = self.validation_tx.send(ValidationRequest {
                uri: uri.clone(),
                content: doc.content.clone(),
                version,
            }).await;
        }
    }

    async fn did_close(&self, params: DidCloseTextDocumentParams) {
        self.documents.remove(&params.text_document.uri);
    }

    async fn completion(&self, params: CompletionParams) -> Result<Option<CompletionResponse>> {
        crate::completion::handle_completion(&self.documents, params).await
    }

    async fn hover(&self, params: HoverParams) -> Result<Option<Hover>> {
        crate::hover::handle_hover(&self.documents, params).await
    }

    async fn goto_definition(
        &self,
        params: GotoDefinitionParams,
    ) -> Result<Option<GotoDefinitionResponse>> {
        crate::definition::handle_definition(&self.documents, params).await
    }
}

/// Validation worker with debouncing
async fn validation_worker(mut rx: mpsc::Receiver<ValidationRequest>, client: Client) {
    use std::collections::HashMap;
    use std::time::{Duration, Instant};

    const DEBOUNCE_MS: u64 = 50;
    let mut pending: HashMap<Url, (ValidationRequest, Instant)> = HashMap::new();

    loop {
        let timeout = if pending.is_empty() {
            Duration::from_secs(3600)
        } else {
            Duration::from_millis(DEBOUNCE_MS)
        };

        tokio::select! {
            Some(req) = rx.recv() => {
                pending.insert(req.uri.clone(), (req, Instant::now()));
            }
            _ = tokio::time::sleep(timeout) => {
                let now = Instant::now();
                let ready: Vec<Url> = pending
                    .iter()
                    .filter(|(_, (_, t))| now.duration_since(*t) >= Duration::from_millis(DEBOUNCE_MS))
                    .map(|(uri, _)| uri.clone())
                    .collect();

                for uri in ready {
                    if let Some((req, _)) = pending.remove(&uri) {
                        let diagnostics = crate::diagnostics::validate(&req.content, &uri);
                        client.publish_diagnostics(uri, diagnostics, Some(req.version)).await;
                    }
                }
            }
        }
    }
}
```

```rust
//! Document state management

use lsp_types::{Position, Range, TextDocumentContentChangeEvent};
use ropey::Rope;

#[derive(Clone)]
pub struct DocumentState {
    /// Document content
    pub content: String,
    /// LSP version
    pub version: i32,
    /// Rope for efficient edits
    rope: Rope,
    /// Cached line starts
    line_starts: Vec<usize>,
}

impl DocumentState {
    pub fn new(content: String, version: i32) -> Self {
        let rope = Rope::from_str(&content);
        let line_starts = compute_line_starts(&content);

        Self {
            content,
            version,
            rope,
            line_starts,
        }
    }

    /// Apply incremental change
    pub fn apply_change(&mut self, change: &TextDocumentContentChangeEvent) {
        match change.range {
            Some(range) => {
                // Incremental update
                let start_offset = self.position_to_offset(range.start);
                let end_offset = self.position_to_offset(range.end);

                // Update rope
                self.rope.remove(start_offset..end_offset);
                self.rope.insert(start_offset, &change.text);

                // Update content
                self.content = self.rope.to_string();
            }
            None => {
                // Full replacement
                self.content = change.text.clone();
                self.rope = Rope::from_str(&self.content);
            }
        }

        // Rebuild line starts
        self.line_starts = compute_line_starts(&self.content);
    }

    /// Convert LSP Position to byte offset
    pub fn position_to_offset(&self, pos: Position) -> usize {
        let line = pos.line as usize;
        if line >= self.line_starts.len() {
            return self.content.len();
        }

        let line_start = self.line_starts[line];
        let col = pos.character as usize;

        // Handle UTF-16 code units (LSP uses UTF-16)
        let line_content = &self.content[line_start..];
        let end = line_content.find('\n').unwrap_or(line_content.len());
        let line_str = &line_content[..end];

        // Convert UTF-16 offset to byte offset
        let mut utf16_offset = 0;
        let mut byte_offset = 0;

        for ch in line_str.chars() {
            if utf16_offset >= col {
                break;
            }
            utf16_offset += ch.len_utf16();
            byte_offset += ch.len_utf8();
        }

        line_start + byte_offset
    }

    /// Convert byte offset to LSP Position
    pub fn offset_to_position(&self, offset: usize) -> Position {
        let line = self.line_starts.partition_point(|&start| start <= offset).saturating_sub(1);
        let line_start = self.line_starts.get(line).copied().unwrap_or(0);

        let col_bytes = offset.saturating_sub(line_start);
        let line_content = &self.content[line_start..];
        let col_str = &line_content[..col_bytes.min(line_content.len())];

        // Convert byte offset to UTF-16 code units
        let col_utf16: u32 = col_str.chars().map(|c| c.len_utf16() as u32).sum();

        Position {
            line: line as u32,
            character: col_utf16,
        }
    }

    /// Get the line at given index
    pub fn line(&self, idx: usize) -> Option<&str> {
        if idx >= self.line_starts.len() {
            return None;
        }

        let start = self.line_starts[idx];
        let end = self.line_starts
            .get(idx + 1)
            .map(|&e| e.saturating_sub(1))
            .unwrap_or(self.content.len());

        Some(&self.content[start..end])
    }
}

fn compute_line_starts(content: &str) -> Vec<usize> {
    let mut starts = vec![0];
    for (i, c) in content.char_indices() {
        if c == '\n' {
            starts.push(i + 1);
        }
    }
    starts
}

#[cfg(test)]
mod tests {
```

```rust
//! Convert Nika validation errors to LSP diagnostics

use lsp_types::{Diagnostic, DiagnosticSeverity, NumberOrString, Position, Range, Url};
use nika::validation::{self, ValidationErrors};

/// Validate document and return LSP diagnostics
pub fn validate(content: &str, uri: &Url) -> Vec<Diagnostic> {
    let path = uri.path();

    match validation::phases::validate(path, content) {
        Ok(()) => vec![],
        Err(errors) => convert_errors(&errors, content),
    }
}

fn convert_errors(errors: &ValidationErrors, content: &str) -> Vec<Diagnostic> {
    let mut diagnostics = Vec::new();

    for error in &errors.errors {
        // Extract error info from miette diagnostic
        let message = error.to_string();
        let code = error.code().map(|c| c.to_string());

        // Get source span
        let range = if let Some(labels) = error.labels() {
            labels
                .into_iter()
                .next()
                .map(|label| {
                    let offset = label.offset();
                    let len = label.len();
                    offset_range_to_lsp_range(content, offset, len)
                })
                .unwrap_or_default()
        } else {
            Range::default()
        };

        diagnostics.push(Diagnostic {
            range,
            severity: Some(DiagnosticSeverity::ERROR),
            code: code.map(NumberOrString::String),
            source: Some("nika".to_string()),
            message,
            related_information: None,
            tags: None,
            code_description: None,
            data: None,
        });
    }

    diagnostics
}

fn offset_range_to_lsp_range(content: &str, offset: usize, len: usize) -> Range {
    let start = offset_to_position(content, offset);
    let end = offset_to_position(content, offset + len);
    Range { start, end }
}

fn offset_to_position(content: &str, offset: usize) -> Position {
    let mut line = 0u32;
    let mut col = 0u32;

    for (i, c) in content.char_indices() {
        if i >= offset {
            break;
        }
        if c == '\n' {
            line += 1;
            col = 0;
        } else {
            col += c.len_utf16() as u32;
        }
    }

    Position { line, character: col }
}
```

```rust
//! Determine completion context from cursor position

use crate::document::DocumentState;
use lsp_types::Position;

#[derive(Debug, Clone)]
pub enum CompletionContext {
    /// Top-level keys: schema, workflow, tasks, mcp, provider, model
    TopLevel,

    /// Task properties: id, description, infer, exec, fetch, invoke, agent
    TaskProperty,

    /// Verb properties (infer:, exec:, etc.)
    VerbProperty { verb: String },

    /// After "use:" key
    UseBlock { available_tasks: Vec<String> },

    /// After task ID in use block (e.g., "alias: task_id.")
    UsePath { task_id: String },

    /// After "invoke:" - MCP tools
    InvokeValue,

    /// After "provider:" - provider names
    ProviderValue,

    /// After "model:" - model names
    ModelValue { provider: Option<String> },

    /// Inside template {{ }}
    Template { partial: String },

    /// No completion available
    None,
}

pub fn detect_context(doc: &DocumentState, position: Position) -> CompletionContext {
    let line_idx = position.line as usize;
    let col = position.character as usize;

    let line = match doc.line(line_idx) {
        Some(l) => l,
        None => return CompletionContext::None,
    };

    let line_before_cursor = &line[..col.min(line.len())];

    // Calculate indentation
    let indent = line.len() - line.trim_start().len();
    let trimmed = line_before_cursor.trim_start();

    // Check for template context {{ }}
    if let Some(brace_pos) = trimmed.rfind("{{") {
        if !trimmed[brace_pos..].contains("}}") {
            let partial = trimmed[brace_pos + 2..].to_string();
            return CompletionContext::Template { partial };
        }
    }

    // Parse what's before cursor
    if trimmed.is_empty() {
        // Empty line - determine context by indentation
        return context_by_indent(doc, line_idx, indent);
    }

    // After a key with colon
    if let Some(colon_idx) = trimmed.find(':') {
        let key = trimmed[..colon_idx].trim();
        let after_colon = trimmed[colon_idx + 1..].trim();

        if after_colon.is_empty() || !after_colon.contains(':') {
            // We're in the value position
            return match key {
                "provider" => CompletionContext::ProviderValue,
                "model" => {
                    let provider = find_provider_in_doc(doc);
                    CompletionContext::ModelValue { provider }
                }
                "invoke" => CompletionContext::InvokeValue,
                key if indent >= 4 && is_use_key(doc, line_idx) => {
                    let tasks = collect_task_ids(doc);
                    CompletionContext::UseBlock { available_tasks: tasks }
                }
                _ => CompletionContext::None,
            };
        }
    }

    // Determine by indentation
    context_by_indent(doc, line_idx, indent)
}

fn context_by_indent(doc: &DocumentState, line_idx: usize, indent: usize) -> CompletionContext {
    match indent {
        0 => CompletionContext::TopLevel,
        2 if is_in_tasks_array(doc, line_idx) => CompletionContext::TaskProperty,
        4 if is_in_task(doc, line_idx) => {
            if let Some(verb) = current_verb(doc, line_idx) {
                CompletionContext::VerbProperty { verb }
            } else {
                CompletionContext::TaskProperty
            }
        }
        _ => CompletionContext::None,
    }
}

fn collect_task_ids(doc: &DocumentState) -> Vec<String> {
    let mut ids = Vec::new();
    let mut in_tasks = false;

    for line in doc.content.lines() {
        let trimmed = line.trim();
        if trimmed == "tasks:" {
            in_tasks = true;
            continue;
        }
        if in_tasks && trimmed.starts_with("- id:") {
            let id = trimmed.strip_prefix("- id:").unwrap().trim();
            ids.push(id.to_string());
        }
        if in_tasks && !line.starts_with("  ") && !line.starts_with("\t") && !trimmed.is_empty() {
            in_tasks = false;
        }
    }

    ids
}

fn find_provider_in_doc(doc: &DocumentState) -> Option<String> {
    for line in doc.content.lines() {
        let trimmed = line.trim();
        if trimmed.starts_with("provider:") {
            return Some(trimmed.strip_prefix("provider:")?.trim().to_string());
        }
    }
    None
}
```

//! Hover documentation

use crate::document::DocumentState;
use dashmap::DashMap;
use lsp_types::*;
use tower_lsp::jsonrpc::Result;

pub async fn handle_hover(
    documents: &DashMap<Url, DocumentState>,
    params: HoverParams,
) -> Result<Option<Hover>> {
    let uri = &params.text_document_position_params.text_document.uri;
    let position = params.text_document_position_params.position;

    let doc = match documents.get(uri) {
        Some(d) => d,
        None => return Ok(None),
    };

    let line = match doc.line(position.line as usize) {
        Some(l) => l,
        None => return Ok(None),
    };

    // Find the word under cursor
    let col = position.character as usize;
    let word = extract_word(line, col);

    let content = get_hover_content(&word);

```rust
//! Go to definition

use crate::document::DocumentState;
use dashmap::DashMap;
use lsp_types::*;
use tower_lsp::jsonrpc::Result;

pub async fn handle_definition(
    documents: &DashMap<Url, DocumentState>,
    params: GotoDefinitionParams,
) -> Result<Option<GotoDefinitionResponse>> {
    let uri = &params.text_document_position_params.text_document.uri;
    let position = params.text_document_position_params.position;

    let doc = match documents.get(uri) {
        Some(d) => d,
        None => return Ok(None),
    };

    let line = match doc.line(position.line as usize) {
        Some(l) => l,
        None => return Ok(None),
    };

    // Find the word under cursor
    let col = position.character as usize;
    let word = extract_word(line, col);

    // Check if this is a task reference (in use: or flow:)
    if let Some(task_line) = find_task_definition(&doc.content, &word) {
        return Ok(Some(GotoDefinitionResponse::Scalar(Location {
            uri: uri.clone(),
            range: Range {
                start: Position { line: task_line as u32, character: 0 },
                end: Position { line: task_line as u32, character: 0 },
            },
        })));
    }

    Ok(None)
}

fn extract_word(line: &str, col: usize) -> String {
    let chars: Vec<char> = line.chars().collect();
    let col = col.min(chars.len());

    let mut start = col;
    while start > 0 && is_word_char(chars[start - 1]) {
        start -= 1;
    }

    let mut end = col;
    while end < chars.len() && is_word_char(chars[end]) {
        end += 1;
    }

    chars[start..end].iter().collect()
}

fn is_word_char(c: char) -> bool {
    c.is_alphanumeric() || c == '_' || c == '-'
}

fn find_task_definition(content: &str, task_id: &str) -> Option<usize> {
    let target = format!("- id: {}", task_id);

    for (i, line) in content.lines().enumerate() {
        if line.trim() == target || line.trim().starts_with(&format!("- id: {}", task_id)) {
            return Some(i);
        }
    }

    None
}
```

```typescript
import * as vscode from 'vscode';
import {
  LanguageClient,
  LanguageClientOptions,
  ServerOptions,
} from 'vscode-languageclient/node';

let client: LanguageClient;

export function activate(context: vscode.ExtensionContext) {
  const config = vscode.workspace.getConfiguration('nika');
  const lspPath = config.get<string>('lsp.path', 'nika-lsp');

  const serverOptions: ServerOptions = {
    command: lspPath,
    args: [],
  };

  const clientOptions: LanguageClientOptions = {
    documentSelector: [
      { scheme: 'file', language: 'nika' },
      { scheme: 'file', pattern: '**/*.nika.yaml' },
      { scheme: 'file', pattern: '**/*.nika.yml' },
    ],
    synchronize: {
      fileEvents: vscode.workspace.createFileSystemWatcher('**/*.nika.{yaml,yml}'),
    },
  };

  client = new LanguageClient(
    'nika',
    'Nika Language Server',
    serverOptions,
    clientOptions
  );

  // Register commands
  context.subscriptions.push(
    vscode.commands.registerCommand('nika.validate', () => {
      const editor = vscode.window.activeTextEditor;
      if (editor) {
        vscode.window.showInformationMessage('Validating...');
        // Trigger validation via LSP
      }
    })
  );

  // Start the client
  client.start();
}

export function deactivate(): Thenable<void> | undefined {
  if (!client) {
    return undefined;
  }
  return client.stop();
}
```
