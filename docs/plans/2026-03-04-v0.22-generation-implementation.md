```rust
use lsp_types::{
    CodeAction, CodeActionKind, CodeActionOrCommand, CodeActionParams,
    Diagnostic, TextEdit, WorkspaceEdit,
};
use std::collections::HashMap;

/// Registry of code action providers by error code
pub struct CodeActionRegistry {
    providers: HashMap<String, Box<dyn CodeActionProvider>>,
}

impl CodeActionRegistry {
    pub fn new() -> Self {
        let mut providers: HashMap<String, Box<dyn CodeActionProvider>> = HashMap::new();

        // Register providers for each error code
        providers.insert("NIKA-001".to_string(), Box::new(UnknownTaskProvider));
        providers.insert("NIKA-003".to_string(), Box::new(InvalidVerbProvider));
        providers.insert("NIKA-004".to_string(), Box::new(InvalidTaskIdProvider));
        providers.insert("NIKA-005".to_string(), Box::new(UnresolvedTemplateProvider));
        providers.insert("NIKA-007".to_string(), Box::new(UnknownProviderProvider));
        providers.insert("NIKA-008".to_string(), Box::new(UnknownModelProvider));

        Self { providers }
    }

    pub fn get_actions(
        &self,
        diagnostic: &Diagnostic,
        document: &DocumentState,
    ) -> Vec<CodeAction> {
        let code = diagnostic.code.as_ref()
            .and_then(|c| match c {
                lsp_types::NumberOrString::String(s) => Some(s.as_str()),
                _ => None,
            });

        match code {
            Some(code) => {
                if let Some(provider) = self.providers.get(code) {
                    provider.provide(diagnostic, document)
                } else {
                    vec![]
                }
            }
            None => vec![],
        }
    }
}

pub trait CodeActionProvider: Send + Sync {
    fn provide(&self, diagnostic: &Diagnostic, document: &DocumentState) -> Vec<CodeAction>;
}

/// NIKA-001: Unknown task reference
/// Fixes: Replace with suggestion, or create new task
pub struct UnknownTaskProvider;

impl CodeActionProvider for UnknownTaskProvider {
    fn provide(&self, diagnostic: &Diagnostic, document: &DocumentState) -> Vec<CodeAction> {
        let mut actions = vec![];

        // Extract suggestion from diagnostic message
        if let Some(suggestion) = extract_suggestion(&diagnostic.message) {
            // Fix 1: Replace with suggestion
            actions.push(CodeAction {
                title: format!("Replace with '{}'", suggestion),
                kind: Some(CodeActionKind::QUICKFIX),
                diagnostics: Some(vec![diagnostic.clone()]),
                edit: Some(WorkspaceEdit {
                    changes: Some(HashMap::from([(
                        document.uri.clone(),
                        vec![TextEdit {
                            range: diagnostic.range,
                            new_text: suggestion.clone(),
                        }],
                    )])),
                    ..Default::default()
                }),
                is_preferred: Some(true),
                ..Default::default()
            });
        }

        // Fix 2: Create new task with stub
        if let Some(task_name) = extract_task_name(&diagnostic.message) {
            let stub = generate_task_stub(&task_name);
            let insert_position = find_task_insert_position(document);

            actions.push(CodeAction {
                title: format!("Create task '{}'", task_name),
                kind: Some(CodeActionKind::QUICKFIX),
                diagnostics: Some(vec![diagnostic.clone()]),
                edit: Some(WorkspaceEdit {
                    changes: Some(HashMap::from([(
                        document.uri.clone(),
                        vec![TextEdit {
                            range: lsp_types::Range {
                                start: insert_position,
                                end: insert_position,
                            },
                            new_text: stub,
                        }],
                    )])),
                    ..Default::default()
                }),
                is_preferred: Some(false),
                ..Default::default()
            });
        }

        actions
    }
}

/// Generate a task stub for "create task" action
fn generate_task_stub(task_name: &str) -> String {
    format!(
        r#"
  - id: {task_name}
    description: "TODO: Add description"
    infer:
      prompt: |
        TODO: Add prompt
    output:
      format: text
"#,
        task_name = task_name
    )
}

/// NIKA-003: Invalid verb
/// Fixes: Replace with valid verb suggestions
pub struct InvalidVerbProvider;

impl CodeActionProvider for InvalidVerbProvider {
    fn provide(&self, diagnostic: &Diagnostic, document: &DocumentState) -> Vec<CodeAction> {
        let valid_verbs = ["infer", "exec", "fetch", "invoke", "agent"];
        let invalid_verb = extract_invalid_value(&diagnostic.message);

        valid_verbs
            .iter()

                    changes: Some(HashMap::from([(

                            new_text: verb.to_string(),
                        }],
                    )])),
                    ..Default::default()
                }),
                is_preferred: Some(false),
                ..Default::default()
            })
            .collect()
    }
}

/// NIKA-005: Unresolved template reference
/// Fixes: Add to use block, or create source task
pub struct UnresolvedTemplateProvider;

impl CodeActionProvider for UnresolvedTemplateProvider {
    fn provide(&self, diagnostic: &Diagnostic, document: &DocumentState) -> Vec<CodeAction> {
        let mut actions = vec![];

        // Parse template reference: {{use.alias.field}}
        if let Some((alias, _field)) = parse_template_ref(&diagnostic.message) {
            // Get available tasks that could be sources

                    title: format!("Bind '{}' to task '{}'", alias, task_id),

                    diagnostics: Some(vec![diagnostic.clone()]),
                    edit: Some(generate_use_binding_edit(document, &alias, task_id)),
                    is_preferred: Some(false),
                    ..Default::default()
                });
            }
        }

        actions
    }
}
```

```rust
use lsp_types::{CodeActionParams, CodeActionResponse};
use tower_lsp::jsonrpc::Result;

impl NikaBackend {
    pub async fn handle_code_action(
        &self,
        params: CodeActionParams,
    ) -> Result<Option<CodeActionResponse>> {
        let uri = params.text_document.uri;

        // Get document
        let document = match self.documents.get(&uri) {
            Some(doc) => doc,
            None => return Ok(None),
        };

        // Get diagnostics in range
        let diagnostics: Vec<_> = params
            .context
            .diagnostics
            .iter()
            .filter(|d| ranges_overlap(&d.range, &params.range))
            .collect();

        if diagnostics.is_empty() {
            return Ok(None);
        }

        // Collect code actions for each diagnostic
        let mut actions = vec![];
        for diagnostic in diagnostics {
            let diag_actions = self.code_actions.get_actions(diagnostic, &document);
            actions.extend(diag_actions.into_iter().map(CodeActionOrCommand::CodeAction));
        }

        if actions.is_empty() {
            Ok(None)
        } else {
            Ok(Some(actions))
        }
    }
}
```

```rust
use dashmap::DashMap;
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use std::time::{Duration, Instant};
use tokio::sync::RwLock;

/// MCP Schema Registry with TTL caching
pub struct McpSchemaRegistry {
    /// Server URI -> SchemaCache
    schemas: DashMap<String, SchemaCache>,
    /// Default TTL for cache entries
    ttl: Duration,
    /// MCP client for fetching schemas
    client: Arc<McpClient>,
}

#[derive(Clone)]
struct SchemaCache {
    tools: Vec<McpToolSchema>,
    fetched_at: Instant,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct McpToolSchema {
    pub name: String,
    pub description: Option<String>,
    pub input_schema: serde_json::Value,
}

    /// Get tools for a server, fetching if not cached or expired

            server_uri.to_string(),

                tools: tools.clone(),
                fetched_at: Instant::now(),
            },
        );

        Ok(tools)
    }

/// Format tool documentation as Markdown
fn format_tool_docs(tool: &McpToolSchema) -> String {
    let mut docs = String::new();

    if let Some(desc) = &tool.description {
        docs.push_str(&format!("{}\n\n", desc));
    }

    docs.push_str("**Parameters:**\n\n");

    if let Some(props) = tool.input_schema.get("properties") {
        if let Some(obj) = props.as_object() {
            for (name, schema) in obj {
                let type_str = schema.get("type")
                    .and_then(|t| t.as_str())
                    .unwrap_or("any");
                let desc = schema.get("description")
                    .and_then(|d| d.as_str())
                    .unwrap_or("");

                docs.push_str(&format!("- `{}` ({}) - {}\n", name, type_str, desc));
            }
        }
    }

    docs
}

/// Generate a snippet for tool invocation
fn generate_tool_snippet(tool: &McpToolSchema) -> String {
    let mut snippet = format!("{}\n    params:", tool.name);

    if let Some(props) = tool.input_schema.get("properties") {
        if let Some(obj) = props.as_object() {
            let required: Vec<&str> = tool.input_schema
                .get("required")
                .and_then(|r| r.as_array())
                .map(|arr| arr.iter().filter_map(|v| v.as_str()).collect())
                .unwrap_or_default();

            let mut placeholder_idx = 1;
            for (name, schema) in obj {
                if required.contains(&name.as_str()) {
                    let default = get_default_placeholder(schema, placeholder_idx);
                    snippet.push_str(&format!("\n      {}: ${{{}}}", name, default));
                    placeholder_idx += 1;
                }
            }
        }
    }

    snippet
}

fn get_default_placeholder(schema: &serde_json::Value, idx: usize) -> String {
    let type_str = schema.get("type").and_then(|t| t.as_str()).unwrap_or("string");

    match type_str {
        "string" => format!("{}:\"${{{}}}\"", idx, idx),
        "number" | "integer" => format!("{}:0", idx),
        "boolean" => format!("{}:true", idx),
        _ => format!("{}:{}", idx, idx),
    }
}
```

```rust
impl SemanticTokenProvider {
    pub fn provide(&self, document: &DocumentState) -> SemanticTokensResult {
        let mut tokens = vec![];
        let mut prev_line = 0u32;
        let mut prev_start = 0u32;

        let ast = match &document.ast {
            Some(ast) => ast,
            None => return SemanticTokensResult::Tokens(SemanticTokens {
                result_id: None,
                data: vec![],
            }),
        };

        // Collect token locations from AST
        let mut token_infos = vec![];

        // Walk AST to find tokenizable elements
        self.collect_tokens(ast, &mut token_infos);

        // Sort by position
        token_infos.sort_by_key(|t| (t.line, t.start));

        // Convert to LSP delta format
        for info in token_infos {
            let delta_line = info.line - prev_line;
            let delta_start = if delta_line == 0 {
                info.start - prev_start
            } else {
                info.start
            };

            tokens.push(SemanticToken {
                delta_line,
                delta_start,
                length: info.length,
                token_type: info.token_type,
                token_modifiers_bitset: info.modifiers,
            });

            prev_line = info.line;
            prev_start = info.start;
        }

        SemanticTokensResult::Tokens(SemanticTokens {
            result_id: Some(format!("{}", document.version)),
            data: tokens,
        })
    }

    fn collect_tokens(&self, ast: &NikaAst, tokens: &mut Vec<TokenInfo>) {
        // Workflow-level keys
        if let Some(span) = ast.get_span("schema") {
            tokens.push(TokenInfo::property(span));
        }
        if let Some(span) = ast.get_span("workflow") {
            tokens.push(TokenInfo::property(span));
        }

        // Provider and model
        if let Some((span, _)) = ast.get_provider() {
            tokens.push(TokenInfo::provider(span));
        }
        if let Some((span, _)) = ast.get_model() {
            tokens.push(TokenInfo::model(span));
        }

        // Tasks
        for task in ast.tasks() {
            // Task ID (definition)
            if let Some(span) = task.id_span() {
                tokens.push(TokenInfo::task_id_def(span));
            }

            // Verb
            if let Some((span, _)) = task.verb() {
                tokens.push(TokenInfo::verb(span));
            }

            // use: references
            for (alias_span, target) in task.use_bindings() {
                tokens.push(TokenInfo::task_ref(alias_span));
                if let Some(target_span) = task.get_target_span(target) {
                    tokens.push(TokenInfo::task_ref(target_span));
                }
            }

            // Templates in strings
            for template_span in task.template_refs() {

            }

            // invoke: tool name
            if let Some((span, _)) = task.invoke_tool() {
                tokens.push(TokenInfo::mcp_tool(span));
            }
        }
```

```rust
use lsp_types::{DocumentFormattingParams, TextEdit, Position, Range};

pub struct WorkflowFormatter {
    options: FormatterOptions,
}

#[derive(Clone)]
pub struct FormatterOptions {
    pub indent_size: usize,
    pub align_values: bool,
    pub sort_tasks_by_deps: bool,
    pub max_line_length: usize,
    pub preserve_comments: bool,
}

impl Default for FormatterOptions {
    fn default() -> Self {
        Self {
            indent_size: 2,
            align_values: false,
            sort_tasks_by_deps: false,
            max_line_length: 100,
            preserve_comments: true,
        }
    }
}

impl WorkflowFormatter {
    pub fn new(options: FormatterOptions) -> Self {
        Self { options }
    }

    pub fn format(&self, document: &DocumentState) -> Vec<TextEdit> {
        let content = &document.content;

        // Parse YAML preserving comments
        let parsed = match parse_yaml_with_comments(content) {
            Ok(p) => p,
            Err(_) => return vec![], // Don't format invalid YAML
        };

        // Build formatted output
        let mut output = String::new();
        let mut current_line = 0;

        // Format top-level keys in order
        let top_level_order = [
            "schema", "workflow", "description",
            "provider", "model", "mcp", "context", "inputs", "tasks"
        ];

        for key in top_level_order {
            if let Some(node) = parsed.get(key) {
                // Prepend comments that belong to this key
                if let Some(comments) = parsed.comments_before(key) {
                    for comment in comments {
                        output.push_str(&comment);
                        output.push('\n');
                    }
                }

                // Format the key-value pair
                output.push_str(&self.format_node(key, node, 0));
                output.push('\n');

                // Add empty line after certain keys
                if key == "model" || key == "inputs" || key == "mcp" {
                    output.push('\n');
                }
            }
        }

        // Return single edit replacing entire document
        let line_count = content.lines().count();

        vec![TextEdit {
            range: Range {
                start: Position { line: 0, character: 0 },
                end: Position {
                    line: line_count as u32,
                    character: 0,
                },
            },
            new_text: output,
        }]
    }

    fn format_node(&self, key: &str, node: &YamlNode, indent: usize) -> String {
        let indent_str = " ".repeat(indent);

        match node {
            YamlNode::Scalar(s) => {
                if s.contains('\n') {
                    // Multiline string
                    format!("{}{}:\n{}", indent_str, key, self.format_multiline(s, indent + self.options.indent_size))
                } else if s.len() > self.options.max_line_length - indent - key.len() - 2 {
                    // Long string, wrap
                    format!("{}{}:\n{}", indent_str, key, self.format_wrapped(s, indent + self.options.indent_size))
                } else {
                    format!("{}{}: {}", indent_str, key, self.quote_if_needed(s))
                }
            }
            YamlNode::Sequence(items) => {
                let mut output = format!("{}{}:", indent_str, key);
                for item in items {
                    output.push('\n');
                    output.push_str(&self.format_sequence_item(item, indent + self.options.indent_size));
                }
                output
            }
            YamlNode::Mapping(map) => {
                let mut output = format!("{}{}:", indent_str, key);
                for (k, v) in map {
                    output.push('\n');
                    output.push_str(&self.format_node(k, v, indent + self.options.indent_size));
                }
                output
            }
        }
    }

    fn format_sequence_item(&self, item: &YamlNode, indent: usize) -> String {
        let indent_str = " ".repeat(indent);

        match item {
            YamlNode::Mapping(map) => {
                let mut output = String::new();
                let mut first = true;

                // Order task keys
                let task_key_order = [
                    "id", "description", "use", "infer", "exec", "fetch", "invoke", "agent",
                    "flow", "output", "retry", "timeout"
                ];

                for key in task_key_order {
                    if let Some(value) = map.get(key) {
                        if first {
                            output.push_str(&format!("{}- ", indent_str));
                            first = false;
                        } else {
                            output.push_str(&format!("{}  ", indent_str));
                        }
                        output.push_str(&self.format_node(key, value, 0).trim_start());
                        output.push('\n');
                    }
                }

                output.trim_end().to_string()
            }
            _ => format!("{}- {}", indent_str, self.format_scalar(item)),
        }
    }

    fn format_multiline(&self, s: &str, indent: usize) -> String {
        let indent_str = " ".repeat(indent);
        let mut output = format!("{}|\n", indent_str);

        for line in s.lines() {
            output.push_str(&format!("{}{}\n", indent_str, line));
        }

        output
    }

    fn format_wrapped(&self, s: &str, indent: usize) -> String {
        let indent_str = " ".repeat(indent);
        let mut output = format!("{}>\n", indent_str);

        // Word wrap at max_line_length - indent
        let max_width = self.options.max_line_length - indent;
        let mut current_line = String::new();

        for word in s.split_whitespace() {
            if current_line.len() + word.len() + 1 > max_width {
                output.push_str(&format!("{}{}\n", indent_str, current_line));
                current_line = word.to_string();
            } else {
                if !current_line.is_empty() {
                    current_line.push(' ');
                }
                current_line.push_str(word);
            }
        }

        if !current_line.is_empty() {
            output.push_str(&format!("{}{}\n", indent_str, current_line));
        }

        output
    }

    fn quote_if_needed(&self, s: &str) -> String {
        // Quote strings that need it
        let needs_quote = s.starts_with(|c: char| c.is_whitespace())
            || s.ends_with(|c: char| c.is_whitespace())
            || s.contains(':')
            || s.contains('#')
            || s.starts_with('*')
            || s.starts_with('&')
            || s.starts_with('!')
            || s.starts_with('@')
            || s.starts_with('%')
            || s == "true"
            || s == "false"
            || s == "null"
            || s.parse::<f64>().is_ok();

        if needs_quote {
            format!("\"{}\"", s.replace('"', "\\\""))
        } else {
            s.to_string()
        }
    }

    fn format_scalar(&self, node: &YamlNode) -> String {
        match node {
            YamlNode::Scalar(s) => self.quote_if_needed(s),
            _ => "<complex>".to_string(),
        }
    }
}
```
