```rust
fn breadcrumb(&self) -> String {
    let home = dirs::home_dir().unwrap_or_default();
    let path = self.root_dir.display().to_string();

    // Remplacer home par ~
    let short = if path.starts_with(home.to_str().unwrap_or("")) {
        path.replace(home.to_str().unwrap_or(""), "~")
    } else {
        path
    };

    // Tronquer si trop long (max 20 chars)
    if short.len() > 20 {
        format!("📁 ..{}", &short[short.len()-17..])
    } else {
        format!("📁 {}", short)
    }
}
```

### 2.2 Rendu

```rust
fn render(&mut self, frame: &mut Frame, area: Rect, items: &[TaskListItem], theme: &Theme, focused: bool) {
    let block = Block::default()
        .title("Tasks")
        .borders(Borders::ALL)
        .border_style(if focused { theme.border_focused } else { theme.border });

    let inner = block.inner(area);
    frame.render_widget(block, area);

    // Render task list
    let visible_height = inner.height as usize;
    let start = self.scroll_offset;
    let end = (start + visible_height).min(items.len());

    for (i, item) in items[start..end].iter().enumerate() {
        let y = inner.y + i as u16;
        let is_selected = start + i == self.selected;

        // Icon based on state
        let icon = match item.state {
            BoxState::Queued => "○",
            BoxState::Running { .. } => item.state.spinner_char(),
            BoxState::Success { .. } => "✓",
            BoxState::Failed { .. } => "✗",
            BoxState::Skipped { .. } => "⊘",
        };

        // Color based on verb
        let color = item.verb.color();

        let line = if is_selected {
            format!("► {} {}", icon, item.id)
        } else {
            format!("  {} {}", icon, item.id)
        };

        let style = Style::default()
            .fg(color)
            .add_modifier(if is_selected { Modifier::BOLD } else { Modifier::empty() });

        frame.render_widget(
            Paragraph::new(line).style(style),
            Rect::new(inner.x, y, inner.width, 1),
        );
    }
}
```

### 3.2 Rendu Stack Vertical

```rust
fn render(
    &mut self,
    frame: &mut Frame,
    area: Rect,
    task_boxes: &[TaskBox],
    streaming_ctx: &StreamingContext,
    theme: &Theme,
    focused: bool,
) {
    let block = Block::default()
        .title(if self.auto_scroll { "TaskBox [AUTO]" } else { "TaskBox [PAUSED]" })
        .borders(Borders::ALL)
        .border_style(if focused { theme.border_focused } else { theme.border });

    let inner = block.inner(area);
    frame.render_widget(block, area);

    // Calculate total height needed
    let mut total_height = 0;
    let heights: Vec<u16> = task_boxes.iter()
        .map(|tb| tb.required_height() + 1)  // +1 for spacing
        .collect();

    for h in &heights {
        total_height += h;
    }

    // Render visible TaskBoxes
    let mut y = inner.y as i32 - self.scroll_offset as i32;

    for (i, task_box) in task_boxes.iter().enumerate() {
        let height = heights[i];

        // Skip if completely above viewport
        if y + height as i32 <= inner.y as i32 {
            y += height as i32;
            continue;
        }

        // Stop if completely below viewport
        if y >= (inner.y + inner.height) as i32 {
            break;
        }

        // Render TaskBox
        let box_area = Rect::new(
            inner.x,
            y.max(inner.y as i32) as u16,
            inner.width,
            height.min((inner.y + inner.height) as u16 - y.max(inner.y as i32) as u16),
        );

        // Use to_list_items for rendering
        let items = task_box.to_list_items(streaming_ctx.clone());
        let list = List::new(items);
        frame.render_widget(list, box_area);

        y += height as i32;
    }

    // Scroll indicator if content overflows
    if total_height > inner.height {
        let indicator = if self.scroll_offset > 0 { "▲" } else { " " };
        // ... render scroll indicators
    }
}
```

### 3.3 Auto-scroll Logic

```rust
pub fn follow_running(&mut self, task_boxes: &[TaskBox]) {
    if !self.auto_scroll {
        return;
    }

    // Find first running task
    let mut offset = 0;
    for task_box in task_boxes {
        if matches!(task_box.state(), BoxState::Running { .. }) {
            self.scroll_offset = offset;
            return;
        }
        offset += task_box.required_height() as usize + 1;
    }
}

pub fn handle_key(&mut self, key: KeyEvent) -> TaskFlowAction {
    match key.code {
        KeyCode::Char('j') | KeyCode::Down => {
            self.auto_scroll = false;  // Pause auto-scroll on manual input
            TaskFlowAction::ScrollDown
        }
        KeyCode::Char('k') | KeyCode::Up => {
            self.auto_scroll = false;
            TaskFlowAction::ScrollUp
        }
        KeyCode::Char('G') => {
            self.auto_scroll = true;  // Resume auto-scroll
            TaskFlowAction::ScrollToBottom
        }
        KeyCode::Char('g') => {
            self.auto_scroll = false;
            TaskFlowAction::ScrollToTop
        }
        _ => TaskFlowAction::None,
    }
}
```

## Phase 5: Refactorer MonitorView (Runner)

### 5.1 Nouvelle Structure

**Fichier:** `src/tui/views/monitor.rs` (refactored)

```rust
pub struct MonitorView {
    // Panels (NEW - shared components)
    pub browser: BrowserPanel,
    pub task_list: TaskListPanel,
    pub task_flow: TaskBoxFlow,
    pub info: InfoPanel,

    // Focus
    pub focus: RunnerFocus,

    // TaskBox cache (built from TuiState)
    task_boxes: Vec<TaskBox>,
}

#[derive(Default, Clone, Copy)]
pub enum RunnerFocus {
    #[default]
    Browser,
    TaskList,
    TaskFlow,
    Info,
}

impl RunnerFocus {
    pub fn next(&self) -> Self {
        match self {
            Self::Browser => Self::TaskList,
            Self::TaskList => Self::TaskFlow,
            Self::TaskFlow => Self::Info,
            Self::Info => Self::Browser,
        }
    }

    pub fn prev(&self) -> Self {
        match self {
            Self::Browser => Self::Info,
            Self::TaskList => Self::Browser,
            Self::TaskFlow => Self::TaskList,
            Self::Info => Self::TaskFlow,
        }
    }
}
```

### 5.2 Layout 4 Colonnes

```rust
impl View for MonitorView {
    fn render(&mut self, frame: &mut Frame, area: Rect, state: &TuiState, theme: &Theme) {
        // 4-column horizontal layout
        let chunks = Layout::default()
            .direction(Direction::Horizontal)
            .constraints([
                Constraint::Percentage(20),  // Browser
                Constraint::Percentage(20),  // TaskList
                Constraint::Percentage(40),  // TaskFlow
                Constraint::Percentage(20),  // Info
            ])
            .split(area);

        // Update TaskBoxes from TuiState
        self.sync_task_boxes(state);

        // Build task list items
        let task_items: Vec<TaskListItem> = state.task_order.iter()
            .filter_map(|id| state.tasks.get(id))
            .map(|task| TaskListItem {
                id: task.id.clone(),
                verb: task.verb_type(),
                state: task.box_state(),
            })
            .collect();

        // Build workflow info
        let info = WorkflowInfo {
            name: state.workflow.name.clone(),
            status: state.workflow.status,
            progress: (state.workflow.completed, state.workflow.total),
            duration: state.workflow.duration(),
            tokens_in: state.metrics.tokens_in,
            tokens_out: state.metrics.tokens_out,
            cost: state.metrics.cost,
        };

        // Streaming context
        let streaming_ctx = StreamingContext {
            render_mode: RenderMode::Expanded,
            is_streaming: state.workflow.is_streaming(),
            partial_response: state.workflow.partial_response(),
            frame: self.frame as u64,
        };

        // Render panels
        self.browser.render(frame, chunks[0], theme, self.focus == RunnerFocus::Browser);
        self.task_list.render(frame, chunks[1], &task_items, theme, self.focus == RunnerFocus::TaskList);
        self.task_flow.render(frame, chunks[2], &self.task_boxes, &streaming_ctx, theme, self.focus == RunnerFocus::TaskFlow);
        self.info.render(frame, chunks[3], &info, theme, self.focus == RunnerFocus::Info);

        // Auto-scroll to running task
        self.task_flow.follow_running(&self.task_boxes);
    }

    fn handle_key(&mut self, key: KeyEvent, state: &mut TuiState) -> ViewAction {
        // Tab cycling
        match (key.code, key.modifiers) {
            (KeyCode::Tab, KeyModifiers::NONE) => {
                self.focus = self.focus.next();
                return ViewAction::None;
            }
            (KeyCode::BackTab, _) => {
                self.focus = self.focus.prev();
                return ViewAction::None;
            }
            _ => {}
        }

        // Dispatch to focused panel
        match self.focus {
            RunnerFocus::Browser => {
                match self.browser.handle_key(key) {
                    BrowserAction::RunWorkflow(path) => {
                        ViewAction::RunWorkflow(path)
                    }
                    _ => ViewAction::None,
                }
            }
            RunnerFocus::TaskList => {
                let count = state.task_order.len();
                match self.task_list.handle_key(key, count) {
                    TaskListAction::SelectChanged(idx) => {
                        // Sync TaskFlow to show selected task
                        if let Some(id) = self.task_list.selected_id(&task_items) {
                            self.task_flow.scroll_to_task(id, &self.task_boxes);
                        }
                        ViewAction::None
                    }
                    _ => ViewAction::None,
                }
            }
            RunnerFocus::TaskFlow => {
                self.task_flow.handle_key(key);
                ViewAction::None
            }
            RunnerFocus::Info => {
                // Info panel: r=retry, e=export
                match key.code {
                    KeyCode::Char('r') => ViewAction::RetryWorkflow,
                    KeyCode::Char('e') => ViewAction::ExportTrace,
                    _ => ViewAction::None,
                }
            }
        }
    }
}
```

### 5.3 Sync TaskBoxes

```rust
impl MonitorView {
    fn sync_task_boxes(&mut self, state: &TuiState) {
        // Rebuild TaskBoxes from TuiState
        self.task_boxes.clear();

        for task_id in &state.task_order {
            if let Some(task) = state.tasks.get(task_id) {
                let task_box = match task.task_type.as_deref() {
                    Some("infer") => TaskBox::Infer(InferBox::from_task(task)),
                    Some("exec") => TaskBox::Exec(ExecBox::from_task(task)),
                    Some("fetch") => TaskBox::Fetch(FetchBox::from_task(task)),
                    Some("invoke") => TaskBox::Invoke(InvokeBox::from_task(task)),
                    Some("agent") => TaskBox::Agent(AgentBox::from_task(task)),
                    _ => continue,
                };
                self.task_boxes.push(task_box);
            }
        }
    }
}
```
