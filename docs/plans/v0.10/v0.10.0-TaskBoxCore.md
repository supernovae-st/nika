```rust
/// Kind of node in the Chat DAG
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum NodeKind {
    /// User message
    User,
    /// Assistant response
    Assistant,
    /// Tool call (MCP invoke)
    ToolCall,
    /// System message
    System,
    /// Error state
    Error,
}

impl NodeKind {
    /// Get icon for this node kind
    pub fn icon(&self) -> &'static str {
        match self {
            NodeKind::User => "👤",
            NodeKind::Assistant => "🤖",
            NodeKind::ToolCall => "🔌",
            NodeKind::System => "⚙️",
            NodeKind::Error => "❌",
        }
    }

    /// All node kinds
    pub fn all() -> &'static [NodeKind] {
        &[
            NodeKind::User,
            NodeKind::Assistant,
            NodeKind::ToolCall,
            NodeKind::System,
            NodeKind::Error,
        ]
    }
}
```

```rust
/// State of a node in execution
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub enum NodeState {
    #[default]
    Idle,
    Running,
    Complete,
    Failed,
}

impl NodeState {
    pub fn is_running(&self) -> bool {
        matches!(self, NodeState::Running)
    }

    pub fn border_color(&self) -> Color {
        match self {
            NodeState::Idle => Color::Gray,
            NodeState::Running => Color::Yellow,
            NodeState::Complete => Color::Green,
            NodeState::Failed => Color::Red,
        }
    }
}
```

```rust
/// A box widget representing a node in the Chat DAG
#[derive(Debug, Clone)]
pub struct NodeBox {
    id: String,
    kind: NodeKind,
    label: String,
    index: u32,
    state: NodeState,
    selected: bool,
    animation_tick: u8,
}

impl NodeBox {
    pub fn new(id: &str, kind: NodeKind) -> Self {
        Self {
            id: id.to_string(),
            kind,
            label: String::new(),
            index: 0,
            state: NodeState::default(),
            selected: false,
            animation_tick: 0,
        }
    }

    pub fn with_label(mut self, label: &str) -> Self {
        self.label = label.to_string();
        self
    }

    pub fn with_index(mut self, index: u32) -> Self {
        self.index = index;
        self
    }

    pub fn with_state(mut self, state: NodeState) -> Self {
        self.state = state;
        self
    }

    pub fn with_selected(mut self, selected: bool) -> Self {
        self.selected = selected;
        self
    }

    // Getters
    pub fn id(&self) -> &str { &self.id }
    pub fn kind(&self) -> NodeKind { self.kind }
    pub fn label(&self) -> &str { &self.label }
    pub fn index(&self) -> u32 { self.index }
    pub fn state(&self) -> NodeState { self.state }
    pub fn selected(&self) -> bool { self.selected }

    /// Advance animation state
    pub fn tick(&mut self) {
        if self.state.is_running() {
            self.animation_tick = self.animation_tick.wrapping_add(1);
        }
    }

    /// Get pulse intensity (0.0 - 1.0) for running animation
    pub fn pulse_intensity(&self) -> f32 {
        if !self.state.is_running() {
            return 0.0;
        }
        // Sine wave pulse
        let t = self.animation_tick as f32 / 30.0;
        (t.sin() + 1.0) / 2.0
    }
}
```

```rust
impl Widget for NodeBox {
    fn render(self, area: Rect, buf: &mut Buffer) {
        // Calculate dimensions
        let width = area.width.min(30);
        let height = 3.min(area.height);

        // Border style based on state
        let border_style = Style::default().fg(self.state.border_color());
        let border_style = if self.selected {
            border_style.add_modifier(Modifier::BOLD)
        } else {
            border_style
        };

        // Draw border
        let block = Block::default()
            .borders(Borders::ALL)
            .border_style(border_style);

        let inner = block.inner(area);
        block.render(area, buf);

        // Draw content: "@N 🔌 Label"
        let content = format!(
            "@{} {} {}",
            self.index,
            self.kind.icon(),
            truncate_label(&self.label, (width - 8) as usize)
        );

        buf.set_string(inner.x, inner.y, &content, Style::default());
    }
}

fn truncate_label(label: &str, max_len: usize) -> String {
    if label.len() <= max_len {
        label.to_string()
    } else {
        format!("{}…", &label[..max_len.saturating_sub(1)])
    }
}
```
