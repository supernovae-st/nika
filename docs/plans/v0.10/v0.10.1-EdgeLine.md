```rust
// src/tui/widgets/edge_line.rs

use ratatui::prelude::*;

/// Position in the terminal grid
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Position {
    pub x: u16,
    pub y: u16,
}

/// A line connecting two nodes in the DAG
#[derive(Debug, Clone)]
pub struct EdgeLine {
    from: Position,
    to: Position,
    label: Option<String>,
    active: bool,
    animation_tick: u8,
}

impl EdgeLine {
    pub fn new(from: Position, to: Position) -> Self {
        Self {
            from,
            to,
            label: None,
            active: false,
            animation_tick: 0,
        }
    }

    pub fn with_label(mut self, label: &str) -> Self {
        self.label = Some(label.to_string());
        self
    }

    pub fn with_active(mut self, active: bool) -> Self {
        self.active = active;
        self
    }

    // Getters
    pub fn from(&self) -> Position { self.from }
    pub fn to(&self) -> Position { self.to }
    pub fn label(&self) -> Option<&str> { self.label.as_deref() }
    pub fn active(&self) -> bool { self.active }

    pub fn is_vertical(&self) -> bool {
        self.from.x == self.to.x
    }

    pub fn is_horizontal(&self) -> bool {
        self.from.y == self.to.y
    }

    pub fn length(&self) -> u16 {
        if self.is_vertical() {
            self.to.y.abs_diff(self.from.y)
        } else {
            self.to.x.abs_diff(self.from.x)
        }
    }

    /// Advance animation state
    pub fn tick(&mut self) {
        if self.active {
            self.animation_tick = self.animation_tick.wrapping_add(1);
        }
    }
}
```

```rust
impl EdgeLine {
    /// Get current position of flow animation dot
    pub fn flow_position(&self) -> Option<Position> {
        if !self.active {
            return None;
        }

        let len = self.length();
        if len == 0 {
            return None;
        }

        // Cycle through edge length
        let offset = (self.animation_tick as u16 % (len * 2)) as i16;
        let offset = if offset > len as i16 {
            (len * 2) as i16 - offset
        } else {
            offset
        };

        if self.is_vertical() {
            let y = if self.from.y < self.to.y {
                self.from.y + offset as u16
            } else {
                self.from.y - offset as u16
            };
            Some(Position { x: self.from.x, y })
        } else {
            let x = if self.from.x < self.to.x {
                self.from.x + offset as u16
            } else {
                self.from.x - offset as u16
            };
            Some(Position { x, y: self.from.y })
        }
    }
}
```

```rust
impl Widget for EdgeLine {
    fn render(self, area: Rect, buf: &mut Buffer) {
        let style = if self.active {
            Style::default().fg(Color::Cyan)
        } else {
            Style::default().fg(Color::DarkGray)
        };

        if self.is_vertical() {
            let x = self.from.x;
            let (start_y, end_y) = if self.from.y < self.to.y {
                (self.from.y, self.to.y)
            } else {
                (self.to.y, self.from.y)
            };

            // Draw vertical line
            for y in start_y..=end_y {
                if y >= area.y && y < area.y + area.height {
                    buf.get_mut(x, y).set_symbol("│").set_style(style);
                }
            }

            // Draw arrow at end
            if end_y < area.y + area.height {
                buf.get_mut(x, end_y).set_symbol("▼").set_style(style);
            }
        } else {
            let y = self.from.y;
            let (start_x, end_x) = if self.from.x < self.to.x {
                (self.from.x, self.to.x)
            } else {
                (self.to.x, self.from.x)
            };

            // Draw horizontal line
            for x in start_x..=end_x {
                if x >= area.x && x < area.x + area.width {
                    buf.get_mut(x, y).set_symbol("─").set_style(style);
                }
            }

            // Draw arrow at end
            if end_x < area.x + area.width {
                buf.get_mut(end_x, y).set_symbol("▶").set_style(style);
            }
        }

        // Draw label if present
        if let Some(label) = &self.label {
            let mid = self.length() / 2;
            let (lx, ly) = if self.is_vertical() {
                (self.from.x + 1, self.from.y + mid)
            } else {
                (self.from.x + mid, self.from.y - 1)
            };

            if ly < area.y + area.height && lx < area.x + area.width {
                buf.set_string(lx, ly, label, Style::default().fg(Color::Yellow));
            }
        }

        // Draw flow dot if active
        if let Some(pos) = self.flow_position() {
            if pos.y >= area.y && pos.y < area.y + area.height
               && pos.x >= area.x && pos.x < area.x + area.width {
                buf.get_mut(pos.x, pos.y).set_symbol("●").set_style(
                    Style::default().fg(Color::Cyan).add_modifier(Modifier::BOLD)
                );
            }
        }
    }
}
```
