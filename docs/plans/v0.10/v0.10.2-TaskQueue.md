```rust
// src/tui/widgets/task_queue.rs

use ratatui::prelude::*;
use std::time::Duration;

/// State of a task in the queue
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub enum TaskState {
    #[default]
    Pending,
    Running,
    Complete,
    Failed,
}

impl TaskState {
    pub fn icon(&self) -> &'static str {
        match self {
            TaskState::Pending => "◦",
            TaskState::Running => "⠹",
            TaskState::Complete => "✓",
            TaskState::Failed => "✗",
        }
    }

    pub fn color(&self) -> Color {
        match self {
            TaskState::Pending => Color::Gray,
            TaskState::Running => Color::Yellow,
            TaskState::Complete => Color::Green,
            TaskState::Failed => Color::Red,
        }
    }
}

/// Verb type for the task
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum TaskVerb {
    Infer,
    Exec,
    Fetch,
    Invoke,
    Agent,
}

impl TaskVerb {
    pub fn icon(&self) -> &'static str {
        match self {
            TaskVerb::Infer => "⚡",
            TaskVerb::Exec => "📟",
            TaskVerb::Fetch => "🛰️",
            TaskVerb::Invoke => "🔌",
            TaskVerb::Agent => "🐔",
        }
    }
}

/// A single item in the task queue
#[derive(Debug, Clone)]
pub struct TaskQueueItem {
    id: String,
    verb: TaskVerb,
    state: TaskState,
    elapsed: Option<Duration>,
    progress: f32, // 0.0 - 1.0
}

impl TaskQueueItem {
    pub fn new(id: &str, verb: TaskVerb) -> Self {
        Self {
            id: id.to_string(),
            verb,
            state: TaskState::default(),
            elapsed: None,
            progress: 0.0,
        }
    }

    pub fn with_state(mut self, state: TaskState) -> Self {
        self.state = state;
        self
    }

    pub fn with_progress(mut self, progress: f32) -> Self {
        self.progress = progress.clamp(0.0, 1.0);
        self
    }

    pub fn with_elapsed(mut self, elapsed: Duration) -> Self {
        self.elapsed = Some(elapsed);
        self
    }

    // Getters
    pub fn id(&self) -> &str { &self.id }
    pub fn verb(&self) -> TaskVerb { self.verb }
    pub fn state(&self) -> TaskState { self.state }
    pub fn progress(&self) -> f32 { self.progress }
    pub fn elapsed(&self) -> Option<Duration> { self.elapsed }
}
```

```rust
/// A queue of tasks with state tracking
#[derive(Debug, Clone, Default)]
pub struct TaskQueue {
    items: Vec<TaskQueueItem>,
    scroll_offset: usize,
    title: String,
}

impl TaskQueue {
    pub fn new() -> Self {
        Self {
            items: Vec::new(),
            scroll_offset: 0,
            title: "TASKS".to_string(),
        }
    }

    pub fn with_title(mut self, title: &str) -> Self {
        self.title = title.to_string();
        self
    }

    pub fn add(&mut self, item: TaskQueueItem) {
        self.items.push(item);
    }

    pub fn items(&self) -> &[TaskQueueItem] {
        &self.items
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.items.is_empty()
    }

    pub fn update_state(&mut self, id: &str, state: TaskState) {
        if let Some(item) = self.items.iter_mut().find(|i| i.id == id) {
            item.state = state;
        }
    }

    pub fn update_progress(&mut self, id: &str, progress: f32) {
        if let Some(item) = self.items.iter_mut().find(|i| i.id == id) {
            item.progress = progress.clamp(0.0, 1.0);
        }
    }

    /// Get count of completed tasks
    pub fn completed_count(&self) -> usize {
        self.items.iter().filter(|i| i.state == TaskState::Complete).count()
    }

    /// Scroll up
    pub fn scroll_up(&mut self) {
        self.scroll_offset = self.scroll_offset.saturating_sub(1);
    }

    /// Scroll down
    pub fn scroll_down(&mut self, visible_height: usize) {
        let max_offset = self.items.len().saturating_sub(visible_height);
        self.scroll_offset = (self.scroll_offset + 1).min(max_offset);
    }
}
```

```rust
impl Widget for TaskQueue {
    fn render(self, area: Rect, buf: &mut Buffer) {
        // Draw border with title
        let header = format!(" {} {}/{} ",
            self.title,
            self.completed_count(),
            self.items.len()
        );

        let block = Block::default()
            .borders(Borders::ALL)
            .title(header);

        let inner = block.inner(area);
        block.render(area, buf);

        if self.items.is_empty() {
            buf.set_string(
                inner.x + 1,
                inner.y,
                "No tasks",
                Style::default().fg(Color::DarkGray)
            );
            return;
        }

        // Render visible items
        let visible_count = inner.height as usize;
        for (i, item) in self.items.iter()
            .skip(self.scroll_offset)
            .take(visible_count)
            .enumerate()
        {
            let y = inner.y + i as u16;
            let style = Style::default().fg(item.state.color());

            // Format: "✓ task_name    ⚡ infer   2.1s   ██████████"
            let elapsed_str = item.elapsed
                .map(|d| format!("{:.1}s", d.as_secs_f32()))
                .unwrap_or_else(|| "—".to_string());

            let progress_bar = render_progress_bar(item.progress, 10);

            let line = format!(
                " {} {:<16} {} {:<6} {:>6}  {}  {:>3}%",
                item.state.icon(),
                truncate(&item.id, 16),
                item.verb.icon(),
                verb_name(item.verb),
                elapsed_str,
                progress_bar,
                (item.progress * 100.0) as u32,
            );

            buf.set_string(inner.x, y, &line, style);
        }
    }
}

fn truncate(s: &str, max: usize) -> String {
    if s.len() <= max {
        s.to_string()
    } else {
        format!("{}…", &s[..max - 1])
    }
}

fn verb_name(verb: TaskVerb) -> &'static str {
    match verb {
        TaskVerb::Infer => "infer",
        TaskVerb::Exec => "exec",
        TaskVerb::Fetch => "fetch",
        TaskVerb::Invoke => "invoke",
        TaskVerb::Agent => "agent",
    }
}

fn render_progress_bar(progress: f32, width: usize) -> String {
    let filled = (progress * width as f32) as usize;
    let empty = width - filled;
    format!("{}{}", "█".repeat(filled), "░".repeat(empty))
}
```
