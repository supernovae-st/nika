```rust
// src/tui/widgets/tree_node.rs

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum NodeType {
    Directory,
    NikaWorkflow,    // .nika.yaml
    NikaConfig,      // .nika/ folder
    YamlFile,
    File,
}

impl NodeType {
    pub fn icon(&self) -> &'static str {
        match self {
            NodeType::Directory => "📁",
            NodeType::NikaWorkflow => "🚂",
            NodeType::NikaConfig => "🪽",
            NodeType::YamlFile => "📄",
            NodeType::File => "📄",
        }
    }
}

#[derive(Debug, Clone)]
pub struct TreeNode {
    pub name: String,
    pub path: Option<PathBuf>,
    pub node_type: NodeType,
    pub expanded: bool,
    pub children: Vec<TreeNode>,
    pub depth: usize,
}

impl TreeNode {
    pub fn new(name: &str, node_type: NodeType) -> Self {
        Self {
            name: name.to_string(),
            path: None,
            node_type,
            expanded: false,
            children: Vec::new(),
            depth: 0,
        }
    }

    pub fn toggle_expand(&mut self) {
        if !self.children.is_empty() {
            self.expanded = !self.expanded;
        }
    }
}
```
