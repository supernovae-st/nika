**Step 3: Write minimal implementation**

```rust
impl EditorView {
    pub fn task_preview_enabled(&self) -> bool {
        true
    }

    /// Get TaskBox preview for task at current cursor line
    pub fn current_task_preview(&self) -> Option<&TaskPreview> {
        let line = self.cursor_line();
        self.task_at_line(line)
    }

    /// Find task that contains the given line number
    fn task_at_line(&self, line: usize) -> Option<&TaskPreview> {
        self.task_previews.iter().find(|t| {
            line >= t.start_line && line <= t.end_line
        })
    }
}
```
