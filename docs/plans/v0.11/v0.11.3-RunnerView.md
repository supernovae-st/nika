```rust
// src/tui/views/runner.rs

use ratatui::prelude::*;
use crate::tui::widgets::{TaskQueue, NodeBox, EdgeLine};

#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub enum RunnerState {
    #[default]
    Idle,
    Running,
    Paused,
    Complete,
    Failed,
}

pub struct RunnerView {
    state: RunnerState,
    task_queue: TaskQueue,
    dag_nodes: Vec<NodeBox>,
    dag_edges: Vec<EdgeLine>,
    selected_task: Option<String>,
    io_content: String,
    reasoning_content: String,
}

impl RunnerView {
    pub fn new() -> Self {
        Self {
            state: RunnerState::default(),
            task_queue: TaskQueue::new(),
            dag_nodes: Vec::new(),
            dag_edges: Vec::new(),
            selected_task: None,
            io_content: String::new(),
            reasoning_content: String::new(),
        }
    }

    pub fn state(&self) -> RunnerState {
        self.state
    }

    pub fn panels(&self) -> &[&'static str] {
        &["Mission Control", "DAG Visualization", "Task I/O", "Reasoning"]
    }
}
```

```rust
impl Widget for RunnerView {
    fn render(self, area: Rect, buf: &mut Buffer) {
        // Split into 2x2 grid
        let vertical = Layout::vertical([
            Constraint::Percentage(60),
            Constraint::Percentage(40),
        ]).split(area);

        let top = Layout::horizontal([
            Constraint::Percentage(40),
            Constraint::Percentage(60),
        ]).split(vertical[0]);

        let bottom = Layout::horizontal([
            Constraint::Percentage(50),
            Constraint::Percentage(50),
        ]).split(vertical[1]);

        // Panel 1: Mission Control (TaskQueue)
        self.render_mission_control(top[0], buf);

        // Panel 2: DAG Visualization
        self.render_dag_panel(top[1], buf);

        // Panel 3: Task I/O
        self.render_io_panel(bottom[0], buf);

        // Panel 4: Reasoning
        self.render_reasoning_panel(bottom[1], buf);
    }
}
```
