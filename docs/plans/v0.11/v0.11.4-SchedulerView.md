// src/scheduler/schedule.rs

use chrono::{DateTime, Utc};
use std::path::PathBuf;

#[derive(Debug, Clone)]
pub struct Schedule {
    pub id: String,
    pub name: String,
    pub workflow_path: PathBuf,
    pub cron_expression: String,
    pub enabled: bool,
    pub last_run: Option<DateTime<Utc>>,
    pub next_run: Option<DateTime<Utc>>,
}

impl Schedule {
    pub fn new(name: &str, workflow_path: PathBuf, cron: &str) -> Result<Self, CronError> {
        // Validate cron expression
        cron::Schedule::from_str(cron)?;

        Ok(Self {
            id: uuid::Uuid::new_v4().to_string(),
            name: name.to_string(),
            workflow_path,
            cron_expression: cron.to_string(),
            enabled: true,
            last_run: None,
            next_run: None,
        })
    }

    /// Calculate next run time
    pub fn calculate_next_run(&mut self) {
        if let Ok(schedule) = cron::Schedule::from_str(&self.cron_expression) {
            self.next_run = schedule.upcoming(Utc).next();
        }
    }

    /// Time until next run
    pub fn time_until_next(&self) -> Option<chrono::Duration> {
        self.next_run.map(|next| next - Utc::now())
    }
}
```

// src/scheduler/history.rs

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ExecutionStatus {
    Success,
    Failed,
    Timeout,
    Skipped,
}

#[derive(Debug, Clone)]
pub struct ExecutionRecord {
    pub schedule_id: String,
    pub timestamp: DateTime<Utc>,
    pub status: ExecutionStatus,
    pub duration: Duration,
    pub error: Option<String>,
}

impl ExecutionRecord {
    pub fn new(schedule_id: &str, status: ExecutionStatus, duration: Duration) -> Self {
        Self {
            schedule_id: schedule_id.to_string(),
            timestamp: Utc::now(),
            status,
            duration,
            error: None,
        }
    }

    pub fn with_error(mut self, error: &str) -> Self {
        self.error = Some(error.to_string());
        self
    }
}
```
