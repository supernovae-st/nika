// src/tui/views/settings.rs

use ratatui::prelude::*;
use crate::tui::widgets::provider_modal::ProviderModalState;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub enum SettingsPanel {
    #[default]
    Cloud,
    Ollama,
    Keys,
    Config,
    Mcp,
    About,
}

impl SettingsPanel {
    pub fn all() -> &'static [SettingsPanel] {
        &[
            SettingsPanel::Cloud,
            SettingsPanel::Ollama,
            SettingsPanel::Keys,
            SettingsPanel::Config,
            SettingsPanel::Mcp,
            SettingsPanel::About,
        ]
    }

    pub fn title(&self) -> &'static str {
        match self {
            SettingsPanel::Cloud => "Cloud Providers",
            SettingsPanel::Ollama => "Ollama",
            SettingsPanel::Keys => "API Keys",
            SettingsPanel::Config => "Configuration",
            SettingsPanel::Mcp => "MCP Servers",
            SettingsPanel::About => "About",
        }
    }
}

pub struct SettingsView {
    active_panel: SettingsPanel,
    provider_state: ProviderModalState,
}

impl SettingsView {
    pub fn new() -> Self {
        Self {
            active_panel: SettingsPanel::default(),
            provider_state: ProviderModalState::new(),
        }
    }

    pub fn active_panel(&self) -> SettingsPanel {
        self.active_panel
    }
}
```

impl SettingsView {
    fn render_mcp_panel(&self, area: Rect, buf: &mut Buffer) {
        let block = Block::default()
            .title(" MCP SERVERS ")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        block.render(area, buf);

        for (i, server) in self.mcp_servers.iter().enumerate() {
            let y = inner.y + i as u16;
            let status_icon = if server.connected { "●" } else { "○" };
            let line = format!("{} {} ({})", status_icon, server.name, server.status);
            buf.set_string(inner.x + 1, y, &line, Style::default());
        }
    }
}
```

    fn render_about_panel(&self, area: Rect, buf: &mut Buffer) {
        let block = Block::default()
            .title(" ABOUT NIKA ")
            .borders(Borders::ALL);

        let inner = block.inner(area);
        block.render(area, buf);

        let version = env!("CARGO_PKG_VERSION");
        let lines = vec![
            format!("Nika v{}", version),
            String::new(),
            "Semantic YAML workflow engine for AI tasks".to_string(),
            String::new(),
            "https://nika.sh".to_string(),
            String::new(),
            "Part of SuperNovae AGI".to_string(),
        ];

        for (i, line) in lines.iter().enumerate() {
            buf.set_string(inner.x + 2, inner.y + i as u16, line, Style::default());
        }
    }
}
