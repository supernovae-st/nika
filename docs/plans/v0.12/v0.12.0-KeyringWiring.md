ModalAction::SaveAndTestApiKey { provider, key } => {
    use crate::tui::widgets::provider_modal::keyring::{validate_key_format, NikaKeyring};

    // 1. Validate key format
    if let Err(e) = validate_key_format(&provider, &key) {
        self.add_system_message(format!("Invalid key format: {}", e));
        return ViewAction::None;
    }

    // 2. Save to keyring
    match NikaKeyring::set(&provider, &key) {
        Ok(()) => {
            self.add_system_message(format!("✓ Saved {} API key to keychain", provider));
            // 3. Trigger verification
            ViewAction::VerifyProviders
        }
        Err(e) => {
            self.add_system_message(format!("✗ Failed to save key: {}", e));
            ViewAction::None
        }
    }
}

**Step 3: Write minimal implementation**

```rust
ModalAction::SelectProvider { provider, model } => {
    // Update active provider state
    self.current_model = model.clone();
    self.current_provider = Some(provider.clone());

    // Update modal state
    if let Some(ref mut modal) = self.provider_modal {
        modal.state.set_active_provider(&provider);
        modal.state.set_active_model(&model);
    }

    // Close modal
    self.close_provider_modal();

    self.add_system_message(format!("Selected {} with model {}", provider, model));
    ViewAction::None
}
```
