// src/tui/widgets/provider_modal/tabs/keys.rs

#[derive(Debug, Clone, PartialEq)]
pub enum ApiKeyState {
    /// Key stored in system keychain (preferred)
    Stored { masked: String },
    /// Key from environment variable (legacy)
    EnvVar { masked: String },
    /// No key configured
    NotConfigured,
}

impl ApiKeyState {
    pub fn is_configured(&self) -> bool {
        !matches!(self, ApiKeyState::NotConfigured)
    }

    pub fn masked(&self) -> Option<&str> {
        match self {
            ApiKeyState::Stored { masked } => Some(masked),
            ApiKeyState::EnvVar { masked } => Some(masked),
            ApiKeyState::NotConfigured => None,
        }
    }

    pub fn source_label(&self) -> &'static str {
        match self {
            ApiKeyState::Stored { .. } => "keychain",
            ApiKeyState::EnvVar { .. } => "env var",
            ApiKeyState::NotConfigured => "not set",
        }
    }
}

// src/tui/widgets/provider_modal/tabs/keys.rs

use super::super::keyring::{mask_api_key, provider_env_var, NikaKeyring};

/// Detect API key state with keyring priority
pub fn detect_state(provider: &str) -> ApiKeyState {
    // Priority 1: Check keyring (v0.12.2)
    if let Ok(key) = NikaKeyring::get(provider) {
        return ApiKeyState::Stored {
            masked: mask_api_key(&key),
        };
    }

    // Priority 2: Check env var fallback
    let env_var = provider_env_var(provider);
    if let Ok(key) = std::env::var(&env_var) {
        if !key.is_empty() {
            return ApiKeyState::EnvVar {
                masked: mask_api_key(&key),
            };
        }
    }

    // Priority 3: Not configured
    ApiKeyState::NotConfigured
}

// src/tui/widgets/provider_modal/handler.rs
// Add in handle_cloud_tab() or similar

KeyCode::Enter => {
    // Get selected provider and model
    let provider = state.selected_provider().to_string();
    let model = state.selected_model().to_string();

    // Only select if configured
    if state.provider_states[state.selected_provider_idx].is_configured() {
        HandleResult::consumed_with_action(ModalAction::SelectProvider {
            provider,
            model,
        })
    } else {
        // Can't select unconfigured provider
        HandleResult::consumed()
    }
}
