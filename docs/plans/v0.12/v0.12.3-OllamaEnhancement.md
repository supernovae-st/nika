ModalAction::DeleteModel { model } => {
    // Get actual selected model from modal state
    let model_to_delete = self.provider_modal
        .as_ref()
        .and_then(|m| m.state.selected_ollama_model())
        .map(|m| m.name.clone())
        .unwrap_or(model);

    // Spawn delete task
    let client = self.ollama_client.clone();
    let model_name = model_to_delete.clone();

    tokio::spawn(async move {
        match client.delete_model(&model_name).await {
            Ok(()) => {
                // Success - will refresh on next tick
            }
            Err(e) => {
                // Log error
                tracing::error!("Failed to delete model {}: {}", model_name, e);
            }
        }
    });

// src/tui/views/chat.rs - in tick() method

fn tick(&mut self) {
    // ... existing tick logic ...

    // Process pull progress
    if let Some(ref mut rx) = self.pull_receiver {
        while let Ok(progress) = rx.try_recv() {
            match progress {
                PullProgress::Downloading { completed, total, status } => {
                    let pct = if total > 0 {
                        completed as f64 / total as f64
                    } else {
                        0.0
                    };

                    if let Some(ref mut modal) = self.provider_modal {
                        modal.state.download_state = DownloadState::Downloading {
                            model: self.pulling_model.clone().unwrap_or_default(),
                            progress: pct,
                            downloaded: completed,
                            total,
                        };
                    }
                }
                PullProgress::Complete => {
                    if let Some(ref mut modal) = self.provider_modal {
                        modal.state.download_state = DownloadState::Complete {
                            model: self.pulling_model.clone().unwrap_or_default(),
                        };
                    }
                    self.pull_receiver = None;
                    self.pulling_model = None;
                    self.add_system_message("Model pull complete!");
                    // Trigger refresh
                }
                PullProgress::Error(e) => {
                    if let Some(ref mut modal) = self.provider_modal {
                        modal.state.download_state = DownloadState::Failed {
                            model: self.pulling_model.clone().unwrap_or_default(),
                            error: e.clone(),
                        };
                    }
                    self.pull_receiver = None;
                    self.pulling_model = None;
                    self.add_system_message(format!("Pull failed: {}", e));
                }
                _ => {}
            }
        }
    }
}
