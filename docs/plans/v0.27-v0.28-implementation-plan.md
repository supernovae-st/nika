### 2.6 New main.rs Structure

```rust
//! Nika CLI entry point
//!
//! This file delegates to src/cli/ modules for all commands.

mod cli;

use clap::Parser;
use cli::args::{Cli, Command};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let cli = Cli::parse();

    match cli.command {
        Command::Run(args) => cli::run::execute(args).await?,
        Command::Check(args) => cli::check::execute(args).await?,
        Command::Chat(args) => cli::chat::execute(args).await?,
        Command::Studio(args) => cli::studio::execute(args).await?,
        Command::Provider(cmd) => cli::provider::execute(cmd).await?,
        Command::Model(cmd) => cli::model::execute(cmd).await?,
        Command::Mcp(cmd) => cli::mcp::execute(cmd).await?,
        Command::Sync(args) => cli::sync::execute(args).await?,
        Command::Daemon(cmd) => cli::daemon::execute(cmd).await?,
        Command::Jobs(cmd) => cli::jobs::execute(cmd).await?,
        Command::Backup(cmd) => cli::backup::execute(cmd).await?,
        Command::Setup(args) => cli::setup::execute(args).await?,
        Command::Trace(cmd) => cli::trace::execute(cmd).await?,
        Command::Config(cmd) => cli::config::execute(cmd).await?,
        Command::Schema(cmd) => cli::schema::execute(cmd).await?,
        Command::New(args) => cli::new::execute(args).await?,
        Command::Completion(args) => cli::completion::execute(args)?,
        None => cli::tui::launch().await?,
    }

    Ok(())
}
```

```rust
//! Autonomy types for Human-in-the-Loop (HITL) workflows.
//!
//! These types control how much autonomy agents have and when
//! human approval is required.

use serde::{Deserialize, Serialize};
use std::fmt;

/// Level of autonomy granted to an agent.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum AutonomyLevel {
    /// Every action requires explicit approval
    Manual,
    /// Low-risk actions auto-approved, high-risk need approval
    Supervised,
    /// All actions auto-approved within policy limits
    Autonomous,
}

impl Default for AutonomyLevel {
    fn default() -> Self {
        Self::Supervised
    }
}

impl fmt::Display for AutonomyLevel {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Manual => write!(f, "manual"),
            Self::Supervised => write!(f, "supervised"),
            Self::Autonomous => write!(f, "autonomous"),
        }
    }
}

/// Risk level of an action.
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum RiskLevel {
    /// No significant impact (read-only, local)
    Low,
    /// Moderate impact (writes, API calls)
    Medium,
    /// Significant impact (external systems, costs)
    High,
    /// Critical impact (irreversible, security)
    Critical,
}

impl Default for RiskLevel {
    fn default() -> Self {
        Self::Low
    }
}

/// Required approval level for an action.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum ApprovalLevel {
    /// No approval needed
    None,
    /// Notify user but proceed
    Notify,
    /// Wait for explicit confirmation
    Confirm,
    /// Block until approved by policy
    Block,
}

impl ApprovalLevel {
    /// Determine approval level based on autonomy and risk.
    pub fn from_policy(autonomy: AutonomyLevel, risk: RiskLevel) -> Self {
        match (autonomy, risk) {
            (AutonomyLevel::Manual, _) => Self::Confirm,
            (AutonomyLevel::Supervised, RiskLevel::Low) => Self::None,
            (AutonomyLevel::Supervised, RiskLevel::Medium) => Self::Notify,
            (AutonomyLevel::Supervised, RiskLevel::High) => Self::Confirm,
            (AutonomyLevel::Supervised, RiskLevel::Critical) => Self::Block,
            (AutonomyLevel::Autonomous, RiskLevel::Low | RiskLevel::Medium) => Self::None,
            (AutonomyLevel::Autonomous, RiskLevel::High) => Self::Notify,
            (AutonomyLevel::Autonomous, RiskLevel::Critical) => Self::Confirm,
        }
    }
}

/// Decision made on an approval request.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum Decision {
    /// Action approved
    Approved,
    /// Action denied
    Denied,
    /// Escalated to higher authority
    Escalated,
    /// Timed out waiting for decision
    TimedOut,
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_autonomy_level_default() {
        assert_eq!(AutonomyLevel::default(), AutonomyLevel::Supervised);
    }

    #[test]
    fn test_risk_level_ordering() {
        assert!(RiskLevel::Low < RiskLevel::Medium);
        assert!(RiskLevel::Medium < RiskLevel::High);
        assert!(RiskLevel::High < RiskLevel::Critical);
    }

    #[test]
    fn test_approval_from_policy_manual() {
        assert_eq!(
            ApprovalLevel::from_policy(AutonomyLevel::Manual, RiskLevel::Low),
            ApprovalLevel::Confirm
        );
    }

    #[test]
    fn test_approval_from_policy_supervised() {
        assert_eq!(
            ApprovalLevel::from_policy(AutonomyLevel::Supervised, RiskLevel::Low),
            ApprovalLevel::None
        );
        assert_eq!(
            ApprovalLevel::from_policy(AutonomyLevel::Supervised, RiskLevel::Critical),
            ApprovalLevel::Block
        );
    }

    #[test]
    fn test_approval_from_policy_autonomous() {
        assert_eq!(
            ApprovalLevel::from_policy(AutonomyLevel::Autonomous, RiskLevel::Medium),
            ApprovalLevel::None
        );
        assert_eq!(
            ApprovalLevel::from_policy(AutonomyLevel::Autonomous, RiskLevel::Critical),
            ApprovalLevel::Confirm
        );
    }

    #[test]
    fn test_autonomy_display() {
        assert_eq!(AutonomyLevel::Manual.to_string(), "manual");
        assert_eq!(AutonomyLevel::Supervised.to_string(), "supervised");
        assert_eq!(AutonomyLevel::Autonomous.to_string(), "autonomous");
    }

    #[test]
    fn test_serde_roundtrip() {
        let level = AutonomyLevel::Supervised;
        let json = serde_json::to_string(&level).unwrap();
        let parsed: AutonomyLevel = serde_json::from_str(&json).unwrap();
        assert_eq!(level, parsed);
    }
}
```

```rust
//! Memory types for persistent agent memory.
//!
//! Agents can store and retrieve data across sessions.

use serde::{Deserialize, Serialize};
use std::fmt;
use std::time::{Duration, SystemTime};

/// Namespace for memory isolation.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum MemoryNamespace {
    /// Project-specific memory (stored in .nika/)
    Project,
    /// Session-specific memory (cleared on exit)
    Session,
    /// Global memory (stored in ~/.nika/)
    Global,
}

impl Default for MemoryNamespace {
    fn default() -> Self {
        Self::Session
    }
}

impl fmt::Display for MemoryNamespace {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Project => write!(f, "project"),
            Self::Session => write!(f, "session"),
            Self::Global => write!(f, "global"),
        }
    }
}

/// Structured key for memory entries.
#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct MemoryKey {
    pub namespace: MemoryNamespace,
    pub key: String,
}

impl MemoryKey {
    pub fn new(namespace: MemoryNamespace, key: impl Into<String>) -> Self {
        Self {
            namespace,
            key: key.into(),
        }
    }

    pub fn project(key: impl Into<String>) -> Self {
        Self::new(MemoryNamespace::Project, key)
    }

    pub fn session(key: impl Into<String>) -> Self {
        Self::new(MemoryNamespace::Session, key)
    }

    pub fn global(key: impl Into<String>) -> Self {
        Self::new(MemoryNamespace::Global, key)
    }
}

impl fmt::Display for MemoryKey {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{}:{}", self.namespace, self.key)
    }
}

/// A memory entry with metadata.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MemoryEntry {
    pub key: MemoryKey,
    pub value: serde_json::Value,
    pub created_at: SystemTime,
    pub ttl: Option<Duration>,
}

impl MemoryEntry {
    pub fn new(key: MemoryKey, value: serde_json::Value) -> Self {
        Self {
            key,
            value,
            created_at: SystemTime::now(),
            ttl: None,
        }
    }

    pub fn with_ttl(mut self, ttl: Duration) -> Self {
        self.ttl = Some(ttl);
        self
    }

    pub fn is_expired(&self) -> bool {
        if let Some(ttl) = self.ttl {
            self.created_at.elapsed().map(|e| e > ttl).unwrap_or(false)
        } else {
            false
        }
    }
```

```rust
//! Trace types for reasoning transparency.

//! Captures agent reasoning process for debugging and audit.

use serde::{Deserialize, Serialize};
use std::fmt;
use std::time::SystemTime;
use uuid::Uuid;

/// Unique identifier for a trace.
#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct TraceId(pub String);

impl TraceId {
    pub fn new() -> Self {
        Self(Uuid::new_v4().to_string())
    }

    pub fn from_string(s: impl Into<String>) -> Self {
        Self(s.into())
    }

    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl Default for TraceId {
    fn default() -> Self {
        Self::new()
    }
}

impl fmt::Display for TraceId {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{}", self.0)
    }
}

/// Kind of trace step.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum TraceStepKind {
    /// Internal reasoning (thinking)
    Reasoning,
    /// Action taken (tool call, output)
    Action,
    /// Observation received (tool result, input)
    Observation,
    /// Decision made (approval, routing)
    Decision,
}

impl fmt::Display for TraceStepKind {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Reasoning => write!(f, "reasoning"),
            Self::Action => write!(f, "action"),
            Self::Observation => write!(f, "observation"),
            Self::Decision => write!(f, "decision"),
        }
    }
}

/// Metadata for a trace.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TraceMetadata {
    pub trace_id: TraceId,
    pub agent_id: Option<String>,
    pub started_at: SystemTime,
    pub ended_at: Option<SystemTime>,
    pub input_tokens: u32,
    pub output_tokens: u32,
    pub estimated_cost_usd: f64,
}

impl TraceMetadata {
    pub fn new(trace_id: TraceId) -> Self {
        Self {
            trace_id,
            agent_id: None,
            started_at: SystemTime::now(),
            ended_at: None,
            input_tokens: 0,
            output_tokens: 0,
            estimated_cost_usd: 0.0,
        }
    }

    pub fn with_agent(mut self, agent_id: impl Into<String>) -> Self {
        self.agent_id = Some(agent_id.into());
        self
    }

    pub fn complete(&mut self) {
        self.ended_at = Some(SystemTime::now());
    }

    pub fn add_tokens(&mut self, input: u32, output: u32) {
        self.input_tokens += input;
        self.output_tokens += output;
    }

    pub fn total_tokens(&self) -> u32 {
        self.input_tokens + self.output_tokens
    }
}
```
