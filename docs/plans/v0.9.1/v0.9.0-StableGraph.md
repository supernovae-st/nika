## Why StableGraph?

```
║  PROBLEM: DiGraph NodeIndex Invalidation                                      ║
╠═══════════════════════════════════════════════════════════════════════════════╣
║                                                                               ║
║  DiGraph (current):                                                           ║
║  ├── Node 0: "msg-001" (index=0)                                              ║
║  ├── Node 1: "msg-002" (index=1)  ← DELETE THIS                               ║
║  └── Node 2: "msg-003" (index=2)  ← BECOMES index=1 ⚠️                         ║
║                                                                               ║
║  User types: "@2" expecting "msg-003" but gets wrong node!                    ║
║                                                                               ║
║  StableGraph (v0.9.0):                                                        ║

║  ├── Node 1: DELETED                                                          ║
║  └── Node 2: "msg-003" (index=2)  ← STAYS index=2 ✅                           ║
║                                                                               ║
║  "@2" always refers to "msg-003" — stable references!                         ║
```

```rust
// src/dag/stable.rs
use petgraph::stable_graph::StableGraph;
use petgraph::Directed;

/// Wrapper around petgraph::StableGraph for DAG operations.
/// NodeIndex values remain stable after node deletion.
pub struct StableFlowGraph<N> {
    inner: StableGraph<N, (), Directed>,
}

impl<N> StableFlowGraph<N> {
    pub fn new() -> Self {
        Self {
            inner: StableGraph::new(),
        }
    }

    pub fn with_capacity(nodes: usize, edges: usize) -> Self {
        Self {
            inner: StableGraph::with_capacity(nodes, edges),
        }
    }

    pub fn node_count(&self) -> usize {
        self.inner.node_count()
    }

    pub fn edge_count(&self) -> usize {
        self.inner.edge_count()
    }
}

impl<N> Default for StableFlowGraph<N> {
    fn default() -> Self {
        Self::new()
    }
}
```

```rust
use petgraph::stable_graph::EdgeIndex;

impl<N> StableFlowGraph<N> {
    /// Add a directed edge from `a` to `b`.
    /// Returns None if either node doesn't exist.
    pub fn add_edge(&mut self, a: NodeIndex, b: NodeIndex) -> Option<EdgeIndex> {
        if !self.contains_node(a) || !self.contains_node(b) {
            return None;
        }
        Some(self.inner.add_edge(a, b, ()))
    }

    /// Check if an edge exists from `a` to `b`.
    pub fn has_edge(&self, a: NodeIndex, b: NodeIndex) -> bool {
        self.inner.find_edge(a, b).is_some()
    }

    /// Remove an edge between nodes.
    pub fn remove_edge(&mut self, a: NodeIndex, b: NodeIndex) -> bool {
        if let Some(edge) = self.inner.find_edge(a, b) {
            self.inner.remove_edge(edge);
            true
        } else {
            false
        }
    }
}
```

```rust
// Update FlowGraph to use StableFlowGraph internally
use crate::dag::stable::StableFlowGraph;

pub struct FlowGraph {
    graph: StableFlowGraph<Task>,
    task_index: HashMap<Arc<str>, NodeIndex>,
}

impl FlowGraph {
    pub fn new() -> Self {
        Self {
            graph: StableFlowGraph::new(),
            task_index: HashMap::new(),
        }
    }

    pub fn add_task(&mut self, task: Task) -> NodeIndex {
        let id = task.id.clone();
        let idx = self.graph.add_node(task);
        self.task_index.insert(id, idx);
        idx
    }

    pub fn remove_task(&mut self, id: &Arc<str>) -> Option<Task> {
        if let Some(idx) = self.task_index.remove(id) {
            self.graph.remove_node(idx)
        } else {
            None
        }
    }

    pub fn get_task_by_index(&self, idx: NodeIndex) -> Option<&Task> {
        self.graph.node_weight(idx)
    }
}
```
