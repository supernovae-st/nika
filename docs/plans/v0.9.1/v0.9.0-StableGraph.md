**Goal:** Migrate FlowGraph from petgraph::DiGraph to StableGraph for stable NodeIndex after deletion.

**Architecture:** StableGraph preserves NodeIndex values when nodes are removed, critical for @mention references.

**Tech Stack:** petgraph (StableGraph), parking_lot

**New Dependencies (add to Cargo.toml):**
```toml
# tools/nika/Cargo.toml
[dependencies]

# parking_lot already present in codebase
```

> **Note:** The `serde-1` feature enables serialization of `StableGraph` and `NodeIndex` for session persistence and export. Uses petgraph's native serde support.

```rust
// src/dag/stable.rs
use petgraph::stable_graph::StableGraph;
use petgraph::Directed;

/// Wrapper around petgraph::StableGraph for DAG operations.
/// NodeIndex values remain stable after node deletion.
pub struct StableFlowGraph<N> {
    inner: StableGraph<N, (), Directed>,
}

impl<N> StableFlowGraph<N> {
    pub fn new() -> Self {
        Self {
            inner: StableGraph::new(),
        }
    }

    pub fn with_capacity(nodes: usize, edges: usize) -> Self {
        Self {
            inner: StableGraph::with_capacity(nodes, edges),
        }
    }

    pub fn node_count(&self) -> usize {
        self.inner.node_count()
    }

    pub fn edge_count(&self) -> usize {
        self.inner.edge_count()
    }
}

impl<N> Default for StableFlowGraph<N> {
    fn default() -> Self {
        Self::new()
    }
}
```

```rust
use petgraph::stable_graph::EdgeIndex;

impl<N> StableFlowGraph<N> {
    /// Add a directed edge from `a` to `b`.
    /// Returns None if either node doesn't exist.
    pub fn add_edge(&mut self, a: NodeIndex, b: NodeIndex) -> Option<EdgeIndex> {
        if !self.contains_node(a) || !self.contains_node(b) {
            return None;
        }
        Some(self.inner.add_edge(a, b, ()))
    }

    /// Check if an edge exists from `a` to `b`.
    pub fn has_edge(&self, a: NodeIndex, b: NodeIndex) -> bool {
        self.inner.find_edge(a, b).is_some()
    }

    /// Remove an edge between nodes.
    pub fn remove_edge(&mut self, a: NodeIndex, b: NodeIndex) -> bool {
        if let Some(edge) = self.inner.find_edge(a, b) {
            self.inner.remove_edge(edge);
            true
        } else {
            false
        }
    }
}
```

```rust
// Update FlowGraph to use StableFlowGraph internally
use crate::dag::stable::StableFlowGraph;

pub struct FlowGraph {
    graph: StableFlowGraph<Task>,
    task_index: HashMap<Arc<str>, NodeIndex>,
}

impl FlowGraph {
    pub fn new() -> Self {
        Self {
            graph: StableFlowGraph::new(),
            task_index: HashMap::new(),
        }
    }

    pub fn add_task(&mut self, task: Task) -> NodeIndex {
        let id = task.id.clone();
        let idx = self.graph.add_node(task);
        self.task_index.insert(id, idx);
        idx
    }

    pub fn remove_task(&mut self, id: &Arc<str>) -> Option<Task> {
        if let Some(idx) = self.task_index.remove(id) {
            self.graph.remove_node(idx)
        } else {
            None
        }
    }

    pub fn get_task_by_index(&self, idx: NodeIndex) -> Option<&Task> {
        self.graph.node_weight(idx)
    }
}
```

### Property-Based Tests (proptest)

```rust
use proptest::prelude::*;

proptest! {
    /// Any sequence of add/remove operations maintains index stability
    #[test]
    fn test_index_stability_under_mutations(
        ops in prop::collection::vec(
            prop_oneof![
                Just("add"),
                Just("remove_first"),
                Just("remove_last"),
                Just("remove_middle"),
            ],
            1..50
        )
    ) {
        let mut graph: StableFlowGraph<usize> = StableFlowGraph::new();
        let mut indices = Vec::new();

        for op in ops {
            match op {
                "add" => {
                    let idx = graph.add_node(indices.len());
                    indices.push((idx, true)); // (index, exists)
                }
                "remove_first" => {
                    if let Some(pos) = indices.iter().position(|(_, e)| *e) {
                        let (idx, _) = indices[pos];
                        graph.remove_node(idx);
                        indices[pos].1 = false;
                    }
                }
                "remove_last" => {
                    if let Some(pos) = indices.iter().rposition(|(_, e)| *e) {
                        let (idx, _) = indices[pos];
                        graph.remove_node(idx);
                        indices[pos].1 = false;
                    }
                }
                "remove_middle" => {
                    let existing: Vec<_> = indices.iter()
                        .enumerate()
                        .filter(|(_, (_, e))| *e)
                        .collect();
                    if existing.len() > 2 {
                        let mid = existing.len() / 2;
                        let (pos, (idx, _)) = existing[mid];
                        graph.remove_node(*idx);
                        indices[pos].1 = false;
                    }
                }
                _ => {}
            }

            // INVARIANT: All remaining indices are valid and unchanged
            for (idx, exists) in &indices {
                assert_eq!(graph.contains_node(*idx), *exists);
            }
        }
    }
}
```

```rust
// benches/stablegraph_bench.rs
use criterion::{criterion_group, criterion_main, Criterion, BenchmarkId};

fn bench_add_nodes(c: &mut Criterion) {
    let mut group = c.benchmark_group("stablegraph_add");

    for size in [100, 1000, 10000] {
        group.bench_with_input(
            BenchmarkId::from_parameter(size),
            &size,
            |b, &n| {
                b.iter(|| {
                    let mut graph: StableFlowGraph<usize> = StableFlowGraph::new();
                    for i in 0..n {
                        graph.add_node(i);
                    }
                });
            },
        );
    }
    group.finish();
}

fn bench_delete_middle(c: &mut Criterion) {
    c.bench_function("delete_middle_1000", |b| {
        b.iter_batched(
            || {
                let mut graph: StableFlowGraph<usize> = StableFlowGraph::new();
                let indices: Vec<_> = (0..1000).map(|i| graph.add_node(i)).collect();
                (graph, indices)
            },
            |(mut graph, indices)| {
                // Delete every other node
                for i in (0..1000).step_by(2) {
                    graph.remove_node(indices[i]);
                }
            },
            criterion::BatchSize::SmallInput,
        );
    });
}

criterion_group!(benches, bench_add_nodes, bench_delete_middle);
criterion_main!(benches);
```
