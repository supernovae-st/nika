**Architecture:** MentionParser extracts @N, @last, @all, @N..M from text. mentions_to_wiring() converts to binding entries.

## Mention Syntax

```
╔═══════════════════════════════════════════════════════════════════════════════╗
║  @MENTION SYNTAX                                                              ║
╠═══════════════════════════════════════════════════════════════════════════════╣
║                                                                               ║
║  @1         → Reference message #1 (msg-001)                                  ║
║  @2         → Reference message #2 (msg-002)                                  ║
║  @last      → Reference the last message                                      ║
║  @all       → Reference all previous messages                                 ║
║  @1..3      → Reference messages 1, 2, and 3 (range)                          ║
║  //         → Parallel marker (no dependency on previous message)             ║
║                                                                               ║
║  Examples:                                                                    ║
║  "Summarize @1"                  → depends on msg-001                         ║
║  "Continue from @last"           → depends on most recent message             ║
║  "Combine @1 and @3"             → depends on msg-001 and msg-003             ║
║  "Based on @all"                 → depends on all previous messages           ║
║  "// Independent task"           → no dependency edges (parallel)             ║
║                                                                               ║
╚═══════════════════════════════════════════════════════════════════════════════╝
```

## Edge Case Behaviors

> **IMPORTANT:** These behaviors MUST be implemented and tested. Each edge case has a defined resolution.

| Edge Case | Input | Behavior | Error Code |
|-----------|-------|----------|------------|
| **Empty chat + @last** | `@last` on empty chat | Return error | `NIKA-260` |
| **Empty chat + @all** | `@all` on empty chat | Return empty array `[]` (valid) | — |
| **@0 reference** | `@0` | Valid - references first message (index 0) | — |
| **Deleted node @N** | `@3` when msg-003 deleted | Return error with suggestion | `NIKA-261` |
| **@N.result on user msg** | `@2.result` when @2 is user | Return error (users have no result) | `NIKA-262` |
| **Self-reference** | `@3` inside message 3 | Return error (circular) | `NIKA-263` |
| **Out of bounds** | `@999` when only 5 messages | Return error with valid range hint | `NIKA-264` |
| **Range invalid** | `@5..2` (start > end) | Return error | `NIKA-265` |
| **Email false positive** | `user@123.com` | NOT parsed as @123 (regex guards) | — |

```rust
use regex::Regex;
use std::sync::LazyLock;

// NOTE: The `regex` crate does NOT support lookbehind assertions.
// Instead, we match @ preceded by whitespace/punctuation/start-of-string.
// Pattern: (^|[\s\[\](),:;])@... then extract the @mention part.
static MENTION_REGEX: LazyLock<Regex> = LazyLock::new(|| {
    // Group 1: prefix (whitespace/punctuation) - discard
    // Group 2,3: Range @N..M
    // Group 4: Number @N
    // Group 5: Keyword (last|all)
    Regex::new(r"(?:^|[\s\[\](),:;])@(?:(\d+)\.\.(\d+)|(\d+)|(last|all))")
        .expect("Invalid mention regex")
});

/// Parse @mentions from text.
///
/// Supports:
/// - `@N` - Reference message N (1-indexed)
/// - `@N..M` - Range of messages from N to M
/// - `@last` - Most recent message
/// - `@all` - All messages
///
/// Does NOT match emails (user@123.com) due to prefix requirement.
pub fn parse_mentions(text: &str) -> Vec<Mention> {
    let mut mentions = Vec::new();

    for cap in MENTION_REGEX.captures_iter(text) {
        if let (Some(start), Some(end)) = (cap.get(1), cap.get(2)) {
            // Range: @N..M
            let start: u32 = start.as_str().parse().unwrap();
            let end: u32 = end.as_str().parse().unwrap();
            mentions.push(Mention::Range { start, end });
        } else if let Some(num) = cap.get(3) {
            // Number: @N
            let n: u32 = num.as_str().parse().unwrap();
            mentions.push(Mention::Number(n));
        } else if let Some(keyword) = cap.get(4) {
            // Keyword: @last or @all
            match keyword.as_str() {
                "last" => mentions.push(Mention::Last),
                "all" => mentions.push(Mention::All),
                _ => {}
            }
        }
    }

    mentions
}
```

```rust
/// Resolve a mention to concrete message numbers.
/// Returns a list of Number mentions.
pub fn resolve_mention(mention: Mention, message_count: u32) -> Vec<Mention> {
    match mention {
        Mention::Number(n) => {
            if n > 0 && n <= message_count {
                vec![Mention::Number(n)]
            } else {
                vec![]
            }
        }
        Mention::Last => {
            if message_count > 0 {
                vec![Mention::Number(message_count)]
            } else {
                vec![]
            }
        }
        Mention::All => {
            (1..=message_count).map(Mention::Number).collect()
        }
        Mention::Range { start, end } => {
            let s = start.max(1);
            let e = end.min(message_count);
            if s <= e {
                (s..=e).map(Mention::Number).collect()
            } else {
                vec![]
            }
        }
    }
}
```

```rust
pub fn strip_parallel(text: &str) -> &str {
    let trimmed = text.trim_start();
    if trimmed.starts_with("//") {
        trimmed[2..].trim_start()
    } else {
        text
    }
}
```

```rust
use crate::binding::{WiringSpec, UseEntry, BindingPath};

/// Convert mentions in text to WiringSpec bindings.
pub fn mentions_to_wiring(
    text: &str,
    message_count: u32,
    _current_id: Option<&str>,
) -> WiringSpec {
    // Parallel marker → no dependencies
    if is_parallel(text) {
        return WiringSpec::default();
    }

    let mentions = parse_mentions(text);
    let mut use_entries = Vec::new();

    for mention in mentions {
        let resolved = resolve_mention(mention, message_count);
        for Mention::Number(n) in resolved {
            let alias = format!("msg_{:03}", n);
            let path = BindingPath::new(format!("msg-{:03}.content", n));

            use_entries.push(UseEntry {
                alias,
                path,
                lazy: false,
                default: None,
            });
        }
    }

    // Deduplicate
    use_entries.sort_by(|a, b| a.alias.cmp(&b.alias));
    use_entries.dedup_by(|a, b| a.alias == b.alias);

    WiringSpec { use_entries }
}
```

```rust
impl ChatWorkflow {
    pub fn add_message_with_mentions(&mut self, content: &str, role: Role) -> NodeIndex {
        let mentions = parse_mentions(content);
        let is_parallel = is_parallel(content);

        self.message_counter += 1;
        let id = format!("msg-{:03}", self.message_counter);

        let message = ChatMessage {
            id: id.clone(),
            content: content.to_string(),
            role,
            timestamp: Utc::now(),
            mentions: mentions.clone(),
        };

        let idx = self.dag.add_node(message);
        self.id_to_index.insert(id, idx);

        // Add edges from mentioned messages
        for mention in &mentions {
            let resolved = resolve_mention(mention.clone(), self.message_counter - 1);
            for Mention::Number(n) in resolved {
                if let Some(source_idx) = self.get_index_by_number(n) {
                    self.dag.add_edge(source_idx, idx);
                }
            }
        }

        // Add sequential edge if not parallel
        if !is_parallel && self.message_counter > 1 {
            if let Some(prev_idx) = self.get_index_by_number(self.message_counter - 1) {
                self.dag.add_edge(prev_idx, idx);
            }
        }

        idx
    }
}
```
