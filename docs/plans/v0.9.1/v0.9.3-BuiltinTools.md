```rust
use crate::error::NikaError;
use std::future::Future;
use std::pin::Pin;

/// Trait for builtin nika:* tools.
pub trait BuiltinTool: Send + Sync {
    /// Tool name (without nika: prefix).
    fn name(&self) -> &'static str;

    /// Tool description for LLM.
    fn description(&self) -> &'static str {
        ""
    }

    /// JSON schema for parameters.
    fn parameters_schema(&self) -> serde_json::Value {
        serde_json::json!({})
    }

    /// Execute the tool with JSON args.
    fn call<'a>(
        &'a self,
        args: String,
    ) -> Pin<Box<dyn Future<Output = Result<String, NikaError>> + Send + 'a>>;
}
```

```rust
// src/runtime/builtin/router.rs
use super::BuiltinTool;
use crate::error::NikaError;
use crate::event::EventLog;
use crate::store::DataStore;
use std::collections::HashMap;
use std::sync::Arc;

pub struct BuiltinToolRouter {
    tools: HashMap<&'static str, Arc<dyn BuiltinTool>>,
    event_log: EventLog,
}

impl BuiltinToolRouter {
    pub fn new(event_log: EventLog, store: DataStore) -> Self {
        let mut tools: HashMap<&'static str, Arc<dyn BuiltinTool>> = HashMap::new();

        // Register all 6 tools
        tools.insert("sleep", Arc::new(super::SleepTool));
        tools.insert("log", Arc::new(super::LogTool::new(event_log.clone())));
        tools.insert("emit", Arc::new(super::EmitTool::new(event_log.clone())));
        tools.insert("assert", Arc::new(super::AssertTool));
        tools.insert("prompt", Arc::new(super::PromptTool));
        tools.insert("run", Arc::new(super::RunTool::new(store)));

        Self { tools, event_log }
    }

    pub fn is_builtin(tool_name: &str) -> bool {
        tool_name.starts_with("nika:")
    }

    pub fn extract_name(tool_name: &str) -> Option<&str> {
        tool_name.strip_prefix("nika:")
    }

    pub fn has_tool(&self, name: &str) -> bool {
        self.tools.contains_key(name)
    }

    pub async fn dispatch(&self, tool_name: &str, args: String) -> Result<String, NikaError> {
        let name = Self::extract_name(tool_name)
            .ok_or_else(|| NikaError::BuiltinToolError {
                tool: tool_name.into(),
                reason: "Not a builtin tool".into(),
            })?;

        let tool = self.tools.get(name)
            .ok_or_else(|| NikaError::BuiltinToolError {
                tool: tool_name.into(),
                reason: format!("Unknown builtin tool: {}", name),
            })?;

        tool.call(args).await
    }
}
```

```rust
// In executor.rs
impl Executor {
    async fn execute_invoke(&self, task: &Task) -> Result<TaskResult, NikaError> {
        let invoke = task.invoke.as_ref().unwrap();

        // Check if builtin
        if BuiltinToolRouter::is_builtin(&invoke.tool) {
            let args = serde_json::to_string(&invoke.params)?;
            let result = self.builtin_router.dispatch(&invoke.tool, args).await?;
            return Ok(TaskResult::success(task.id.clone(), result));
        }

        // Otherwise route to MCP
        self.execute_mcp_invoke(task).await
    }
}
```
