// src/tui/widgets/node_box.rs
use ratatui::prelude::*;
use ratatui::widgets::{Block, Borders, Paragraph};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum NodeKind {
    UserMessage,
    AssistantMessage,
    ToolCall,
    SystemMessage,
}

#[derive(Debug, Clone)]
pub struct NodeBox {
    pub id: String,
    pub kind: NodeKind,
    pub is_selected: bool,
    pub is_running: bool,
    pub label: Option<String>,
}

impl NodeBox {
    pub fn new(id: &str, kind: NodeKind) -> Self {
        Self {
            id: id.to_string(),
            kind,
            is_selected: false,
            is_running: false,
            label: None,
        }
    }

    pub fn selected(mut self, selected: bool) -> Self {
        self.is_selected = selected;
        self
    }

    pub fn running(mut self, running: bool) -> Self {
        self.is_running = running;
        self
    }

    pub fn icon(kind: NodeKind) -> &'static str {
        match kind {
            NodeKind::UserMessage => "👤",
            NodeKind::AssistantMessage => "🤖",
            NodeKind::ToolCall => "🔧",
            NodeKind::SystemMessage => "⚙️",
        }
    }

    pub fn width(&self) -> u16 {
        // id + space + icon + borders
        (self.id.len() + 4) as u16
    }

    pub fn height(&self) -> u16 {
        3 // top border + content + bottom border
    }
}

impl Widget for NodeBox {
    fn render(self, area: Rect, buf: &mut Buffer) {
        let border_color = if self.is_selected {
            Color::LightBlue
        } else if self.is_running {
            Color::Yellow
        } else {
            Color::Gray
        };

        let block = Block::default()
            .borders(Borders::ALL)
            .border_style(Style::default().fg(border_color));

        let content = format!("{} {}", self.id, Self::icon(self.kind));
        let para = Paragraph::new(content).block(block);
        para.render(area, buf);
    }
}
```

// src/tui/widgets/edge_line.rs
use ratatui::prelude::*;

pub struct EdgeLine {
    pub from_y: u16,
    pub to_y: u16,
    pub x: u16,
    pub is_active: bool,
}

impl EdgeLine {
    pub const VERTICAL: char = '│';
    pub const ARROW_DOWN: char = '▼';
    pub const CORNER_RIGHT: char = '└';
    pub const CORNER_LEFT: char = '┘';
    pub const HORIZONTAL: char = '─';

    pub fn vertical(from_y: u16, to_y: u16, x: u16) -> Self {
        Self {
            from_y,
            to_y,
            x,
            is_active: false,
        }
    }

    pub fn active(mut self, active: bool) -> Self {
        self.is_active = active;
        self
    }
}

impl Widget for EdgeLine {
    fn render(self, _area: Rect, buf: &mut Buffer) {
        let color = if self.is_active {
            Color::Green
        } else {
            Color::DarkGray
        };

        let style = Style::default().fg(color);

        // Draw vertical line
        for y in self.from_y..self.to_y {
            if let Some(cell) = buf.cell_mut((self.x, y)) {
                cell.set_char(Self::VERTICAL).set_style(style);
            }
        }

        // Draw arrow at end
        if let Some(cell) = buf.cell_mut((self.x, self.to_y)) {
            cell.set_char(Self::ARROW_DOWN).set_style(style);
        }
    }
}
```

// src/tui/widgets/chat_dag_panel.rs
use super::{NodeBox, NodeKind, EdgeLine};
use ratatui::prelude::*;
use std::collections::HashMap;

pub struct ChatDagPanel {
    pub nodes: Vec<(String, NodeKind)>,
    pub edges: Vec<(String, String)>,
    pub selected: Option<String>,
    pub scroll_offset: u16,
    node_positions: HashMap<String, (u16, u16)>,
}

impl ChatDagPanel {
    pub fn new() -> Self {
        Self {
            nodes: Vec::new(),
            edges: Vec::new(),
            selected: None,
            scroll_offset: 0,
            node_positions: HashMap::new(),
        }
    }

    pub fn add_node(&mut self, id: &str, kind: NodeKind) {
        self.nodes.push((id.to_string(), kind));
    }

    pub fn add_edge(&mut self, from: &str, to: &str) {
        self.edges.push((from.to_string(), to.to_string()));
    }

    pub fn select(&mut self, id: &str) {
        self.selected = Some(id.to_string());
    }

    pub fn clear_selection(&mut self) {
        self.selected = None;
    }

    fn calculate_layout(&mut self, area: Rect) {
        self.node_positions.clear();

        let node_width = 15u16;
        let node_height = 3u16;
        let vertical_gap = 2u16;

        let center_x = area.x + (area.width.saturating_sub(node_width)) / 2;
        let mut y = area.y;

        for (id, _kind) in &self.nodes {
            self.node_positions.insert(id.clone(), (center_x, y));
            y += node_height + vertical_gap;
        }
    }
}

impl Widget for ChatDagPanel {
    fn render(mut self, area: Rect, buf: &mut Buffer) {
        self.calculate_layout(area);

        // Render edges first (behind nodes)
        for (from, to) in &self.edges {
            if let (Some(&(_, from_y)), Some(&(x, to_y))) =
                (self.node_positions.get(from), self.node_positions.get(to))
            {
                let edge = EdgeLine::vertical(from_y + 3, to_y, x + 7);
                edge.render(area, buf);
            }
        }

        // Render nodes
        for (id, kind) in &self.nodes {
            if let Some(&(x, y)) = self.node_positions.get(id) {
                let is_selected = self.selected.as_ref() == Some(id);
                let node = NodeBox::new(id, *kind).selected(is_selected);

                let node_area = Rect::new(x, y, 15, 3);
                node.render(node_area, buf);
            }
        }
    }
}

impl Default for ChatDagPanel {
    fn default() -> Self {
        Self::new()
    }
}
