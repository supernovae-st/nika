## Overview

TaskBox widgets are the **visual representation of verbs in action**. They appear in:
- **Chat View** — Inline compact boxes during conversation
- **Runner View** — Expanded boxes during workflow execution
- **Monitor View** — Historical execution traces
- **DAG Panel** — Node visualization in graph
- **History/Export** — Serialized snapshots

This spec defines the **shared foundation** that all 5 verb-specific boxes inherit.

### Pattern 3: FormatBuf for Metrics (Zero Allocation)

```rust
use std::fmt::Write;

/// Stack-allocated buffer for formatting metrics (no heap allocation)
pub struct FormatBuf {
    buf: [u8; 64],
    len: usize,
}

impl FormatBuf {
    pub fn new() -> Self {
        Self { buf: [0; 64], len: 0 }
    }

    pub fn format_tokens(&mut self, tokens: u32) -> &str {
        self.len = 0;
        if tokens >= 1_000_000 {
            write!(self, "{:.1}M", tokens as f64 / 1_000_000.0).ok();
        } else if tokens >= 1_000 {
            write!(self, "{:.1}K", tokens as f64 / 1_000.0).ok();
        } else {
            write!(self, "{}", tokens).ok();
        }
        std::str::from_utf8(&self.buf[..self.len]).unwrap_or("")
    }
}

impl Write for FormatBuf {
    fn write_str(&mut self, s: &str) -> std::fmt::Result {
        let bytes = s.as_bytes();
        let remaining = self.buf.len() - self.len;
        let to_copy = bytes.len().min(remaining);
        self.buf[self.len..self.len + to_copy].copy_from_slice(&bytes[..to_copy]);
        self.len += to_copy;
        Ok(())
    }
}
```

### Pattern 4: AnimationState Machine

```rust
/// Animation state for TaskBox widgets
#[derive(Debug, Clone, Copy, Default)]
pub struct AnimationState {
    /// Global frame counter (wraps at 65536)
    pub frame: u16,
    /// Spinner frame (0-7 for braille)
    pub spinner_frame: u8,
    /// Decrypt reveal progress (0.0 - 1.0)
    pub reveal_progress: f32,
    /// Pulse phase (0.0 - 1.0)
    pub pulse_phase: f32,
    /// Whether animation is active
    pub active: bool,
}

impl AnimationState {
    /// Advance animation by one frame (~16ms at 60 FPS)
    pub fn tick(&mut self) {
        self.frame = self.frame.wrapping_add(1);
        self.spinner_frame = (self.frame / 2) as u8 % 8;
        self.pulse_phase = ((self.frame % 60) as f32) / 60.0;

        if self.active && self.reveal_progress < 1.0 {
            self.reveal_progress = (self.reveal_progress + 0.02).min(1.0);
        }
    }

    /// Reset for new content
    pub fn reset(&mut self) {
        self.reveal_progress = 0.0;
        self.active = true;
    }
}

/// Braille spinner chars (8 frames)
pub const BRAILLE_SPINNER: [char; 8] = ['⣾', '⣽', '⣻', '⢿', '⡿', '⣟', '⣯', '⣷'];
```

### Pattern 5: LayoutConstraints for Adaptive Rendering

```rust
/// Layout constraints for responsive rendering
#[derive(Debug, Clone, Copy)]
pub struct LayoutConstraints {
    /// Available width
    pub width: u16,
    /// Available height
    pub height: u16,
    /// Render mode
    pub mode: RenderMode,
    /// Whether terminal supports Unicode
    pub unicode: bool,
    /// Whether terminal supports 256 colors
    pub colors_256: bool,
}

impl LayoutConstraints {
    pub fn from_area(area: Rect) -> Self {
        Self {
            width: area.width,
            height: area.height,
            mode: if area.height < 6 {
                RenderMode::Compact
            } else if area.height < 20 {
                RenderMode::Expanded
            } else {
                RenderMode::Full
            },
            unicode: true,  // Assume modern terminal
            colors_256: true,
        }
    }

    /// Check if we should show section
    pub fn can_show_section(&self, min_lines: u16) -> bool {
        self.height >= min_lines
    }

    /// Get max content width (excluding borders)
    pub fn content_width(&self) -> u16 {
        self.width.saturating_sub(4)  // 2 border + 2 padding
    }
}
```

```rust
fn test_render_no_allocation() {
    use std::alloc::{GlobalAlloc, Layout, System};
    use std::sync::atomic::{AtomicUsize, Ordering};

    static ALLOCATIONS: AtomicUsize = AtomicUsize::new(0);

    struct CountingAlloc;
    unsafe impl GlobalAlloc for CountingAlloc {
        unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
            ALLOCATIONS.fetch_add(1, Ordering::SeqCst);
            System.alloc(layout)
        }
        unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
            System.dealloc(ptr, layout)
        }
    }

    // Setup
    let box_ = InferBox::new("claude-sonnet").with_state(BoxState::running());
    let area = Rect::new(0, 0, 80, 10);
    let mut buf = Buffer::empty(area);

    // Warm up caches
    (&box_).render(area, &mut buf);

    // Measure
    ALLOCATIONS.store(0, Ordering::SeqCst);
    (&box_).render(area, &mut buf);
    let allocs = ALLOCATIONS.load(Ordering::SeqCst);
```

```rust
// src/tui/widgets/task_box/effects.rs

//! Visual effects library for TaskBox widgets
//!
//! Provides reusable animation primitives:
//! - Decrypt/reveal effect
//! - Progress bars
//! - Sparklines
//! - Cursor blink
//! - Glitch animations

/// Characters for decrypt effect: ghost → light → medium → reveal
pub const DECRYPT_CHARS: [char; 4] = ['░', '▒', '▓', '█'];

/// Frames per character reveal
pub const FRAMES_PER_CHAR: usize = 4;

/// Progressive text reveal effect
///
/// Simulates "decryption" by revealing characters one at a time
/// with a ghost → solid transition animation.
#[derive(Debug, Clone)]
pub struct DecryptEffect {
    /// Text to reveal
    text: String,
    /// Number of fully revealed characters
    revealed_chars: usize,
    /// Current animation frame (0-3 within current char)
    frame: usize,
    /// Whether reveal is complete
    complete: bool,
}

impl DecryptEffect {
    /// Create new effect for text
    pub fn new(text: impl Into<String>) -> Self {
        let text = text.into();
        let complete = text.is_empty();
        Self {
            text,
            revealed_chars: 0,
            frame: 0,
            complete,
        }
    }

    /// Advance animation by one frame
    pub fn tick(&mut self) {
        if self.complete {
            return;
        }

        self.frame += 1;
        if self.frame >= FRAMES_PER_CHAR {
            self.frame = 0;
            self.revealed_chars += 1;
            if self.revealed_chars >= self.text.chars().count() {
                self.complete = true;
            }
        }
    }

    /// Check if reveal is complete
    pub fn is_complete(&self) -> bool {
        self.complete
    }

    /// Render current state as string
    pub fn render(&self) -> String {
        let mut result = String::with_capacity(self.text.len());

        for (i, c) in self.text.chars().enumerate() {
            if i < self.revealed_chars {
                // Fully revealed
                result.push(c);
            } else if i == self.revealed_chars && !self.complete {
                // Currently revealing - show animation frame
                result.push(DECRYPT_CHARS[self.frame]);
            } else {
                // Not yet revealed - ghost
                result.push(DECRYPT_CHARS[0]);
            }
        }

        result
    }

    /// Force complete reveal
    pub fn complete(&mut self) {
        self.revealed_chars = self.text.chars().count();
        self.complete = true;
    }

    /// Reset to initial state
    pub fn reset(&mut self) {
        self.revealed_chars = 0;
        self.frame = 0;
        self.complete = self.text.is_empty();
    }

    /// Update text (resets animation)
    pub fn set_text(&mut self, text: impl Into<String>) {
        self.text = text.into();
        self.reset();
    }

    /// Append text (continues animation from current position)
    pub fn append(&mut self, text: &str) {
        self.text.push_str(text);
        self.complete = false;
    }
}
```

```rust
/// Sparkline characters (8 levels)
pub const SPARKLINE_CHARS: [char; 8] = ['▁', '▂', '▃', '▄', '▅', '▆', '▇', '█'];

/// Sparkline renderer for time-series data
pub struct Sparkline;

impl Sparkline {
    /// Render sparkline from values (0.0 - 1.0 normalized)
    pub fn render(values: &[f32], max_width: usize) -> String {
        if values.is_empty() {
            return String::new();
        }

        let take_count = values.len().min(max_width);
        let skip_count = values.len().saturating_sub(max_width);

        values
            .iter()
            .skip(skip_count)
            .take(take_count)
            .map(|&v| {
                let v = v.clamp(0.0, 1.0);
                let idx = (v * 7.0).round() as usize;
                SPARKLINE_CHARS[idx.min(7)]
            })
            .collect()
    }

    /// Render from raw values (auto-normalizes)
    pub fn render_auto(values: &[f32], max_width: usize) -> String {
        if values.is_empty() {
            return String::new();
        }

        let max = values.iter().cloned().fold(f32::MIN, f32::max);
        let min = values.iter().cloned().fold(f32::MAX, f32::min);
        let range = max - min;

        if range < f32::EPSILON {
            // All values same - render as middle
            return SPARKLINE_CHARS[4].to_string().repeat(values.len().min(max_width));
        }

        let normalized: Vec<f32> = values.iter().map(|v| (v - min) / range).collect();
        Self::render(&normalized, max_width)
    }
}
```

```rust
        }
    }
```

```rust
    pub fn with_duration(frames: usize) -> Self {
        Self {
            duration: frames,
            frame: frames,
        }
    }

    /// Trigger the effect (reset to start)
    pub fn trigger(&mut self) {
        self.frame = self.duration;
    }

    /// Advance animation
    pub fn tick(&mut self) {
        if self.frame > 0 {
            self.frame -= 1;
        }
    }

    /// Check if effect is active
    pub fn is_active(&self) -> bool {
        self.frame > 0
    }

    /// Get horizontal offset for current frame
    pub fn offset(&self) -> i8 {
        if !self.is_active() {
            return 0;
        }

        // Shake pattern: 0, -1, -1, 1, 1, 0
        match self.frame % 6 {
            5 | 0 => 0,
            4 | 3 => -1,
            2 | 1 => 1,
            _ => 0,
        }
    }

    /// Get intensity (1.0 at start, 0.0 when done)
    pub fn intensity(&self) -> f32 {
        if self.duration == 0 {
            return 0.0;
        }
        self.frame as f32 / self.duration as f32
    }
}

    fn default() -> Self {
        Self::new()
    }
```

```rust
        Self {
            cycle_frames: 30,
            min_brightness: 0.85,
            max_brightness: 1.0,
        }

    /// Create strong pulse (15 frame cycle, 50-100%)
    pub fn strong() -> Self {
        Self {
            cycle_frames: 15,
            min_brightness: 0.5,
            max_brightness: 1.0,
        }
    }

    /// Get brightness multiplier at given frame
    pub fn brightness(&self, frame: usize) -> f32 {
        let progress = (frame % self.cycle_frames) as f32 / self.cycle_frames as f32;
        let sine = (progress * std::f32::consts::PI * 2.0).sin();
        let normalized = (sine + 1.0) / 2.0; // 0.0 - 1.0
        self.min_brightness + (self.max_brightness - self.min_brightness) * normalized
    }

        let brightness = self.brightness(frame);
        match color {
            Color::Rgb(r, g, b) => Color::Rgb(
                (r as f32 * brightness) as u8,
                (g as f32 * brightness) as u8,
                (b as f32 * brightness) as u8,
            ),
            other => other,
        }
    }
}
```

```rust
/// Status of a TaskBox execution
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]

pub struct BoxState {
    pub status: BoxStatus,
    pub started_at: Option<Instant>,
    pub completed_at: Option<Instant>,
    pub error_message: Option<String>,
}

impl Default for BoxState {
    fn default() -> Self {
        Self::queued()
    }
}

impl BoxState {
    // === Constructors ===

    pub fn queued() -> Self {
        Self {
            status: BoxStatus::Queued,
            started_at: None,
            completed_at: None,
            error_message: None,
        }
    }

    pub fn running() -> Self {
        Self {
            status: BoxStatus::Running,
            started_at: Some(Instant::now()),
            completed_at: None,
            error_message: None,
        }
    }

    // === State Transitions ===
    //
    // Valid transitions (enforced by assertions in debug mode):
    //
    //   Queued ──► Running ──► Success
    //      │          │
    //      │          └──────► Failed
    //      │
    //      └──────────────────► Skipped
    //
    // INVALID transitions (will panic in debug, no-op in release):
    //   - Success/Failed/Skipped → any state (terminal states)
    //   - Running → Queued (cannot un-start)
    //   - Running → Skipped (must fail or succeed first)

    /// Transition to Running state
    /// Valid from: Queued
    pub fn run(&mut self) {
        debug_assert_eq!(self.status, BoxStatus::Queued, "Can only run from Queued");
        if self.status == BoxStatus::Queued {
            self.status = BoxStatus::Running;
            self.started_at = Some(Instant::now());
        }
    }

    /// Transition to Success state
    /// Valid from: Running
    pub fn complete(&mut self) {
        debug_assert_eq!(self.status, BoxStatus::Running, "Can only complete from Running");
        if self.status == BoxStatus::Running {
            self.status = BoxStatus::Success;
            self.completed_at = Some(Instant::now());
        }
    }

    /// Transition to Failed state with error message
    /// Valid from: Running
    pub fn fail(&mut self, error: impl Into<String>) {
        debug_assert_eq!(self.status, BoxStatus::Running, "Can only fail from Running");
        if self.status == BoxStatus::Running {
            self.status = BoxStatus::Failed;
            self.completed_at = Some(Instant::now());
            self.error_message = Some(error.into());
        }
    }

    /// Transition to Skipped state (dependency failed)
    /// Valid from: Queued
    pub fn skip(&mut self) {
        debug_assert_eq!(self.status, BoxStatus::Queued, "Can only skip from Queued");
        if self.status == BoxStatus::Queued {
            self.status = BoxStatus::Skipped;
        }
    }

    // === Query Methods ===

    /// Check if state is terminal (cannot transition further)
    pub fn is_terminal(&self) -> bool {
        matches!(self.status, BoxStatus::Success | BoxStatus::Failed | BoxStatus::Skipped)
    }

    /// Get elapsed duration since start
    pub fn elapsed(&self) -> Option<Duration> {
        self.started_at.map(|start| {
            self.completed_at
                .map(|end| end.duration_since(start))
                .unwrap_or_else(|| start.elapsed())
        })
    }

    /// Get formatted duration string
    pub fn duration_str(&self) -> String {
        match self.elapsed() {
            Some(d) if d.as_secs() >= 60 => format!("{}m{}s", d.as_secs() / 60, d.as_secs() % 60),
            Some(d) if d.as_millis() >= 1000 => format!("{:.1}s", d.as_secs_f32()),
            Some(d) => format!("{}ms", d.as_millis()),
            None => "—".to_string(),
        }
    }
}
```

```rust
/// Render action bar for TaskBox footer
pub fn render_action_bar(
    actions: &[(&str, &str)],
    area: Rect,
    buf: &mut Buffer,
    style: Style,
) {
    if actions.is_empty() || area.width < 10 {
        return;
    }

    let mut x = area.x + 1;
    let max_x = area.x + area.width - 1;

    // Format: [R] Retry  [C] Copy  [Esc] Close
    for (key, label) in actions {
        let text = format!("[{}] {} ", key, label);
        if x + text.len() as u16 > max_x {
            break;
        }

        // Key in brackets - highlighted
        buf.set_string(x, area.y, "[", style);
        x += 1;
        buf.set_string(x, area.y, *key, style.add_modifier(Modifier::BOLD));
        x += key.len() as u16;
        buf.set_string(x, area.y, "] ", style);
        x += 2;

        // Label
        buf.set_string(x, area.y, format!("{}  ", label), style);
        x += label.len() as u16 + 2;
    }
}
```

```rust
/// Stack-allocated buffer for metrics formatting (64 bytes, no heap)
pub struct FormatBuf {
    buf: [u8; 64],
    len: usize,
}

impl FormatBuf {
    pub const fn new() -> Self {
        Self { buf: [0; 64], len: 0 }
    }

    pub fn format_tokens(&mut self, tokens: u32) -> &str {
        self.len = 0;
        if tokens >= 1_000_000 {
            let _ = write!(self, "{:.1}M", tokens as f64 / 1_000_000.0);
        } else if tokens >= 1_000 {
            let _ = write!(self, "{:.1}K", tokens as f64 / 1_000.0);
        } else {
            let _ = write!(self, "{}", tokens);
        }
        std::str::from_utf8(&self.buf[..self.len]).unwrap_or("")
    }

    pub fn format_duration(&mut self, secs: u64) -> &str {
        self.len = 0;
        if secs >= 3600 {
            let _ = write!(self, "{}h{}m", secs / 3600, (secs % 3600) / 60);
        } else if secs >= 60 {
            let _ = write!(self, "{}m{}s", secs / 60, secs % 60);
        } else {
            let _ = write!(self, "{}s", secs);
        }
        std::str::from_utf8(&self.buf[..self.len]).unwrap_or("")
    }

    pub fn clear(&mut self) {
        self.len = 0;
    }
}

impl Write for FormatBuf {
    fn write_str(&mut self, s: &str) -> std::fmt::Result {
        let bytes = s.as_bytes();
        let remaining = self.buf.len() - self.len;
        let to_copy = bytes.len().min(remaining);
        self.buf[self.len..self.len + to_copy].copy_from_slice(&bytes[..to_copy]);
        self.len += to_copy;
        Ok(())
    }
}

impl Default for FormatBuf {
    fn default() -> Self {
        Self::new()
    }
}

/// Braille spinner characters (8 frames)
pub const BRAILLE_SPINNER: [char; 8] = ['⣾', '⣽', '⣻', '⢿', '⡿', '⣟', '⣯', '⣷'];

/// Animation state for TaskBox widgets
#[derive(Debug, Clone, Copy, Default)]
pub struct AnimationState {
    pub frame: u16,
    pub spinner_frame: u8,
    pub reveal_progress: f32,
    pub pulse_phase: f32,
    pub active: bool,
}

impl AnimationState {
    pub fn tick(&mut self) {
        self.frame = self.frame.wrapping_add(1);
        self.spinner_frame = (self.frame / 2) as u8 % 8;
        self.pulse_phase = ((self.frame % 60) as f32) / 60.0;

        if self.active && self.reveal_progress < 1.0 {
            self.reveal_progress = (self.reveal_progress + 0.02).min(1.0);
        }
    }

    pub fn spinner_char(&self) -> char {
        BRAILLE_SPINNER[self.spinner_frame as usize]
    }

    pub fn reset(&mut self) {
        self.reveal_progress = 0.0;
        self.active = true;
    }
}
```

```rust
/// Layout constraints for responsive rendering
#[derive(Debug, Clone, Copy)]
pub struct LayoutConstraints {
    pub width: u16,
    pub height: u16,
    pub mode: RenderMode,
    pub unicode: bool,
    pub colors_256: bool,
}

impl LayoutConstraints {
    pub fn from_area(area: Rect) -> Self {
        Self {
            width: area.width,
            height: area.height,
            mode: if area.height < 6 {
                RenderMode::Compact
            } else if area.height < 20 {
                RenderMode::Expanded
            } else {
                RenderMode::Full
            },
            unicode: true,
            colors_256: true,
        }
    }

    pub fn can_show_section(&self, min_lines: u16) -> bool {
        self.height >= min_lines
    }

    pub fn content_width(&self) -> u16 {
        self.width.saturating_sub(4)
    }
}

#[cfg(test)]
pub mod test_helpers {
    use ratatui::buffer::Buffer;

    pub fn buffer_to_string(buf: &Buffer) -> String {
        let mut output = String::new();
        for y in 0..buf.area.height {
            for x in 0..buf.area.width {
                output.push_str(buf.get(x, y).symbol());
            }
            output.push('\n');
        }
        output
    }

    pub fn assert_buffer_contains(buf: &Buffer, x: u16, y: u16, expected: &str) {
        let mut actual = String::new();
        for i in 0..expected.len() as u16 {
            actual.push_str(buf.get(x + i, y).symbol());
        }
        assert_eq!(actual, expected, "Buffer at ({}, {}) mismatch", x, y);
    }

    pub fn assert_buffer_line_eq(buf: &Buffer, y: u16, expected: &str) {
        let mut actual = String::new();
        for x in 0..buf.area.width {
            actual.push_str(buf.get(x, y).symbol());
        }
        assert_eq!(actual, expected, "Buffer line {} mismatch", y);
    }
}
```
