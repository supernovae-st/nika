## Overview

InferBox is the visual representation of the `infer:` verb — LLM text generation.

**Key Features:**
- Real-time streaming response with decrypt effect
- Token metrics (input/output/thinking)
- Provider icon detection (Claude 🧠, OpenAI 🤖, Mistral 🌀, etc.)
- Expandable prompt/response sections
- Thinking tokens display (Claude extended thinking)

**Color:** Violet 500 `#8b5cf6`
**Icon:** ⚡ INFER

## Data Structure

```rust
/// InferBox widget data
#[derive(Debug, Clone)]
pub struct InferBox {
    // === Identity ===
    /// Task ID from workflow
    pub task_id: Arc<str>,

    // === Input ===
    /// LLM model name (e.g., "claude-sonnet-4-6")
    pub model: String,
    /// Input prompt (may be multi-line)
    pub prompt: String,
    /// System prompt (optional)
    pub system: Option<String>,
    /// Temperature setting (optional)
    pub temperature: Option<f32>,
    /// Max tokens setting (optional)
    pub max_tokens: Option<u32>,

    // === Output ===
    /// Generated response (streamed incrementally)
    pub response: String,
    /// Input tokens count
    pub tokens_in: u32,
    /// Output tokens count
    pub tokens_out: u32,
    /// Thinking tokens (Claude extended thinking)
    pub thinking_tokens: Option<u32>,
    /// Thinking content (if captured)
    pub thinking_content: Option<String>,

    // === State ===
    /// Current execution state
    pub state: BoxState,
    /// Animation frame counter
    pub frame: usize,

    // === UI State ===
    /// Is prompt section expanded
    pub expanded_prompt: bool,
    /// Is response section expanded
    pub expanded_response: bool,
    /// Is thinking section expanded (if available)
    pub expanded_thinking: bool,
    /// Show streaming cursor during generation
    pub streaming_cursor: bool,

    // === Effects ===
    /// Decrypt effect for response streaming
    pub decrypt_effect: Option<DecryptEffect>,
    /// Border pulse for running state
    pub border_pulse: BorderPulse,
}
```

```rust
impl InferBox {
    /// Append to response (streaming) with decrypt effect
    pub fn append_response(&mut self, text: &str) {
        self.response.push_str(text);

        // Update decrypt effect
        match &mut self.decrypt_effect {
            Some(effect) => effect.append(text),
            None => {
                self.decrypt_effect = Some(DecryptEffect::new(&self.response));
            }
        }
    }

    /// Tick animation frame
    pub fn tick(&mut self) {
        self.frame = self.frame.wrapping_add(1);

        if let Some(effect) = &mut self.decrypt_effect {
            effect.tick();
        }

        // State-specific ticking
        self.state.tick();
    }
}
```

**Step 3: Write implementation**

```rust
impl TaskBoxBehavior for InferBox {
    fn state(&self) -> &BoxState {
        &self.state
    }

    fn state_mut(&mut self) -> &mut BoxState {
        &mut self.state
    }

    fn required_height(&self, mode: RenderMode) -> u16 {
        match mode {
            RenderMode::Compact => {
                let mut height: u16 = 8; // Header + model + prompt section + response section + footer + action bar

                // Add thinking section if present
                if self.has_thinking() {
                    height += 3;
                }

                height
            }
            RenderMode::Expanded => {
                let mut height: u16 = 10; // Base expanded

                // Prompt lines
                let prompt_lines = self.prompt.lines().count().min(10) as u16;
                height += prompt_lines;

                // Response lines
                let response_lines = self.response.lines().count().min(20) as u16;
                height += response_lines.max(2);

                // Thinking section
                if self.has_thinking() && self.expanded_thinking {
                    let thinking_lines = self.thinking_content
                        .as_ref()
                        .map(|t| t.lines().count().min(10))
                        .unwrap_or(0) as u16;
                    height += thinking_lines + 2; // Header + content
                }

                height
            }
            RenderMode::Full => u16::MAX, // Unlimited
        }
    }

    fn is_expanded(&self) -> bool {
        self.expanded_response || self.expanded_prompt || self.expanded_thinking
    }

    fn toggle_expand(&mut self) {
        if self.is_expanded() {
            self.expanded_response = false;
            self.expanded_prompt = false;
            self.expanded_thinking = false;
        } else {
            self.expanded_response = true;
        }
    }

    fn handle_key(&mut self, key: KeyEvent) -> TaskBoxAction {
        match key.code {
            // Copy actions
            KeyCode::Char('c') | KeyCode::Char('C') => {
                TaskBoxAction::Copy(CopyTarget::Result)
            }
            KeyCode::Char('p') | KeyCode::Char('P') => {
                self.toggle_prompt();
                TaskBoxAction::None
            }
            KeyCode::Char('t') | KeyCode::Char('T') if self.has_thinking() => {
                self.toggle_thinking();
                TaskBoxAction::None
            }
            KeyCode::Char('r') | KeyCode::Char('R') if self.state.is_terminal() => {
                TaskBoxAction::Retry
            }
            KeyCode::Char('j') | KeyCode::Char('J') => {
                TaskBoxAction::Copy(CopyTarget::AsJson)
            }
            KeyCode::Char('e') | KeyCode::Char('E') => {
                self.toggle_expand();
                if self.is_expanded() {
                    TaskBoxAction::Expand
                } else {
                    TaskBoxAction::Collapse
                }
            }
            KeyCode::Esc => {
                if self.is_expanded() {
                    self.toggle_expand();
                    TaskBoxAction::Collapse
                } else {
                    TaskBoxAction::None
                }
            }
            KeyCode::Up | KeyCode::Char('k') => TaskBoxAction::NavigatePrev,
            KeyCode::Down | KeyCode::Char('j') if !key.modifiers.contains(KeyModifiers::NONE) => {
                TaskBoxAction::NavigateNext
            }
            KeyCode::Enter => TaskBoxAction::DrillDown,
            _ => TaskBoxAction::None,
        }
    }

    fn available_actions(&self) -> Vec<(&'static str, &'static str)> {
        let mut actions = vec![
            ("E", "Expand"),
            ("C", "Copy"),
        ];

        if self.has_thinking() {
            actions.push(("T", "Think"));
        }

        if self.state.is_terminal() {
            actions.push(("R", "Retry"));
        }

        actions.push(("J", "JSON"));
        actions
    }

    fn tick(&mut self) {
        self.frame = self.frame.wrapping_add(1);

        if let Some(effect) = &mut self.decrypt_effect {
            effect.tick();
        }

        self.state.tick();
    }
}
```

```rust
impl InferBox {
    /// Render in compact mode (Chat view inline)
    pub fn render_compact(&self, area: Rect, buf: &mut Buffer) {
        if area.width < 40 || area.height < 8 {
            return;
        }

        let verb = VerbColor::Infer;
        let border_color = self.state.border_color(verb.rgb());

        // Apply pulse if running
        let border_color = if self.state.is_running() {
            self.border_pulse.apply(border_color, self.frame)
        } else {
            border_color
        };

        let border_style = Style::default().fg(border_color);
        let dim_style = Style::default().fg(Color::Rgb(100, 116, 139));
        let content_style = Style::default().fg(Color::Rgb(226, 232, 240));

        let inner_width = (area.width - 2) as usize;
        let status_icon = self.state.icon();
        let status_suffix = self.state.suffix();

        // Top border
        let title_prefix = format!("╭─ {} ", verb.icon_label());
        let title_suffix = format!(" {} {} ─╮", status_icon, status_suffix);
        let dash_count = inner_width
            .saturating_sub(title_prefix.chars().count())
            .saturating_sub(title_suffix.chars().count());
        let title = format!("{}{}{}", title_prefix, "─".repeat(dash_count), title_suffix);
        buf.set_string(area.x, area.y, &title, border_style);

        let mut y = area.y + 1;

        // Model line
        self.render_model_line(area, buf, &mut y, border_style, dim_style, inner_width);

        // Separator
        self.render_separator(area, buf, &mut y, border_style, inner_width);

        // Prompt section
        self.render_prompt_section(area, buf, &mut y, border_style, dim_style, content_style, inner_width, false);

        // Separator
        self.render_separator(area, buf, &mut y, border_style, inner_width);

        // Response section
        self.render_response_section(area, buf, &mut y, border_style, dim_style, content_style, inner_width, false);

        // Footer with metrics
        self.render_metrics_footer(area, buf, &mut y, border_style, dim_style, inner_width);

        // Action bar
        let actions = self.available_actions();
        self.render_action_bar(area, buf, &mut y, border_style, &actions);

        // Bottom border
        let bottom = format!("╰{}╯", "─".repeat(inner_width));
        buf.set_string(area.x, area.y + area.height - 1, &bottom, border_style);
    }

    fn render_model_line(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        dim_style: Style,
        inner_width: usize,
    ) {
        if *y >= area.y + area.height - 1 {
            return;
        }

        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let icon = self.provider_icon();
        let model_display = truncate(&self.model, inner_width - 12);
        let model_text = format!("model: {} {}", icon, model_display);
        buf.set_string(area.x + 2, *y, &model_text, dim_style);

        *y += 1;
    }

    fn render_prompt_section(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        dim_style: Style,
        content_style: Style,
        inner_width: usize,
        expanded: bool,
    ) {
        if *y >= area.y + area.height - 2 {
            return;
        }

        // Header
        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);
        let expand_icon = if self.expanded_prompt { "[▲]" } else { "[▼]" };
        buf.set_string(area.x + 2, *y, "PROMPT", dim_style);
        buf.set_string(area.x + area.width - 6, *y, expand_icon, dim_style);
        *y += 1;

        // Content
        if *y < area.y + area.height - 2 {
            buf.set_string(area.x, *y, "│", border_style);
            buf.set_string(area.x + area.width - 1, *y, "│", border_style);

            let prompt_display = if expanded {
                self.prompt.lines().next().unwrap_or("").to_string()
            } else {
                truncate(&self.prompt.replace('\n', " "), inner_width - 6)
            };
            buf.set_string(area.x + 2, *y, &format!("┊ {}", prompt_display), content_style);
            *y += 1;
        }
    }

    fn render_response_section(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        dim_style: Style,
        content_style: Style,
        inner_width: usize,
        expanded: bool,
    ) {
        if *y >= area.y + area.height - 3 {
            return;
        }

        // Header
        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let status_text = if self.streaming_cursor && self.state.is_running() {
            "▼ streaming".to_string()
        } else if !self.response.is_empty() {
            format!("▼ {} ch", self.response.len())
        } else {
            String::new()
        };
        buf.set_string(area.x + 2, *y, "RESPONSE", dim_style);
        buf.set_string(area.x + area.width - status_text.len() as u16 - 3, *y, &status_text, dim_style);
        *y += 1;

        // Content
        if *y < area.y + area.height - 3 {
            buf.set_string(area.x, *y, "│", border_style);
            buf.set_string(area.x + area.width - 1, *y, "│", border_style);

            let response_display = if self.response.is_empty() {
                if self.state.is_running() {
                    "┊ ...".to_string()
                } else {
                    "┊ (waiting...)".to_string()
                }
            } else {
                // Use decrypt effect if available
                let text = if let Some(effect) = &self.decrypt_effect {
                    effect.render()
                } else {
                    truncate(&self.response.replace('\n', " "), inner_width - 8)
                };

                let cursor = if self.streaming_cursor && self.state.is_running() {
                    "▌"
                } else {
                    ""
                };
                format!("┊ {}{}", truncate(&text, inner_width - 8), cursor)
            };
            buf.set_string(area.x + 2, *y, &response_display, content_style);
            *y += 1;
        }
    }

    fn render_metrics_footer(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        dim_style: Style,
        inner_width: usize,
    ) {
        if *y >= area.y + area.height - 2 {
            return;
        }

        // Separator
        self.render_separator(area, buf, y, border_style, inner_width);

        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let thinking_str = self.thinking_tokens
            .map(|t| format!(" │ 💭 {}", t))
            .unwrap_or_default();

        let metrics = format!(
            "📊 {} in │ {} out{} │ {} {}",
            self.tokens_in,
            self.tokens_out,
            thinking_str,
            self.provider_icon(),
            truncate(&self.model, 12)
        );
        buf.set_string(area.x + 2, *y, &truncate(&metrics, inner_width - 2), dim_style);
        *y += 1;
    }

    fn render_separator(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        inner_width: usize,
    ) {
        if *y >= area.y + area.height - 1 {
            return;
        }
        let sep = format!("├{}┤", "─".repeat(inner_width));
        buf.set_string(area.x, *y, &sep, border_style);
        *y += 1;
    }

    fn render_action_bar(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        actions: &[(&'static str, &'static str)],
    ) {
        use super::effects::render_action_bar;

        if *y >= area.y + area.height - 2 {
            return;
        }

        // Separator
        self.render_separator(area, buf, y, border_style, (area.width - 2) as usize);

        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let action_area = Rect::new(area.x + 1, *y, area.width - 2, 1);
        render_action_bar(actions, action_area, buf, border_style);
        *y += 1;
    }
}

/// Truncate string preserving UTF-8
fn truncate(s: &str, max_len: usize) -> String {
    let char_count = s.chars().count();
    if char_count > max_len && max_len > 3 {
        let truncated: String = s.chars().take(max_len - 3).collect();
        format!("{}...", truncated)
    } else {
        s.to_string()
    }
}
```

```rust
    pub fn render_expanded(&self, area: Rect, buf: &mut Buffer) {
        if area.width < 50 || area.height < 15 {
            // Fallback to compact if area too small
            return self.render_compact(area, buf);
        }

        let verb = VerbColor::Infer;
        let border_color = self.state.border_color(verb.rgb());
        let border_color = if self.state.is_running() {
            self.border_pulse.apply(border_color, self.frame)
        } else {
            border_color
        };

        let border_style = Style::default().fg(border_color);
        let dim_style = Style::default().fg(Color::Rgb(100, 116, 139));
        let content_style = Style::default().fg(Color::Rgb(226, 232, 240));
        let thinking_style = Style::default().fg(Color::Rgb(168, 85, 247)); // Purple

        let inner_width = (area.width - 2) as usize;
        let status_icon = self.state.icon();
        let status_suffix = self.state.suffix();

        // Top border with full details
        let title_prefix = format!("╭─ {} ", verb.icon_label());
        let title_suffix = format!(" {} {} ─╮", status_icon, status_suffix);
        let dash_count = inner_width
            .saturating_sub(title_prefix.chars().count())
            .saturating_sub(title_suffix.chars().count());

        if self.has_thinking() {
            self.render_thinking_section(area, buf, &mut y, border_style, thinking_style, inner_width);
            self.render_separator(area, buf, &mut y, border_style, inner_width);
        }

        self.render_response_section_expanded(area, buf, &mut y, border_style, dim_style, content_style, inner_width);

        // Fill remaining space
        while y < area.y + area.height - 3 {
            buf.set_string(area.x, y, "│", border_style);
            buf.set_string(area.x + area.width - 1, y, "│", border_style);
            y += 1;

        let actions = self.available_actions();
```

```rust
    /// Token velocity history (for sparkline)
    pub token_history: Vec<u32>,

    /// Record token count for sparkline
    pub fn record_tokens(&mut self, count: u32) {
        self.token_history.push(count);
        // Keep last 20 samples
        if self.token_history.len() > 20 {
            self.token_history.remove(0);
        }
    }

    /// Generate sparkline from token history
    pub fn token_sparkline(&self, width: usize) -> String {
        if self.token_history.is_empty() {
            return String::new();
        }

        let values: Vec<f32> = self.token_history
            .iter()
            .map(|&t| t as f32)
            .collect();

        Sparkline::render_auto(&values, width)
    }
```

```rust
impl InferBox {
    /// Estimate cost based on model and tokens
    pub fn estimated_cost(&self) -> f64 {
        // Pricing per 1K tokens (approximate, varies by model)
        let (input_rate, output_rate) = match self.model.to_lowercase().as_str() {
            m if m.contains("claude-3-5-sonnet") => (0.003, 0.015),
            m if m.contains("claude-3-opus") => (0.015, 0.075),
            m if m.contains("claude-3-haiku") => (0.00025, 0.00125),
            m if m.contains("gpt-4o") => (0.005, 0.015),
            m if m.contains("gpt-4-turbo") => (0.01, 0.03),
            m if m.contains("gpt-3.5") => (0.0005, 0.0015),
            m if m.contains("mistral-large") => (0.004, 0.012),
            m if m.contains("deepseek") => (0.0001, 0.0002),
            _ => (0.001, 0.002), // Default estimate
        };

        let input_cost = (self.tokens_in as f64 / 1000.0) * input_rate;
        let output_cost = (self.tokens_out as f64 / 1000.0) * output_rate;

        input_cost + output_cost
    }

    /// Format cost for display
    pub fn cost_display(&self) -> String {
        let cost = self.estimated_cost();
        if cost < 0.001 {
            format!("$<0.001")
        } else if cost < 1.0 {
            format!("${:.4}", cost)
        } else {
            format!("${:.2}", cost)
        }
    }
}
```

```rust
use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn bench_infer_render_compact(c: &mut Criterion) {
```
