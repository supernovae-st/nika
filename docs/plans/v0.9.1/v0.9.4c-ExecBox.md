## Overview

ExecBox is the visual representation of the `exec:` verb — shell command execution.

**Key Features:**
- Real-time stdout/stderr streaming
- Exit code display with semantic coloring
- Process ID and working directory display
- Expandable output sections
- Timeout indicator with countdown
- Signal handling indicators (SIGTERM, SIGKILL)

**Color:** Amber 500 `#f59e0b`
**Icon:** 📟 EXEC

```rust
/// ExecBox widget data
#[derive(Debug, Clone)]
pub struct ExecBox {
    // === Identity ===
    /// Task ID from workflow
    pub task_id: Arc<str>,

    // === Input ===
    /// Shell command to execute
    pub command: String,
    /// Working directory (optional)
    pub cwd: Option<String>,
    /// Environment variables override (optional)
    pub env: Option<HashMap<String, String>>,
    /// Timeout in seconds (optional)
    pub timeout_secs: Option<u64>,

    // === Output ===
    /// Standard output (streamed)
    pub stdout: String,
    /// Standard error (streamed)
    pub stderr: String,
    /// Exit code (None if still running)
    pub exit_code: Option<i32>,
    /// Process ID
    pub pid: Option<u32>,
    /// Signal received (if killed)
    pub signal: Option<String>,

    // === State ===
    /// Current execution state
    pub state: BoxState,
    /// Animation frame counter
    pub frame: usize,
    /// Elapsed time in milliseconds
    pub elapsed_ms: u64,

    // === UI State ===
    /// Is stdout section expanded
    pub expanded_stdout: bool,
    /// Is stderr section expanded
    pub expanded_stderr: bool,
    /// Auto-scroll stdout to bottom
    pub auto_scroll: bool,

    // === Effects ===
    /// Border pulse for running state
    pub border_pulse: BorderPulse,
    /// Glitch shake for errors
    pub glitch_shake: GlitchShake,
}
```

## Exit Code Color Mapping

```rust
/// Get color for exit code
pub fn exit_code_color(code: i32) -> Color {
    match code {
        0 => Color::Rgb(34, 197, 94),      // Green 500 - success
        1 => Color::Rgb(239, 68, 68),      // Red 500 - general error
        2 => Color::Rgb(239, 68, 68),      // Red 500 - misuse of command
        126 => Color::Rgb(249, 115, 22),   // Orange 500 - permission denied
        127 => Color::Rgb(249, 115, 22),   // Orange 500 - command not found
        128..=159 => Color::Rgb(234, 179, 8), // Yellow 500 - signal (128+N)
        _ => Color::Rgb(239, 68, 68),      // Red 500 - other errors
    }
}

/// Get signal name from exit code (128+N pattern)
pub fn signal_name(code: i32) -> Option<&'static str> {
    match code {
        130 => Some("SIGINT (Ctrl+C)"),
        137 => Some("SIGKILL"),
        143 => Some("SIGTERM"),
        139 => Some("SIGSEGV"),
        134 => Some("SIGABRT"),
        _ if code > 128 => Some("Unknown signal"),
        _ => None,
    }
}
```

```rust
impl TaskBoxBehavior for ExecBox {
    fn state(&self) -> &BoxState {
        &self.state
    }

    fn state_mut(&mut self) -> &mut BoxState {
        &mut self.state
    }

    fn required_height(&self, mode: RenderMode) -> u16 {
        match mode {
            RenderMode::Compact => {
                let mut height: u16 = 6; // Header + command + stdout + footer + action bar

                // Stderr section if present
                if !self.stderr.is_empty() {
                    height += 3;
                }

                height
            }
            RenderMode::Expanded => {
                let mut height: u16 = 8;

                // Stdout lines
                let stdout_lines = if self.expanded_stdout {
                    self.stdout.lines().count().min(20) as u16
                } else {
                    3
                };
                height += stdout_lines;

                // Stderr lines
                if !self.stderr.is_empty() {
                    let stderr_lines = if self.expanded_stderr {
                        self.stderr.lines().count().min(10) as u16
                    } else {
                        2
                    };
                    height += stderr_lines + 2; // Header + content
                }

                height
            }
            RenderMode::Full => u16::MAX,
        }
    }

    fn is_expanded(&self) -> bool {
        self.expanded_stdout || self.expanded_stderr
    }

    fn toggle_expand(&mut self) {
        if self.is_expanded() {
            self.expanded_stdout = false;
            self.expanded_stderr = false;
        } else {
            self.expanded_stdout = true;
            if !self.stderr.is_empty() {
                self.expanded_stderr = true;
            }
        }
    }

    fn handle_key(&mut self, key: KeyEvent) -> TaskBoxAction {
        match key.code {
            KeyCode::Char('c') | KeyCode::Char('C') => {
                // Copy stderr if present and has errors, otherwise stdout
                if !self.stderr.is_empty() && self.exit_code != Some(0) {
                    TaskBoxAction::Copy(CopyTarget::Error)
                } else {
                    TaskBoxAction::Copy(CopyTarget::Result)
                }
            }
            KeyCode::Char('o') | KeyCode::Char('O') => {
                self.expanded_stdout = !self.expanded_stdout;
                TaskBoxAction::None
            }
            KeyCode::Char('e') | KeyCode::Char('E') => {
                if !self.stderr.is_empty() {
                    self.expanded_stderr = !self.expanded_stderr;
                } else {
                    self.toggle_expand();
                }
                TaskBoxAction::None
            }
            KeyCode::Char('s') | KeyCode::Char('S') if self.state.is_running() => {
                // Signal to stop (SIGTERM)
                TaskBoxAction::Copy(CopyTarget::Input) // Placeholder - actual impl sends signal
            }
            KeyCode::Char('k') | KeyCode::Char('K') if self.state.is_running() => {
                // Signal to kill (SIGKILL)
                TaskBoxAction::Copy(CopyTarget::Input) // Placeholder
            }
            KeyCode::Char('r') | KeyCode::Char('R') if self.state.is_terminal() => {
                TaskBoxAction::Retry
            }
            KeyCode::Char('j') | KeyCode::Char('J') => {
                TaskBoxAction::Copy(CopyTarget::AsJson)
            }
            KeyCode::Esc => {
                if self.is_expanded() {
                    self.toggle_expand();
                    TaskBoxAction::Collapse
                } else {
                    TaskBoxAction::None
                }
            }
            _ => TaskBoxAction::None,
        }
    }

    fn available_actions(&self) -> Vec<(&'static str, &'static str)> {
        let mut actions = vec![];

        if self.state.is_running() {
            actions.push(("S", "Stop"));
            actions.push(("K", "Kill"));
        }

        actions.push(("E", "Expand"));
        actions.push(("C", "Copy"));

        if !self.stderr.is_empty() {
            actions.push(("O", "Stdout"));
        }

        if self.state.is_terminal() {
            actions.push(("R", "Retry"));
        }

        actions.push(("J", "JSON"));
        actions
    }

    fn tick(&mut self) {
        self.frame = self.frame.wrapping_add(1);
        self.state.tick();

        // Update elapsed time if running
        if let BoxState::Running { start, .. } = &self.state {
            self.elapsed_ms = start.elapsed().as_millis() as u64;
        }

        // Check for timeout warning
        if self.state.is_running() {
            if let Some(timeout) = self.timeout_secs {
                let remaining = timeout.saturating_sub(self.elapsed_ms / 1000);
                if remaining <= 10 {
                    // Fast pulse when approaching timeout
                    self.border_pulse = BorderPulse::strong();
                }
            }
        }
    }
}
```

```rust
impl ExecBox {
    pub fn render_compact(&self, area: Rect, buf: &mut Buffer) {
        if area.width < 40 || area.height < 6 {
            return;
        }

        let verb = VerbColor::Exec;
        let border_color = self.state.border_color(verb.rgb());

        // Apply effects
        let border_color = if self.state.is_running() {
            self.border_pulse.apply(border_color, self.frame)
        } else {
            border_color
        };

        let border_style = Style::default().fg(border_color);
        let dim_style = Style::default().fg(Color::Rgb(100, 116, 139));
        let content_style = Style::default().fg(Color::Rgb(226, 232, 240));
        let stderr_style = Style::default().fg(Color::Rgb(251, 191, 36)); // Amber

        let inner_width = (area.width - 2) as usize;
        let status_icon = self.state.icon();
        let status_suffix = self.state.suffix();

        // Top border
        let title_prefix = format!("╭─ {} ", verb.icon_label());
        let title_suffix = format!(" {} {} ─╮", status_icon, status_suffix);
        let dash_count = inner_width
            .saturating_sub(title_prefix.chars().count())
            .saturating_sub(title_suffix.chars().count());
        buf.set_string(area.x, area.y, &format!("{}{}{}", title_prefix, "─".repeat(dash_count), title_suffix), border_style);

        let mut y = area.y + 1;

        // Command line
        self.render_command_line(area, buf, &mut y, border_style, content_style, inner_width);

        // Separator
        self.render_separator(area, buf, &mut y, border_style, inner_width);

        // Stdout section
        self.render_stdout_section(area, buf, &mut y, border_style, dim_style, content_style, inner_width, false);

        // Stderr section (if present)
        if !self.stderr.is_empty() {
            self.render_separator(area, buf, &mut y, border_style, inner_width);
            self.render_stderr_section(area, buf, &mut y, border_style, stderr_style, inner_width, false);
        }

        // Footer with exit code and pid
        self.render_footer(area, buf, &mut y, border_style, dim_style, inner_width);

        // Action bar
        let actions = self.available_actions();
        self.render_action_bar(area, buf, &mut y, border_style, &actions, inner_width);

        // Bottom border
        let bottom = format!("╰{}╯", "─".repeat(inner_width));
        buf.set_string(area.x, area.y + area.height - 1, &bottom, border_style);
    }

    fn render_command_line(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        content_style: Style,
        inner_width: usize,
    ) {
        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let cmd_display = truncate(&self.command, inner_width - 4);
        buf.set_string(area.x + 2, *y, &format!("$ {}", cmd_display), content_style);
        *y += 1;
    }

    fn render_stdout_section(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        dim_style: Style,
        content_style: Style,
        inner_width: usize,
        expanded: bool,
    ) {
        // Header
        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let line_count = self.stdout.lines().count();
        let header = format!("STDOUT                                         ▼ {} ln", line_count);
        buf.set_string(area.x + 2, *y, &truncate(&header, inner_width - 2), dim_style);
        *y += 1;

        // Content
        let max_lines = if expanded { 20 } else { 3 };
        let lines: Vec<&str> = if self.auto_scroll {
            self.stdout_tail(max_lines).collect()
        } else {
            self.stdout.lines().take(max_lines).collect()
        };

        for line in lines {
            if *y >= area.y + area.height - 4 {
                break;
            }
            buf.set_string(area.x, *y, "│", border_style);
            buf.set_string(area.x + area.width - 1, *y, "│", border_style);

            let display = truncate(line, inner_width - 6);
            buf.set_string(area.x + 2, *y, &format!("┊ {}", display), content_style);
            *y += 1;
        }

        // Streaming cursor if running
        if self.state.is_running() && *y > area.y + 1 {
            // Would add blinking cursor here
        }
    }

    fn render_stderr_section(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        stderr_style: Style,
        inner_width: usize,
        expanded: bool,
    ) {
        // Header
        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        let line_count = self.stderr.lines().count();
        let icon = if self.exit_code.unwrap_or(0) != 0 { "❌" } else { "⚠️" };
        let header = format!("STDERR {}                                      ▼ {} ln", icon, line_count);
        buf.set_string(area.x + 2, *y, &truncate(&header, inner_width - 2), stderr_style);
        *y += 1;

        // Content
        let max_lines = if expanded { 10 } else { 2 };
        for line in self.stderr.lines().take(max_lines) {
            if *y >= area.y + area.height - 3 {
                break;
            }
            buf.set_string(area.x, *y, "│", border_style);
            buf.set_string(area.x + area.width - 1, *y, "│", border_style);

            let display = truncate(line, inner_width - 6);
            buf.set_string(area.x + 2, *y, &format!("┊ {}", display), stderr_style);
            *y += 1;
        }
    }

    fn render_footer(
        &self,
        area: Rect,
        buf: &mut Buffer,
        y: &mut u16,
        border_style: Style,
        dim_style: Style,
        inner_width: usize,
    ) {
        self.render_separator(area, buf, y, border_style, inner_width);

        buf.set_string(area.x, *y, "│", border_style);
        buf.set_string(area.x + area.width - 1, *y, "│", border_style);

        // Exit code with color
        let exit_str = match self.exit_code {
            Some(0) => "exit: 0 ✓".to_string(),
            Some(code) => {
                let signal = signal_name(code).unwrap_or("");
                if signal.is_empty() {
                    format!("exit: {} ✗", code)
                } else {
                    format!("exit: {} ({})", code, signal)
                }
            }
            None => "exit: ?".to_string(),
        };

        let pid_str = self.pid.map(|p| format!(" │ pid: {}", p)).unwrap_or_default();
        let cwd_str = self.cwd.as_ref().map(|c| {
            let display = if c.len() > 20 {
                format!("...{}", &c[c.len()-17..])
            } else {
                c.clone()
            };
            format!(" │ cwd: {}", display)
        }).unwrap_or_default();

        let timeout_str = if self.state.is_running() {
            self.timeout_secs.map(|t| {
                let remaining = t.saturating_sub(self.elapsed_ms / 1000);
                format!(" │ timeout: {}s ({}s left)", t, remaining)
            }).unwrap_or_default()
        } else {
            String::new()
        };

        let footer = format!("{}{}{}{}", exit_str, pid_str, cwd_str, timeout_str);

        // Color exit code part
        let exit_color = self.exit_code.map(exit_code_color).unwrap_or(Color::Rgb(100, 116, 139));
        buf.set_string(area.x + 2, *y, &truncate(&footer, inner_width - 2), Style::default().fg(exit_color));
        *y += 1;
    }
}
```

```rust
impl ExecBox {
    /// Get remaining timeout seconds
    pub fn timeout_remaining(&self) -> Option<u64> {
        self.timeout_secs.map(|t| t.saturating_sub(self.elapsed_ms / 1000))
    }

    /// Check if approaching timeout (< 10s)
    pub fn is_timeout_warning(&self) -> bool {
        self.timeout_remaining().map(|r| r <= 10).unwrap_or(false)

        self.timeout_remaining() == Some(0)
```
