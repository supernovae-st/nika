## Overview

FetchBox is the visual representation of the `fetch:` verb — HTTP requests.

**Key Features:**
- HTTP method badge (GET, POST, PUT, DELETE, PATCH)
- Status code with semantic coloring (2xx green, 4xx amber, 5xx red)
- Request/response body display with JSON formatting
- TTFB (time to first byte) metric
- Retry indicator with exponential backoff display
- Header masking for sensitive values (Authorization, API keys)
- cURL export command

**Color:** Cyan 500 `#06b6d4`
**Icon:** 🛰️ FETCH

```rust
/// FetchBox widget data
#[derive(Debug, Clone)]
pub struct FetchBox {
    // === Identity ===
    /// Task ID from workflow
    pub task_id: Arc<str>,

    // === Request ===
    /// HTTP method (GET, POST, PUT, DELETE, PATCH)
    pub method: String,
    /// Request URL
    pub url: String,
    /// Request headers
    pub request_headers: HashMap<String, String>,
    /// Request body (if any)
    pub request_body: Option<String>,
    /// Content-Type for request
    pub content_type: Option<String>,

    // === Response ===
    /// HTTP status code
    pub status_code: Option<u16>,
    /// Response headers
    pub response_headers: HashMap<String, String>,
    /// Response body
    pub response_body: Option<String>,
    /// Response Content-Type
    pub response_content_type: Option<String>,
    /// Response size in bytes
    pub response_size: u64,

    // === Timing ===
    /// Time to first byte (ms)
    pub ttfb_ms: u64,
    /// Total request time (ms)
    pub total_time_ms: u64,
    /// DNS lookup time (ms)
    pub dns_ms: Option<u64>,
    /// TLS handshake time (ms)
    pub tls_ms: Option<u64>,

    // === Retry ===
    /// Current retry count
    pub retries: u32,
    /// Max retries configured
    pub max_retries: Option<u32>,
    /// Next retry delay (ms) if retrying
    pub retry_delay_ms: Option<u64>,

    // === State ===
    /// Current execution state
    pub state: BoxState,
    /// Animation frame counter
    pub frame: usize,

    // === UI State ===
    /// Is request section expanded
    pub expanded_request: bool,
    /// Is response section expanded
    pub expanded_response: bool,
    /// Show response as formatted JSON
    pub json_formatted: bool,

    // === Effects ===
    /// Border pulse for running state
    pub border_pulse: BorderPulse,
}
```

## HTTP Status Color Mapping

```rust
/// Get color for HTTP status code
pub fn status_color(code: u16) -> Color {
    match code {
        100..=199 => Color::Rgb(148, 163, 184), // Slate 400 - informational
        200..=299 => Color::Rgb(34, 197, 94),   // Green 500 - success
        300..=399 => Color::Rgb(59, 130, 246),  // Blue 500 - redirect
        400..=499 => Color::Rgb(245, 158, 11),  // Amber 500 - client error
        500..=599 => Color::Rgb(239, 68, 68),   // Red 500 - server error
        _ => Color::Rgb(100, 116, 139),          // Unknown
    }
}

/// Get text for HTTP status code

    match code {
        200 => "OK",
        201 => "Created",
        204 => "No Content",

        304 => "Not Modified",
        400 => "Bad Request",
        401 => "Unauthorized",
        403 => "Forbidden",
        404 => "Not Found",
        405 => "Method Not Allowed",
        408 => "Request Timeout",
        422 => "Unprocessable Entity",
        429 => "Too Many Requests",
        500 => "Internal Server Error",
        502 => "Bad Gateway",
        503 => "Service Unavailable",
        504 => "Gateway Timeout",
        _ => "Unknown",
    }
}
```

### Task 3.2: Implement TaskBoxBehavior trait

```rust
impl TaskBoxBehavior for FetchBox {
    fn state(&self) -> &BoxState { &self.state }
    fn state_mut(&mut self) -> &mut BoxState { &mut self.state }

    fn required_height(&self, mode: RenderMode) -> u16 {
        match mode {
            RenderMode::Compact => {
                let mut height: u16 = 8;
                if self.request_body.is_some() { height += 1; }
                if self.retries > 0 { height += 1; }
                height
            }
            RenderMode::Expanded => {
                let mut height: u16 = 10;
                height += self.request_headers.len().min(5) as u16;
                if self.request_body.is_some() {
                    height += self.request_body.as_ref()
                        .map(|b| b.lines().count().min(10))
                        .unwrap_or(0) as u16;
                }
                if let Some(body) = &self.response_body {
                    height += body.lines().count().min(15) as u16;
                }
                height
            }
            RenderMode::Full => u16::MAX,
        }
    }

    fn handle_key(&mut self, key: KeyEvent) -> TaskBoxAction {
        match key.code {
            KeyCode::Char('c') | KeyCode::Char('C') => {
                TaskBoxAction::Copy(CopyTarget::Result)
            }
            KeyCode::Char('u') | KeyCode::Char('U') => {
                TaskBoxAction::Copy(CopyTarget::Input) // URL
            }
            KeyCode::Char('x') | KeyCode::Char('X') => {
                TaskBoxAction::Copy(CopyTarget::AsCurl)
            }
            KeyCode::Char('f') | KeyCode::Char('F') => {
                self.json_formatted = !self.json_formatted;
                TaskBoxAction::None
            }
            KeyCode::Char('r') | KeyCode::Char('R') if self.state.is_terminal() => {
                TaskBoxAction::Retry
            }
            KeyCode::Char('j') | KeyCode::Char('J') => {
                TaskBoxAction::Copy(CopyTarget::AsJson)
            }
            _ => TaskBoxAction::None,
        }
    }

    fn available_actions(&self) -> Vec<(&'static str, &'static str)> {
        let mut actions = vec![
            ("E", "Expand"),
            ("C", "Copy"),
            ("U", "URL"),
            ("X", "cURL"),
        ];

        if self.response_content_type.as_ref()
            .map(|ct| ct.contains("json"))
            .unwrap_or(false)
        {
            actions.push(("F", "Format"));
        }

        if self.state.is_terminal() {
            actions.push(("R", "Retry"));
        }

        actions
    }

    fn tick(&mut self) {
        self.frame = self.frame.wrapping_add(1);
        self.state.tick();
    }

    fn is_expanded(&self) -> bool {
        self.expanded_request || self.expanded_response
    }

    fn toggle_expand(&mut self) {
        if self.is_expanded() {
            self.expanded_request = false;
            self.expanded_response = false;
        } else {
            self.expanded_response = true;
        }
    }
}
```

```rust
impl FetchBox {
    /// Generate cURL command
    pub fn to_curl(&self) -> String {
        let mut parts = vec!["curl".to_string()];

        // Method
        if self.method != "GET" {
            parts.push(format!("-X {}", self.method));
        }

        // Headers
        for (key, value) in &self.request_headers {
            let masked = Self::mask_header_value(key, value);
            parts.push(format!("-H '{}: {}'", key, masked));
        }

        // Body
        if let Some(body) = &self.request_body {
            // Escape single quotes in body
            let escaped = body.replace('\'', "'\\''");
            parts.push(format!("-d '{}'", escaped));
        }

        // URL (quoted)
        parts.push(format!("'{}'", self.url));

        parts.join(" \\\n  ")
    }

    /// Generate cURL with actual values (for clipboard)
    pub fn to_curl_unmasked(&self) -> String {
        let mut parts = vec!["curl".to_string()];

        if self.method != "GET" {
            parts.push(format!("-X {}", self.method));
        }

        for (key, value) in &self.request_headers {
            parts.push(format!("-H '{}: {}'", key, value.replace('\'', "'\\''")));
        }

        if let Some(body) = &self.request_body {
            parts.push(format!("-d '{}'", body.replace('\'', "'\\''")));
        }

        parts.push(format!("'{}'", self.url));
        parts.join(" \\\n  ")
    }
}
```

```rust
    pub fn response_formatted(&self) -> Option<String> {
        self.response_body.as_ref().and_then(|body| {
            if self.json_formatted {
                serde_json::from_str::<serde_json::Value>(body)

                    .map(|v| serde_json::to_string_pretty(&v).unwrap_or_else(|_| body.clone()))

                Some(body.clone())
            }
        })
```

```rust
    /// Get retry progress for display
    pub fn retry_progress(&self) -> Option<(u32, u32, u64)> {
        if self.retries > 0 && self.retry_delay_ms.is_some() {
            Some((
                self.retries,
                self.max_retries.unwrap_or(3),
                self.retry_delay_ms.unwrap_or(0)
            ))
        } else {
            None
        }
    }

    /// Render retry progress bar
    pub fn retry_progress_display(&self) -> String {
        if let Some((current, max, delay_ms)) = self.retry_progress() {
            format!(
                "🔄 Retry {}/{} │ waiting {}s",
                current, max, delay_ms / 1000
            )
        } else {
            String::new()
        }
    }
```

```rust
use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn bench_fetch_render_compact(c: &mut Criterion) {
    let box_ = FetchBox::new("POST", "https://api.example.com/data")
        .with_status(201)
        .with_response_body(r#"{"id": 123}"#)
        .with_ttfb(89)
        .with_total_time(150);

    let area = Rect::new(0, 0, 60, 12);
    let mut buffer = Buffer::empty(area);

    c.bench_function("fetch_render_compact", |b| {
        b.iter(|| {
            box_.render_compact(black_box(area), &mut buffer);
        })
    });
}

fn bench_fetch_status_color(c: &mut Criterion) {
    c.bench_function("status_color", |b| {
        b.iter(|| {
            black_box(status_color(200));
            black_box(status_color(404));
            black_box(status_color(500));
        })
    });
}

fn bench_fetch_to_curl(c: &mut Criterion) {
    let box_ = FetchBox::new("POST", "https://api.example.com")
        .with_header("Content-Type", "application/json")
        .with_request_body(r#"{"key": "value"}"#);

    c.bench_function("fetch_to_curl", |b| {
        b.iter(|| {
            black_box(box_.to_curl());
        })
    });
}

criterion_group!(benches, bench_fetch_render_compact, bench_fetch_status_color, bench_fetch_to_curl);
criterion_main!(benches);
```
