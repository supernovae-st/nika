```rust
/// InvokeBox — MCP tool call visualization
#[derive(Debug, Clone)]
pub struct InvokeBox {
    // === Identity ===
    /// Tool name (e.g., "novanet_describe")
    pub tool: String,

    // === Timing ===
    /// Time to first byte (ms)
    pub ttfb_ms: Option<u64>,
    /// Total duration (ms)
    pub duration_ms: Option<u64>,
    /// Timeout setting (ms)
    pub timeout_ms: u64,
    /// Number of retries attempted
    pub retry_count: u32,

    // === UI State ===
    pub expanded_params: bool,
    pub expanded_result: bool,
    pub render_mode: RenderMode,

    // === PERF: JSON Cache (v0.8.1) ===
    params_oneline_cached: Option<String>,
    params_pretty_cached: Option<String>,
    result_oneline_cached: Option<String>,
    result_pretty_cached: Option<String>,
}

/// Structured MCP error information
#[derive(Debug, Clone)]
pub struct InvokeError {
    /// Error message
    pub message: String,
    /// Nika error code (e.g., "NIKA-102")
    pub code: String,
    /// JSON-RPC error code if available
    pub jsonrpc_code: Option<i32>,
    /// Hint for resolution
    pub hint: Option<String>,
}

impl InvokeError {
    pub fn new(message: impl Into<String>, code: impl Into<String>) -> Self {
        Self {
            message: message.into(),
            code: code.into(),
            jsonrpc_code: None,
            hint: None,
        }
    }

    pub fn with_jsonrpc(mut self, code: i32) -> Self {
        self.jsonrpc_code = Some(code);
        self
    }

    pub fn with_hint(mut self, hint: impl Into<String>) -> Self {
        self.hint = Some(hint.into());
        self
    }

    /// Format JSON-RPC code for display
    pub fn jsonrpc_label(&self) -> Option<&'static str> {
        self.jsonrpc_code.map(|code| match code {
            -32700 => "ParseError",
            -32600 => "InvalidRequest",
            -32601 => "MethodNotFound",
            -32602 => "InvalidParams",
            -32603 => "InternalError",
            _ if (-32099..=-32000).contains(&code) => "ServerError",
            _ => "Unknown",
        })
    }
}
```

## Performance Requirements

**From v0.9.4a-TaskBoxFoundation:**

### Mandatory Patterns

| Pattern | Implementation | Budget |
|---------|----------------|--------|
| Widget for &InvokeBox | `impl Widget for &InvokeBox` | 0 clones/render |
| StyleCache | Use `STYLE_CACHE.invoke_border` | <10ns lookup |
| FormatBuf | Format TTFB/total/retries | 0 heap allocs |
| AnimationState | Share animation frame | <1µs/tick |
| JSON Caching | Pre-computed oneline/pretty | 0 parse on render |

### JSON Cache Strategy

```rust
// Already implemented in existing InvokeBox — retain this pattern
pub struct InvokeBox {
    // ... other fields ...

    // PERF: Cache JSON strings to avoid re-serialization on each render
    params_oneline_cached: Option<String>,
    params_pretty_cached: Option<String>,
    result_oneline_cached: Option<String>,
    result_pretty_cached: Option<String>,
}

impl InvokeBox {
    fn ensure_params_cached(&mut self) {
        if self.params_oneline_cached.is_none() {
            self.params_oneline_cached = Some(self.params.to_string());
            self.params_pretty_cached = serde_json::to_string_pretty(&self.params).ok();
        }
    }
}
```

### Memory Bounds

| Component | Limit | Enforcement |
|-----------|-------|-------------|
| params JSON cache | 4KB oneline | Truncate with `...` |
| result JSON cache | 16KB pretty | Truncate with `[truncated]` |
| Error message | 1KB | Truncate |
| Server name | 64 chars | Already enforced by MCP |

### StreamChunk Integration

```rust
// In ChatView (see Foundation spec for full wiring)
StreamChunk::McpCallStart { tool, server, params } => {
    let box_ = InvokeBox::new(&tool, &server)
        .with_params_json(&params)
        .with_state(BoxState::running());
    self.active_boxes.push(TaskBox::Invoke(box_));
}
StreamChunk::McpCallComplete { result } => {
    if let Some(TaskBox::Invoke(ref mut box_)) = self.active_boxes.last_mut() {
        box_.set_result_json(&result);
        box_.state.complete();
    }
}
StreamChunk::McpCallFailed { error } => {
    if let Some(TaskBox::Invoke(ref mut box_)) = self.active_boxes.last_mut() {
        box_.set_error(&error);
        box_.state.fail(&error);
    }
}
```

```rust
impl InvokeBox {
    /// Calculate timeout progress (0.0 to 1.0+)
    pub fn timeout_progress(&self, elapsed_ms: u64) -> f64 {
        if self.timeout_ms == 0 {
            return 0.0;
        }
        elapsed_ms as f64 / self.timeout_ms as f64
    }

    /// Format remaining timeout for display
    pub fn format_timeout_remaining(&self, elapsed_ms: u64) -> String {
        let remaining_ms = self.timeout_ms.saturating_sub(elapsed_ms);
        let remaining_secs = remaining_ms as f64 / 1000.0;
        format!("{:.1}s remaining", remaining_secs)
    }

    /// Check if timeout has been exceeded
    pub fn is_timeout_exceeded(&self, elapsed_ms: u64) -> bool {
        elapsed_ms > self.timeout_ms
    }
}
```

```rust
use super::{TaskBoxBehavior, TaskBoxAction, RenderMode};

impl TaskBoxBehavior for InvokeBox {
    fn verb_icon(&self) -> &'static str {
        "🔌"
    }

    fn verb_label(&self) -> &'static str {
        "INVOKE"
    }

    fn verb_color(&self) -> (u8, u8, u8) {
        (16, 185, 129) // Emerald
    }

    fn state(&self) -> &BoxState {
        &self.state
    }

    fn state_mut(&mut self) -> &mut BoxState {
        &mut self.state
    }

    fn render_mode(&self) -> RenderMode {
        self.render_mode
    }

    fn set_render_mode(&mut self, mode: RenderMode) {
        self.render_mode = mode;
    }

    fn required_height(&self) -> u16 {
        let base = match self.render_mode {
            RenderMode::Compact => 8,
            RenderMode::Expanded => 15,
            RenderMode::Full => 25,
        };

        let params_extra = if self.expanded_params && !self.params.is_null() {
            self.params_pretty_cached
                .as_ref()
                .map(|s| s.lines().count().min(10))
                .unwrap_or(0) as u16
        } else {
            0
        };

        let result_extra = if self.expanded_result && self.result.is_some() {
            self.result_pretty_cached
                .as_ref()
                .map(|s| s.lines().count().min(10))
                .unwrap_or(0) as u16
        } else {
            0
        };

        base + params_extra + result_extra
    }

    fn available_actions(&self) -> Vec<TaskBoxAction> {
        let mut actions = vec![TaskBoxAction::ToggleExpand];

        if self.result.is_some() {
            actions.push(TaskBoxAction::CopyResult);
        }

        if self.state.is_failed() {
            actions.push(TaskBoxAction::Retry);
        }

        actions
    }

    fn handle_action(&mut self, action: TaskBoxAction) -> bool {
        match action {
            TaskBoxAction::ToggleExpand => {
                self.toggle_result();
                true
            }
            TaskBoxAction::ToggleParams => {
                self.toggle_params();
                true
            }
            _ => false,
        }
    }
}
```

```rust
impl Widget for InvokeBox {
    fn render(self, area: Rect, buf: &mut Buffer) {
        // ... existing top border and params rendering ...

        // ERROR section (enhanced with structured error)
        if let Some(ref structured_err) = self.structured_error {
            if y < area.y + area.height - 2 {
                buf.set_string(area.x, y, "│", border_style);
                buf.set_string(area.x + area.width - 1, y, "│", border_style);

                let error_line = format!("❌ {}", structured_err.message);
                buf.set_string(area.x + 2, y, Self::truncate(&error_line, inner_width - 4), error_style);
                y += 1;
            }

            // Error code line
            if y < area.y + area.height - 2 {
                buf.set_string(area.x, y, "│", border_style);
                buf.set_string(area.x + area.width - 1, y, "│", border_style);

                let code_line = if let Some(label) = structured_err.jsonrpc_label() {
                    format!("   Code: {} ({} {})",
                        structured_err.code,
                        label,
                        structured_err.jsonrpc_code.unwrap_or(0)
                    )
                } else {
                    format!("   Code: {}", structured_err.code)
                };
                buf.set_string(area.x + 2, y, &code_line, dim_style);
                y += 1;
            }

            // Hint line
            if let Some(ref hint) = structured_err.hint {
                if y < area.y + area.height - 2 {
                    buf.set_string(area.x, y, "│", border_style);
                    buf.set_string(area.x + area.width - 1, y, "│", border_style);
                    buf.set_string(area.x + 2, y, format!("   Hint: {}", hint), dim_style);
                    y += 1;
                }
            }
        }

        // Timing footer (Expanded mode only)
        if matches!(self.render_mode, RenderMode::Expanded | RenderMode::Full) {
            if y < area.y + area.height - 2 {
                let sep = format!("├{}┤", "─".repeat(inner_width));
                buf.set_string(area.x, y, &sep, border_style);
                y += 1;
            }

            if y < area.y + area.height - 1 {
                buf.set_string(area.x, y, "│", border_style);
                buf.set_string(area.x + area.width - 1, y, "│", border_style);

                let timing = format!(
                    "⏱️  TTFB: {}ms │ Total: {} │ Retries: {}",
                    self.ttfb_ms.unwrap_or(0),
                    self.duration_ms.map(|d| format!("{:.1}s", d as f64 / 1000.0))
                        .unwrap_or_else(|| "—".to_string()),
                    self.retry_count
                );
                buf.set_string(area.x + 2, y, Self::truncate(&timing, inner_width - 2), metric_style);
                y += 1;
            }
        }

        // ... rest of render ...
    }
}
```

```rust
impl InvokeBox {
    /// Export to JSON for clipboard/logging
    pub fn to_json(&self) -> serde_json::Value {
        let mut json = serde_json::json!({
            "tool": self.tool,
            "server": self.server,
            "params": self.params,
            "state": self.state.label(),
        });

        if let Some(ref task_id) = self.task_id {
            json["task_id"] = serde_json::json!(task_id);
        }

        if let Some(ref result) = self.result {
            json["result"] = result.clone();
        }

        if let Some(ref err) = self.structured_error {
            json["error"] = serde_json::json!({
                "message": err.message,
                "code": err.code,
                "jsonrpc_code": err.jsonrpc_code,
                "hint": err.hint,
            });
        }

        json["timing"] = serde_json::json!({
            "ttfb_ms": self.ttfb_ms,
            "duration_ms": self.duration_ms,
            "timeout_ms": self.timeout_ms,
            "retry_count": self.retry_count,
        });

        json
    }
}
```

```rust
fn update_invoke_box(&mut self, event: &McpInvoke) {
```
