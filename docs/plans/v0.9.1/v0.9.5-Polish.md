#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum AnimationState {
    Idle,
    Pulsing,
    Completed,
}

impl NodeBox {
    // Add to struct
    pub animation_state: AnimationState,
    pulse_frame: u8,

    pub fn running(mut self, running: bool) -> Self {
        self.is_running = running;
        self.animation_state = if running {
            AnimationState::Pulsing
        } else {
            AnimationState::Idle
        };
        self
    }

    pub fn tick(&mut self) {
        if self.animation_state == AnimationState::Pulsing {
            self.pulse_frame = (self.pulse_frame + 1) % 60;
        }
    }

    pub fn pulse_intensity(&self) -> f32 {
        if self.animation_state != AnimationState::Pulsing {
            return 0.0;
        }

        // Sine wave pulse: 0.0 to 1.0 over 60 frames
        let t = self.pulse_frame as f32 / 60.0;
        (t * std::f32::consts::PI * 2.0).sin().abs()
    }
}
```

impl EdgeLine {
    // Add to struct
    flow_frame: u8,

    pub fn tick(&mut self) {
        if self.is_active {
            self.flow_frame = (self.flow_frame + 1) % 30;
        }
    }

    pub fn flow_position(&self) -> u16 {
        if !self.is_active {
            return 0;
        }

        let length = self.to_y.saturating_sub(self.from_y);
        let progress = self.flow_frame as u16 * length / 30;
        self.from_y + progress
    }
}

// In render, add flowing dot
impl Widget for EdgeLine {
    fn render(self, _area: Rect, buf: &mut Buffer) {
        // ... existing code ...

        // Draw flowing dot on active edges
        if self.is_active {
            let dot_y = self.flow_position();
            if let Some(cell) = buf.cell_mut((self.x, dot_y)) {
                cell.set_char('●').set_style(Style::default().fg(Color::Cyan));
            }
        }
    }
}

impl ChatView {
    pub fn handle_key(&mut self, key: KeyEvent) -> KeyAction {
        match (key.code, key.modifiers) {

// src/dag/export.rs
use serde::{Serialize, Deserialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SerializableGraph {
    pub nodes: Vec<String>,
    pub edges: Vec<(String, String)>,
}

impl ChatDagPanel {
    pub fn to_serializable(&self) -> SerializableGraph {
        SerializableGraph {
            nodes: self.nodes.iter().map(|(id, _)| id.clone()).collect(),
            edges: self.edges.clone(),
        }
    }
}

pub fn export_json_pretty(panel: &ChatDagPanel) -> String {
    let graph = panel.to_serializable();
    serde_json::to_string_pretty(&graph).unwrap_or_default()
}

pub fn export_mermaid(panel: &ChatDagPanel) -> String {
    let mut output = String::from("graph TD\n");

    // Add nodes with labels
    for (id, kind) in &panel.nodes {
        let icon = match kind {
            NodeKind::UserMessage => "👤",
            NodeKind::AssistantMessage => "🤖",
            NodeKind::ToolCall => "🔧",
            NodeKind::SystemMessage => "⚙️",
        };
        output.push_str(&format!("    {}[\"{} {}\"]\n", id, id, icon));
    }

    // Add edges
    for (from, to) in &panel.edges {
        output.push_str(&format!("    {} --> {}\n", from, to));
    }

    output
}

// In session.rs
#[derive(Serialize, Deserialize)]
pub struct ChatSession {
    pub messages: Vec<SerializedMessage>,
    pub dag: SerializableGraph,
    pub timestamp: DateTime<Utc>,
}

impl ChatView {
    pub fn save_session(&self) {
        if let Some(ref dir) = self.session_dir {
            let session = ChatSession {
                messages: self.messages.iter().map(Into::into).collect(),
                dag: self.dag_panel.as_ref()
                    .map(|p| p.to_serializable())
                    .unwrap_or_default(),
                timestamp: Utc::now(),
            };

            let path = dir.join("current_session.json");
            if let Ok(json) = serde_json::to_string_pretty(&session) {
                let _ = std::fs::write(path, json);
            }
        }
    }

    pub fn restore_session(&mut self) {
        if let Some(ref dir) = self.session_dir {
            let path = dir.join("current_session.json");
            if let Ok(content) = std::fs::read_to_string(path) {
                if let Ok(session) = serde_json::from_str::<ChatSession>(&content) {
                    // Restore messages
                    for msg in session.messages {
                        self.messages.push(msg.into());
                    }
                    // Restore DAG
                    if let Some(ref mut panel) = self.dag_panel {
                        for node in session.dag.nodes {
                            panel.add_node(&node, NodeKind::UserMessage);
                        }
                        for (from, to) in session.dag.edges {
                            panel.add_edge(&from, &to);
                        }
                    }
                }
            }
        }
    }
}
```
