// src/lsp/server.rs

use tower_lsp::{
    jsonrpc::Result,
    lsp_types::*,
    Client, LanguageServer, LspService, Server,
};
use std::sync::Arc;
use tokio::sync::RwLock;

use crate::ast::{raw, analyzer};

pub struct NikaLanguageServer {
    client: Client,
    documents: Arc<RwLock<DocumentStore>>,
}

impl NikaLanguageServer {
    pub fn new(client: Client) -> Self {
        Self {
            client,
            documents: Arc::new(RwLock::new(DocumentStore::new())),
        }
    }

    /// Parse document and publish diagnostics
    async fn analyze_document(&self, uri: &Url, text: &str) {
        let file_id = FileId::from_uri(uri);

        // Phase 1: Parse to Raw AST
        let raw_result = raw::parse(text, file_id);

        let diagnostics = match raw_result {
            Ok(raw_workflow) => {
                // Phase 2: Analyze
                let analyze_result = analyzer::analyze(raw_workflow);
                self.errors_to_diagnostics(&analyze_result.errors)
            }
            Err(parse_error) => {
                vec![self.parse_error_to_diagnostic(&parse_error)]
            }
        };

        self.client
            .publish_diagnostics(uri.clone(), diagnostics, None)
            .await;
    }

    fn errors_to_diagnostics(&self, errors: &[AnalyzeError]) -> Vec<Diagnostic> {
        errors.iter().map(|e| {
            Diagnostic {
                range: self.span_to_range(&e.span),
                severity: Some(match e.severity {
                    Severity::Error => DiagnosticSeverity::ERROR,
                    Severity::Warning => DiagnosticSeverity::WARNING,
                    Severity::Hint => DiagnosticSeverity::HINT,
                }),
                code: Some(NumberOrString::String(e.code.to_string())),
                source: Some("nika".to_string()),
                message: e.message.clone(),
                related_information: None,
                tags: None,
                ..Default::default()
            }
        }).collect()
    }
}

#[tower_lsp::async_trait]
impl LanguageServer for NikaLanguageServer {
    async fn initialize(&self, _: InitializeParams) -> Result<InitializeResult> {
        Ok(InitializeResult {
            capabilities: ServerCapabilities {
                text_document_sync: Some(TextDocumentSyncCapability::Kind(
                    TextDocumentSyncKind::INCREMENTAL,
                )),
                completion_provider: Some(CompletionOptions {
                    trigger_characters: Some(vec![
                        ":".to_string(),  // After verb
                        ".".to_string(),  // Binding path
                        "$".to_string(),  // Implicit output
                        "{".to_string(),  // Template start
                    ]),
                    ..Default::default()
                }),
                hover_provider: Some(HoverProviderCapability::Simple(true)),
                definition_provider: Some(OneOf::Left(true)),
                code_action_provider: Some(CodeActionProviderCapability::Simple(true)),
                ..Default::default()
            },
            ..Default::default()
        })
    }

    async fn initialized(&self, _: InitializedParams) {
        self.client
            .log_message(MessageType::INFO, "Nika LSP initialized!")
            .await;
    }

    async fn shutdown(&self) -> Result<()> {
        Ok(())
    }

    async fn did_open(&self, params: DidOpenTextDocumentParams) {
        let uri = params.text_document.uri;
        let text = params.text_document.text;

        // Store document
        self.documents.write().await.insert(uri.clone(), text.clone());

        // Analyze and publish diagnostics
        self.analyze_document(&uri, &text).await;
    }

    async fn did_change(&self, params: DidChangeTextDocumentParams) {
        let uri = params.text_document.uri;

        // Apply incremental changes
        let text = {
            let mut docs = self.documents.write().await;
            for change in params.content_changes {
                docs.apply_change(&uri, change);
            }
            docs.get(&uri).cloned().unwrap_or_default()
        };

        // Re-analyze
        self.analyze_document(&uri, &text).await;
    }

    async fn completion(&self, params: CompletionParams) -> Result<Option<CompletionResponse>> {
        let uri = params.text_document_position.text_document.uri;
        let position = params.text_document_position.position;

        let docs = self.documents.read().await;
        let text = docs.get(&uri).cloned().unwrap_or_default();

        let completions = self.compute_completions(&text, position).await;
        Ok(Some(CompletionResponse::Array(completions)))
    }

// src/lsp/conversion.rs

use crate::ast::raw::Span;
use tower_lsp::lsp_types::{Position, Range};

/// Convert Nika Span to LSP Range
pub fn span_to_range(span: &Span, source: &str) -> Range {
    let start = offset_to_position(span.start, source);
    let end = offset_to_position(span.end, source);
    Range { start, end }
}

/// Convert byte offset to LSP Position (line, character)
pub fn offset_to_position(offset: usize, source: &str) -> Position {
    let mut line = 0u32;
    let mut col = 0u32;

    for (i, ch) in source.char_indices() {
        if i >= offset {
            break;
        }
        if ch == '\n' {
            line += 1;
            col = 0;
        } else {
            col += 1;
        }
    }

    Position { line, character: col }
}

/// Convert LSP Position to byte offset
pub fn position_to_offset(pos: Position, source: &str) -> usize {
    let mut current_line = 0u32;
    let mut current_col = 0u32;

    for (i, ch) in source.char_indices() {
        if current_line == pos.line && current_col == pos.character {
            return i;
        }
        if ch == '\n' {
            current_line += 1;
            current_col = 0;
        } else {
            current_col += 1;
        }
    }

    source.len()
}

use serde::{Deserialize, Serialize};
use serde_json::Value;
use std::collections::HashMap;

/// Result with named output namespaces
#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct NamespacedResult {
    /// Default output (agent's final response)
    pub output: Value,

    /// Named namespaces emitted during execution
    #[serde(flatten)]
    pub namespaces: HashMap<String, Value>,

    /// Execution metadata
    pub metadata: ResultMetadata,
}

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct ResultMetadata {
    pub turns: u32,
    pub tokens_in: u32,
    pub tokens_out: u32,
    pub duration_ms: u64,
    pub stop_reason: String,
}

impl NamespacedResult {
    pub fn new(output: impl Into<Value>) -> Self {
        Self {
            output: output.into(),
            namespaces: HashMap::new(),
            metadata: ResultMetadata::default(),
        }
    }

    /// Add a named namespace
    pub fn with_namespace(mut self, name: impl Into<String>, value: Value) -> Self {
        self.namespaces.insert(name.into(), value);
        self
    }

    /// Get value by path (e.g., "summary.title" or "artifacts[0]")
    pub fn get_path(&self, path: &str) -> Option<&Value> {
        let parts: Vec<&str> = path.split('.').collect();

        if parts.is_empty() {
            return None;
        }

        // First part is namespace or "output"
        let (first, array_idx) = parse_array_access(parts[0]);

        let mut current = if first == "output" {
            &self.output
        } else {
            self.namespaces.get(first)?
        };

        // Apply array index if present
        if let Some(idx) = array_idx {
            current = current.get(idx)?;
        }

        // Navigate rest of path
        for part in &parts[1..] {
            let (key, array_idx) = parse_array_access(part);
            current = current.get(key)?;
            if let Some(idx) = array_idx {
                current = current.get(idx)?;
            }
        }

        Some(current)
    }
}

/// Parse "name[0]" into ("name", Some(0))
fn parse_array_access(s: &str) -> (&str, Option<usize>) {
    if let Some(bracket_pos) = s.find('[') {
        let name = &s[..bracket_pos];
        let idx_str = &s[bracket_pos + 1..s.len() - 1];
        let idx = idx_str.parse().ok();
        (name, idx)
    } else {
        (s, None)
    }
}

use crate::tools::{FileTool, ToolOutput};
use crate::error::NikaError;
use serde::{Deserialize, Serialize};
use serde_json::{json, Value};
use std::sync::Arc;
use tokio::sync::RwLock;

/// Shared namespace storage for current agent execution
pub type NamespaceStore = Arc<RwLock<std::collections::HashMap<String, Value>>>;

/// Tool for agents to emit named outputs
pub struct EmitOutputTool {
    store: NamespaceStore,
}

#[derive(Debug, Deserialize)]
struct EmitOutputParams {
    /// Namespace name (e.g., "summary", "artifacts")
    name: String,

    /// Value to store in namespace
    value: Value,

    /// Optional: merge with existing namespace (default: replace)
    #[serde(default)]
    merge: bool,
}

impl EmitOutputTool {
    pub fn new(store: NamespaceStore) -> Self {
        Self { store }
    }
}

#[async_trait::async_trait]
impl FileTool for EmitOutputTool {
    fn name(&self) -> &'static str {
        "nika:emit_output"
    }

    fn description(&self) -> &'static str {
        "Emit a named output namespace. Use to structure agent results \
         with multiple named values that downstream tasks can reference."
    }

    fn parameters_schema(&self) -> Value {
        json!({
            "type": "object",
            "properties": {
                "name": {
                    "type": "string",
                    "description": "Namespace name (e.g., 'summary', 'artifacts')"
                },
                "value": {
                    "description": "Value to store (any JSON type)"
                },
                "merge": {
                    "type": "boolean",
                    "description": "Merge with existing namespace instead of replacing",
                    "default": false
                }
            },
            "required": ["name", "value"]
        })
    }

    async fn call(&self, params: Value) -> Result<ToolOutput, NikaError> {
        let params: EmitOutputParams = serde_json::from_value(params)
            .map_err(|e| NikaError::ToolError {
                code: "NIKA-290".to_string(),
                message: format!("Invalid emit_output parameters: {}", e),
            })?;

        // Validate namespace name
        if params.name.is_empty() || params.name == "output" || params.name == "metadata" {
            return Err(NikaError::ToolError {
                code: "NIKA-291".to_string(),
                message: format!(
                    "Invalid namespace name '{}'. Cannot use reserved names: output, metadata",
                    params.name
                ),
            });
        }

        let mut store = self.store.write().await;

        if params.merge {
            // Merge with existing
            if let Some(existing) = store.get_mut(&params.name) {
                if let (Value::Object(existing_obj), Value::Object(new_obj)) =
                    (existing, &params.value)
                {
                    for (k, v) in new_obj {
                        existing_obj.insert(k.clone(), v.clone());
                    }
                } else {
                    // Can't merge non-objects, replace instead
                    store.insert(params.name.clone(), params.value.clone());
                }
            } else {
                store.insert(params.name.clone(), params.value.clone());
            }
        } else {
            // Replace
            store.insert(params.name.clone(), params.value.clone());
        }

        Ok(ToolOutput::success(json!({
            "status": "emitted",
            "namespace": params.name,
        })))
    }
}

use crate::error::NikaError;
use crate::store::namespaced::NamespacedResult;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::path::{Path, PathBuf};
use std::time::{Duration, SystemTime, UNIX_EPOCH};
use tokio::fs;

/// Persistent storage for workflow results
pub struct PersistentStore {
    /// Base directory for store files
    base_dir: PathBuf,
}

#[derive(Debug, Serialize, Deserialize)]
struct StoreEntry {
    /// Task result
    result: NamespacedResult,

    /// When this entry was created
    created_at: u64,

    /// Optional expiry timestamp
    expires_at: Option<u64>,
}

#[derive(Debug, Serialize, Deserialize, Default)]
struct StoreFile {
    /// Version for future migrations
    version: u32,

    /// Task results by task_id
    entries: HashMap<String, StoreEntry>,
}

impl PersistentStore {
    pub fn new(base_dir: impl AsRef<Path>) -> Self {
        Self {
            base_dir: base_dir.as_ref().to_path_buf(),
        }
    }

    /// Initialize store directory
    pub async fn init(&self) -> Result<(), NikaError> {
        fs::create_dir_all(&self.base_dir).await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to create store directory: {}", e),
            }
        })?;
        Ok(())
    }

    /// Get store file path for workflow
    fn store_path(&self, workflow_id: &str, global: bool) -> PathBuf {
        if global {
            self.base_dir.join("global.json")
        } else {
            self.base_dir.join(format!("workflow-{}.json", workflow_id))
        }
    }

    /// Load stored results for workflow
    pub async fn load(&self, workflow_id: &str, global: bool) -> Result<HashMap<String, NamespacedResult>, NikaError> {
        let path = self.store_path(workflow_id, global);

        if !path.exists() {
            return Ok(HashMap::new());
        }

        let content = fs::read_to_string(&path).await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to read store file: {}", e),
            }
        })?;

        let store_file: StoreFile = serde_json::from_str(&content).map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to parse store file: {}", e),
            }
        })?;

        let now = current_timestamp();

        // Filter out expired entries
        let results = store_file
            .entries
            .into_iter()
            .filter(|(_, entry)| {
                entry.expires_at.map(|exp| exp > now).unwrap_or(true)
            })
            .map(|(k, v)| (k, v.result))
            .collect();

        Ok(results)
    }

    /// Save task result
    pub async fn save(
        &self,
        workflow_id: &str,
        task_id: &str,
        result: NamespacedResult,
        ttl: Option<Duration>,
        global: bool,
    ) -> Result<(), NikaError> {
        let path = self.store_path(workflow_id, global);

        // Load existing or create new
        let mut store_file = if path.exists() {
            let content = fs::read_to_string(&path).await.unwrap_or_default();
            serde_json::from_str(&content).unwrap_or_default()
        } else {
            StoreFile::default()
        };

        let now = current_timestamp();
        let expires_at = ttl.map(|d| now + d.as_secs());

        store_file.entries.insert(
            task_id.to_string(),
            StoreEntry {
                result,
                created_at: now,
                expires_at,
            },
        );

        // Atomic write
        let temp_path = path.with_extension("json.tmp");
        let content = serde_json::to_string_pretty(&store_file).map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to serialize store: {}", e),
            }
        })?;

        fs::write(&temp_path, &content).await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to write temp file: {}", e),
            }
        })?;

        fs::rename(&temp_path, &path).await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to finalize store file: {}", e),
            }
        })?;

        Ok(())
    }

    /// Expire old entries across all store files
    pub async fn expire(&self) -> Result<u32, NikaError> {
        let mut expired_count = 0u32;
        let now = current_timestamp();

        let mut dir = fs::read_dir(&self.base_dir).await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to read store directory: {}", e),
            }
        })?;

        while let Some(entry) = dir.next_entry().await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to read directory entry: {}", e),
            }
        })? {
            let path = entry.path();
            if path.extension().map(|e| e == "json").unwrap_or(false) {
                expired_count += self.expire_file(&path, now).await?;
            }
        }

        Ok(expired_count)
    }

    async fn expire_file(&self, path: &Path, now: u64) -> Result<u32, NikaError> {
        let content = fs::read_to_string(path).await.map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to read store file: {}", e),
            }
        })?;

        let mut store_file: StoreFile = serde_json::from_str(&content).map_err(|e| {
            NikaError::StorageError {
                message: format!("Failed to parse store file: {}", e),
            }
        })?;

        let before = store_file.entries.len();
        store_file.entries.retain(|_, entry| {
            entry.expires_at.map(|exp| exp > now).unwrap_or(true)
        });
        let expired = (before - store_file.entries.len()) as u32;

        if expired > 0 {
            let content = serde_json::to_string_pretty(&store_file).unwrap();
            fs::write(path, content).await.ok();
        }

        Ok(expired)
    }
}

fn current_timestamp() -> u64 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap()
        .as_secs()
}

// src/runtime/runner.rs - Add persistence support

impl Runner {
    pub async fn run_with_persistence(
        &mut self,
        workflow: &AnalyzedWorkflow,
    ) -> Result<WorkflowResult, NikaError> {
        let persist_config = workflow.persist.clone().unwrap_or_default();

        // Initialize persistent store if enabled
        let persistent_store = if persist_config.enabled {
            let store = PersistentStore::new(".nika/store");
            store.init().await?;

            // Load cached results
            let cached = store.load(
                &workflow.id,
                persist_config.scope == PersistScope::Global,
            ).await?;

            // Populate datastore with cached results
            for (task_id, result) in cached {
                self.datastore.set_namespaced(&task_id, result);
            }

            Some(store)
        } else {
            None
        };

        // Run workflow
        let result = self.run_internal(workflow).await?;

        // Persist results if enabled
        if let Some(store) = &persistent_store {
            for (task_id, task_result) in &result.task_results {
                if persist_config.should_persist_task(task_id) {
                    store.save(
                        &workflow.id,
                        task_id,
                        task_result.to_namespaced(),
                        persist_config.ttl,
                        persist_config.scope == PersistScope::Global,
                    ).await?;
                }
            }
        }

        Ok(result)
    }
}
